module StepTypes {A : Set} (_~A_ : A → A → Prop)
  {B : A → Set} (_~B_ : ∀ {a₀ a₁} → B a₀ → B a₁ → Prop)
  (ε : EqSys₀ A B)
  (P : (w : 𝒲 A B) → QW-rel.Ext B _~A_ _~B_ ε w → Set ℓ)
  (P~ : ∀ {w₀ w₁} → QW-rel.QW₀~ B _~A_ _~B_ ε w₀ w₁
    → {extw₀ : QW-rel.Ext B _~A_ _~B_ ε w₀}
    → {extw₁ : QW-rel.Ext B _~A_ _~B_ ε w₁}
    → P w₀ extw₀ → P w₁ extw₁ → Prop ℓ) where

  open QW-rel B _~A_ _~B_ ε using (QW₀~; Ext)

  Step : Set _
  Step = {a : A} {f : B a → 𝒲 A B}
    → (extf : (b : B a) → Ext (f b))
    → (f~ : ∀ {b₀ b₁} → b₀ ~B b₁ → QW₀~ (f b₀) (f b₁))
    → (ih : (b : B a) → P (f b) (extf b))
    → (ih~ : ∀ {b₀ b₁} (b₀₁ : b₀ ~B b₁)
      → P~ (f~ b₀₁) (ih b₀) (ih b₁))
    → P (sup₀ a f) (f~ , extf)

  Step~ : Step → Prop _
  Step~ p = {a₀ : A} {f₀ : B a₀ → 𝒲 A B}
    → (f₀~ : ∀ {b₀ b₀'} → b₀ ~B b₀' → QW₀~ (f₀ b₀) (f₀ b₀'))
    → (extf₀ : (b₀ : B a₀) → Ext (f₀ b₀))
    → (ih₀ : (b₀ : B a₀) → P (f₀ b₀) (extf₀ b₀))
    → (ih₀~ : ∀ {b₀ b₀'} (b₀₀' : b₀ ~B b₀')
      → P~ (f₀~ b₀₀') (ih₀ b₀) (ih₀ b₀'))
    → {a₁ : A} {f₁ : B a₁ → 𝒲 A B}
    → (f₁~ : ∀ {b₁ b₁'} → b₁ ~B b₁' → QW₀~ (f₁ b₁) (f₁ b₁'))
    → (extf₁ : (b₁ : B a₁) → Ext (f₁ b₁))
    → (ih₁ : (b₁ : B a₁) → P (f₁ b₁) (extf₁ b₁))
    → (ih₁~ : ∀ {b₁ b₁'} (b₁₁' : b₁ ~B b₁')
      → P~ (f₁~ b₁₁') (ih₁ b₁) (ih₁ b₁'))
    → (a₀₁ : a₀ ~A a₁)
    → (f₀₁ : ∀ {b₀ b₁} → b₀ ~B b₁ → QW₀~ (f₀ b₀) (f₁ b₁))
    → (ih₀₁ : ∀ {b₀ b₁} (b₀₁ : b₀ ~B b₁)
      → P~ (f₀₁ b₀₁) (ih₀ b₀) (ih₁ b₁))
    → P~ (qw-sup a₀₁ f₀₁)
        (p extf₀ f₀~ ih₀ ih₀~) (p extf₁ f₁~ ih₁ ih₁~)

module _
  {A : Set} (_~A_ : A → A → Prop) (Arefl : ∀ a → a ~A a)
  (Asym : ∀ {a₀ a₁} → a₀ ~A a₁ → a₁ ~A a₀)
  {B : A → Set} (_~B_ : ∀ {a₀ a₁} → B a₀ → B a₁ → Prop)
  (Bsym : ∀ {a₀ a₁} {b₀ : B a₀} {b₁ : B a₁} → b₀ ~B b₁ → b₁ ~B b₀)
  (Btrans : ∀ {a₀ a₁ a₂} {b₀ : B a₀} {b₁ : B a₁} {b₂ : B a₂}
    → b₀ ~B b₁ → b₁ ~B b₂ → b₀ ~B b₂)
  (Bcoe : ∀ {a₀ a₁} → a₀ ~A a₁ → B a₀ → B a₁)
  (Bcoh : ∀ {a₀ a₁} (aa : a₀ ~A a₁) (b : B a₀) → b ~B Bcoe aa b)
  (ε : EqSys₀ A B)
  (Vrefl : (e : E ε) (x : V ε e) → V~ ε e x x)
  (lhs-ext : ∀ e → extTerm _~A_ _~B_ (V~ ε e) (lhs ε e))
  (rhs-ext : ∀ e → extTerm _~A_ _~B_ (V~ ε e) (rhs ε e))
  where

  open QW-rel B _~A_ _~B_ ε
    using (eval~; QW₀~; Ext; eval-QExt; |𝒲|; qw-refl)

  module QW-elim₀
    -- Use the same carrier relation as the internal free-term recursor.
    (Q~ : |𝒲| → |𝒲| → Prop)
    (to-QEq : ∀ {u v} → Q~ u v → QW₀~ (u .₁) (v .₁))
    (from-QEq : ∀ {u v} → QW₀~ (u .₁) (v .₁) → Q~ u v)
    (P : (w : 𝒲 A B) → Ext w → Set ℓ)
    (P~ : ∀ {w₀ w₁} → QW₀~ w₀ w₁
      → {extw₀ : Ext w₀} {extw₁ : Ext w₁}
      → P w₀ extw₀ → P w₁ extw₁ → Prop ℓ)
    (P~refl : ∀ (w : 𝒲 A B) (extw : Ext w) (v : P w extw)
      → P~ (qw-refl Arefl extw) v v)
    (P~sym : ∀ {w₀ w₁} {extw₀ : Ext w₀} {extw₁ : Ext w₁}
      {v₀ : P w₀ extw₀} {v₁ : P w₁ extw₁} {w₀₁ : QW₀~ w₀ w₁}
      → P~ w₀₁ v₀ v₁ → P~ (qw-sym w₀₁) v₁ v₀)
    (P~trans : ∀ {w₀ w₁ w₂}
      {extw₀ : Ext w₀} {extw₁ : Ext w₁} {extw₂ : Ext w₂}
      {v₀ : P w₀ extw₀} {v₁ : P w₁ extw₁} {v₂ : P w₂ extw₂}
      {w₀₁ : QW₀~ w₀ w₁} {w₁₂ : QW₀~ w₁ w₂}
      → P~ w₀₁ v₀ v₁ → P~ w₁₂ v₁ v₂
      → P~ (qw-trans extw₁ w₀₁ w₁₂) v₀ v₂)
    (p : StepTypes.Step _~A_ _~B_ ε P P~)
    (p~ : StepTypes.Step~ _~A_ _~B_ ε P P~ p)
    where

    private
      module Eval = QW-rel.Evaluation B _~A_ _~B_ ε Arefl Q~ to-QEq from-QEq

    module _ {X : Set} (X~ : X → X → Prop)
      (ρ : X → |𝒲|)
      (ρ~ : ∀ {x y} → X~ x y → QW₀~ (ρ x .₁) (ρ y .₁)) where

      open Eval X~ ρ ρ~
        public using (eval-W; eval-W~; rec-raw)

      module _ (ρP : (x : X) → P (ρ x .₁) (ρ x .₂))
        (ρP~ : ∀ {x y} (xy : X~ x y) → P~ (ρ~ xy) (ρP x) (ρP y)) where

        private
          evaluated : Σₚ (Term₀ A B X) (extTerm _~A_ _~B_ X~) → |𝒲|
          evaluated (t , extt) = eval-W t extt

          evaluated~ : ∀ {t u}
            → Term₀~ _~A_ B B _~B_ X~ (t .₁) (u .₁)
            → QW₀~ (evaluated t .₁) (evaluated u .₁)
          evaluated~ {t} {u} = eval-W~ {t .₁} {u .₁} {t .₂} {u .₂}

        open FreeRaw.Elim₀ _~A_ _~B_ X~
          (λ t → P (evaluated t .₁) (evaluated t .₂))
          (λ {t} {t'} tt' → P~ (evaluated~ {t} {t'} tt'))
          ρP ρP~
          (λ a f f~ ih ih~ → p (λ b → evaluated (f b) .₂)
            (λ {b} {b'} bb → evaluated~ {f b} {f b'} (f~ bb)) ih ih~)
          (λ {f = f} {f' = f'} {f~} {f'~} ih~ ih'~ aa ff ii →
            p~ (λ {b} {b'} bb → evaluated~ {f b} {f b'} (f~ bb)) (λ b → evaluated (f b) .₂) _ ih~
               (λ {b} {b'} bb → evaluated~ {f' b} {f' b'} (f'~ bb)) (λ b → evaluated (f' b) .₂) _ ih'~
               aa (λ {b} {b'} bb → evaluated~ {f b} {f' b'} (ff bb)) ii)

          public using () renaming (val to deval; val~ to deval~)

    QW-elim₀-eq : ∀ (e : E ε) (ρ : V ε e → |𝒲|)
        (ρ~ : ∀ {x y} → V~ ε e x y → QW₀~ (ρ x .₁) (ρ y .₁))
        (ρP : (x : V ε e) → P (ρ x .₁) (ρ x .₂))
        (ρP~ : ∀ {x y} (xy : V~ ε e x y) → P~ (ρ~ xy) (ρP x) (ρP y))
      → Prop ℓ
    QW-elim₀-eq e ρ ρ~ ρP ρP~ =
      P~ (Eval.rec-eq (V~ ε e) ρ ρ~
          (lhs ε e) (rhs ε e) (lhs-ext e) (rhs-ext e) (qw-equ e ρ ρ~))
        (deval (V~ ε e) ρ ρ~ ρP ρP~ (lhs ε e) (lhs-ext e))
        (deval (V~ ε e) ρ ρ~ ρP ρP~ (rhs ε e) (rhs-ext e))
    {-# NOINLINE QW-elim₀-eq #-}

    QW-elim₀-coh : Prop _
    QW-elim₀-coh =
      ∀ (e : E ε) (ρ : V ε e → |𝒲|)
        (ρ~ : ∀ {x y} → V~ ε e x y → QW₀~ (ρ x .₁) (ρ y .₁))
        (ρP : (x : V ε e) → P (ρ x .₁) (ρ x .₂))
        (ρP~ : ∀ {x y} (xy : V~ ε e x y) → P~ (ρ~ xy) (ρP x) (ρP y))
      → QW-elim₀-eq e ρ ρ~ ρP ρP~

    elim-par : (w : 𝒲 A B) (extw : Ext w) → ℒ (P w extw)
    elim-par (sup₀ a f) (f~ , extf) =
      (∃[ ds ∈ ((b : B a) → elim-par (f b) (extf b) .extent) ]
        (∀ {b₀ b₁} (bb : b₀ ~B b₁) → P~ (f~ bb)
          (elim-par (f b₀) (extf b₀) .value (ds b₀))
          (elim-par (f b₁) (extf b₁) .value (ds b₁))))
      , λ { (ds , coh) →
          p extf f~ (partial-values (λ b → elim-par (f b) (extf b)) ds) coh }

    module _ {X : Set} (X~ : X → X → Prop)
      (ρ : X → |𝒲|)
      (ρ~ : ∀ {x y} → X~ x y → QW₀~ (ρ x .₁) (ρ y .₁))
      (ds : (x : X) → elim-par (ρ x .₁) (ρ x .₂) .extent)
      (coh : ∀ {x y} (xy : X~ x y) → P~ (ρ~ xy)
        (elim-par (ρ x .₁) (ρ x .₂) .value (ds x))
        (elim-par (ρ y .₁) (ρ y .₂) .value (ds y))) where

      private
        ρP = partial-values (λ x → elim-par (ρ x .₁) (ρ x .₂)) ds
        ev = deval X~ ρ ρ~ ρP coh
        ev~ = deval~ X~ ρ ρ~ ρP coh

      eval-domain : (t : Term₀ A B X) (extt : extTerm _~A_ _~B_ X~ t)
        → let w = eval-QExt X~ (t , extt) ρ ρ~
          in ∃[ d ∈ elim-par (w .₁) (w .₂) .extent ]
            P~ (qw-sym (rec-raw X~ ρ ρ~ t extt))
              (elim-par (w .₁) (w .₂) .value d) (ev t extt)
      eval-domain (var₀ x) _ = ds x , P~refl _ _ _
      eval-domain (op₀ a f) (f~ , extf) = ((λ b → ih b .₁) , coherent) ,
        p~ fr (λ b → ws b .₂) vals coherent
          (λ {b} {b'} bb → eval-W~ X~ ρ ρ~ {f b} {f b'} {extf b} {extf b'} (f~ bb))
          (λ b → eval-W X~ ρ ρ~ (f b) (extf b) .₂)
          (λ b → ev (f b) (extf b))
          (λ {b₀} {b₁} bb → ev~ (f b₀) (f b₁) (extf b₀) (extf b₁) (f~ bb))
          (Arefl a)
          (λ {b₀} {b₁} bb → qw-trans (ws b₁ .₂) (fr bb)
            (qw-sym (rec-raw X~ ρ ρ~ (f b₁) (extf b₁))))
          (λ {b₀} {b₁} bb → P~trans (ih b₀ .₂)
            (ev~ (f b₀) (f b₁) (extf b₀) (extf b₁) (f~ bb)))
        where
          ws = λ b → eval-QExt X~ (f b , extf b) ρ ρ~
          fr : ∀ {b₀ b₁} → b₀ ~B b₁ → QW₀~ (ws b₀ .₁) (ws b₁ .₁)
          fr {b₀} {b₁} bb = eval~ X~ {t₀ = f b₀} {t₁ = f b₁} (f~ bb) ρ ρ ρ~
          ih = λ b → eval-domain (f b) (extf b)
          vals = λ b → elim-par (ws b .₁) (ws b .₂) .value (ih b .₁)
          coherent : ∀ {b₀ b₁} (bb : b₀ ~B b₁) → P~ (fr bb) (vals b₀) (vals b₁)
          coherent {b₀} {b₁} bb = P~trans (ih b₀ .₂)
            (P~trans (ev~ (f b₀) (f b₁) (extf b₀) (extf b₁) (f~ bb)) (P~sym (ih b₁ .₂)))

    Sound : ∀ {w₀ w₁} → QW₀~ w₀ w₁ → Ext w₀ → Ext w₁ → Prop ℓ
    Sound {w₀} {w₁} eq ext₀ ext₁ =
      ℒ↓ (P~ eq) (elim-par w₀ ext₀) (elim-par w₁ ext₁)

    module _ (equ-p : QW-elim₀-coh) where

      sound : (w₀ w₁ : 𝒲 A B) (eq : QW₀~ w₀ w₁) (ext₀ : Ext w₀) (ext₁ : Ext w₁)
        → Sound eq ext₀ ext₁
      sound _ _ (qw-sym eq) ext₀ ext₁ with sound _ _ eq ext₁ ext₀
      ... | related d₁ d₀ eqP = related d₀ d₁ (P~sym eqP)
      sound _ _ (qw-trans mid eq₀ eq₁) ext₀ ext₂
        with sound _ _ eq₀ ext₀ mid | sound _ _ eq₁ mid ext₂
      ... | related d₀ _ eqP₀ | related _ d₂ eqP₁ = related d₀ d₂ (P~trans eqP₀ eqP₁)
      sound _ _ (qw-sup {f₀ = f₀} {f₁} aa ff) (f₀~ , ext₀) (f₁~ , ext₁) =
        related (ds₀ , coh₀) (ds₁ , coh₁)
          (p~ f₀~ ext₀ (partial-values (λ b → elim-par (f₀ b) (ext₀ b)) ds₀) coh₀
            f₁~ ext₁ (partial-values (λ b → elim-par (f₁ b) (ext₁ b)) ds₁) coh₁
            aa ff (λ bb → cross bb .val~))
        where
          cross : ∀ {b₀ b₁} (bb : b₀ ~B b₁) → Sound (ff bb) (ext₀ b₀) (ext₁ b₁)
          cross {b₀} {b₁} bb = sound _ _ (ff bb) (ext₀ b₀) (ext₁ b₁)

          ds₀ = λ b → cross (Bcoh aa b) .def₀
          ds₁ = λ b → cross (Bsym (Bcoh (Asym aa) b)) .def₁

          coh₀ = λ {b₀} {b₁} (bb : b₀ ~B b₁) →
            P~trans (cross (Btrans bb (Bcoh aa b₁)) .val~)
              (P~sym (cross (Bcoh aa b₁) .val~))
          coh₁ = λ {b₀} {b₁} (bb : b₀ ~B b₁) →
            P~trans (P~sym (cross (Bsym (Bcoh (Asym aa) b₀)) .val~))
              (cross (Btrans (Bsym (Bcoh (Asym aa) b₀)) bb) .val~)
      sound _ _ (qw-equ e ρ ρ~) ext₀ ext₁ = related (l .₁) (r .₁)
        (P~trans (l .₂)
          (P~trans (equ-p e ρ ρ~ vals coherent) (P~sym (r .₂))))
        where
          defined = λ x → sound _ _ (ρ~ (Vrefl e x)) (ρ x .₂) (ρ x .₂) .def₀
          vals = λ x → elim-par (ρ x .₁) (ρ x .₂) .value (defined x)
          coherent : ∀ {x y} (xy : V~ ε e x y) → P~ (ρ~ xy) (vals x) (vals y)
          coherent {x} {y} xy = sound _ _ (ρ~ xy) (ρ x .₂) (ρ y .₂) .val~
          l = eval-domain (V~ ε e) ρ ρ~ defined coherent (lhs ε e) (lhs-ext e)
          r = eval-domain (V~ ε e) ρ ρ~ defined coherent (rhs ε e) (rhs-ext e)

      elim-QW₀ : (w : 𝒲 A B) (extw : Ext w) → P w extw
      elim-QW₀ w extw = elim-par w extw .value (sound w w (qw-refl Arefl extw) extw extw .def₀)

      elim-QW₀~
        : ∀ {w₀ w₁} (w₀₁ : QW₀~ w₀ w₁)
        → (extw₀ : Ext w₀) (extw₁ : Ext w₁)
        → P~ w₀₁ (elim-QW₀ w₀ extw₀) (elim-QW₀ w₁ extw₁)
      elim-QW₀~ w₀₁ extw₀ extw₁ = sound _ _ w₀₁ extw₀ extw₁ .val~
