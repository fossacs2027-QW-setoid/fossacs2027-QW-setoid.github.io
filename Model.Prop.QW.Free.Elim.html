Var-step : (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero)
  (X : Ty Γ lzero) (P : Ty (Γ ▷ Term A B X) ℓ) → Set _
Var-step {Γ = Γ} A B X P =
  Tm (Γ ▷ X) (P [ wk ▸ var (A [ wk ]) (B [ wk ⁺ ]) (X [ wk ]) vz ])

Op-step : (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero)
  (X : Ty Γ lzero) (P : Ty (Γ ▷ Term A B X) ℓ) → Set _
Op-step {Γ = Γ} A B X P =
  let Γ₀ = Γ ▷ A ▷ B ⇒ Term A B X [ wk ]
  in Tm (Γ₀ ▷ Π (B [ wk ])
      (P [ wk∘wk∘wk ▸ app {Γ = Γ₀ ▷ B [ wk ]}
        {A = B [ wk∘wk ]} {B = Term A B X [ wk∘wk∘wk∘wk ]}
        (vs vz) ⟨ vz ⟩t ]))
    (P [ wk∘wk∘wk ▸
      (op (A [ wk∘wk ]) (B [ wk∘wk ⁺ ]) (X [ wk∘wk ])
        (vs vz) vz) [ wk ]t ])

module _ {Γ : Con ℓΓ} (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero)
  (X : Ty Γ lzero) (P : Ty (Γ ▷ Term A B X) ℓ)
  (v : Var-step A B X P)
  (p : Op-step A B X P) where

  open Term-rel using (Term₀~)

  elim₀ : (γ : ∣ Γ ∣)
    → (t : Term₀ (∣ A ∣ γ) (λ a → ∣ B ∣ (γ , a)) (∣ X ∣ γ))
    → (extt : extTerm (A ∋ Γ .ref γ ⊢_~_) (B~ A B (Γ .ref γ))
        (X ∋ Γ .ref γ ⊢_~_) t)
    → ∣ P ∣ (γ , (t , extt))
  elim₀ γ = Elim₀.val
    (A ∋ Γ .ref γ ⊢_~_) (B~ A B (Γ .ref γ)) (X ∋ Γ .ref γ ⊢_~_)
    (λ t → ∣ P ∣ (γ , t)) (λ tt' → P .heq (Γ .ref γ , tt'))
    (λ x → ∣ v ∣ (γ , x)) (λ xy → v .resp (Γ .ref γ , xy))
    (λ a f f~ ih ih~ → ∣ p ∣
      (((γ , a) , (f , λ bb → f~ (A .refl* a , bb)))
        , ih , λ bb → ih~ (A .refl* a , bb)))
    (λ ih~ ih'~ aa ff ii → p .resp
      (((Γ .ref γ , aa) , λ b b' bb → ff (aa , bb))
        , λ b b' bb → ii (aa , bb)))

  elim~ : ∀ {γ₀ γ₁} (q : Γ ∋ γ₀ ~ γ₁)
    → (t₀ : Term₀ (∣ A ∣ γ₀) (λ a → ∣ B ∣ (γ₀ , a)) (∣ X ∣ γ₀))
    → (t₁ : Term₀ (∣ A ∣ γ₁) (λ a → ∣ B ∣ (γ₁ , a)) (∣ X ∣ γ₁))
    → (ext₀ : extTerm (A ∋ Γ .ref γ₀ ⊢_~_) (B~ A B (Γ .ref γ₀))
        (X ∋ Γ .ref γ₀ ⊢_~_) t₀)
    → (ext₁ : extTerm (A ∋ Γ .ref γ₁ ⊢_~_) (B~ A B (Γ .ref γ₁))
        (X ∋ Γ .ref γ₁ ⊢_~_) t₁)
    → (tt' : Term₀~ (A ∋ q ⊢_~_) _ _ (B~ A B q) (X ∋ q ⊢_~_) t₀ t₁)
    → P ∋ (q , tt') ⊢ elim₀ γ₀ t₀ ext₀ ~ elim₀ γ₁ t₁ ext₁
  elim~ q (var₀ x₀) (var₀ x₁) _ _ xx = v .resp (q , xx)
  elim~ q (op₀ a₀ f₀) (op₀ a₁ f₁) (f₀~ , extf₀) (f₁~ , extf₁) (aa , ff) =
    p .resp (((q , aa) , λ b₀ b₁ bb → ff (aa , bb))
      , λ b₀ b₁ bb → elim~ q (f₀ b₀) (f₁ b₁) (extf₀ b₀) (extf₁ b₁)
          (ff (aa , bb)))

  elim-Term : Tm (Γ ▷ (Term A B X)) P
  ∣ elim-Term ∣ (γ , t) = elim₀ γ (t .₁) (t .₂)
  elim-Term .resp {γ₀ , t₀ , extt₀} {γ₁ , t₁ , extt₁} (γ₀₁ , t₀₁) =
    elim~ γ₀₁ t₀ t₁ extt₀ extt₁ t₀₁
  {-# INJECTIVE_FOR_INFERENCE elim-Term #-}

  elim-Termₑ : (t : Tm Γ (Term A B X)) → Tm Γ (P ⟨ t ⟩)
  elim-Termₑ t = elim-Term [ idS ▸ t ]t
  {-# INJECTIVE_FOR_INFERENCE elim-Termₑ #-}
