U₀-rec
  : (P : U₀ → Set ℓ)
  → P nat₀
  → (∀ (A : U₀) → P A
      → (B : El₀ A → U₀) → ((a : El₀ A) → P (B a))
      → (Bext : ∀ {a₀ a₁} → eq₀ A A a₀ a₁ → eqU₀ (B a₀) (B a₁))
      → P (sigma₀ A B Bext))
  → (∀ (A : U₀) → P A
      → (B : El₀ A → U₀) → ((a : El₀ A) → P (B a))
      → (Bext : ∀ {a₀ a₁} → eq₀ A A a₀ a₁ → eqU₀ (B a₀) (B a₁))
      → P (pi₀ A B Bext))
  → (∀ (A : U₀) → P A → (a b : El₀ A) → P (id₀ A a b))
  → (A : U₀) → P A
U₀-rec P nat* sigma* pi* id* (pack A Au Ae) = go Ae
  where
  go : ∀ {C : Set} {Cu : InU C} (Ce : Ext Cu) → P (pack C Cu Ce)
  go extNat = nat*
  go (extΣ Ce Cext Cf) =
    sigma* (pack _ _ Ce) (go Ce)
      (λ c → pack _ _ (Cf c)) (λ c → go (Cf c))
      (λ p → Cext _ _ p)
  go (extΠ Ce Cext Cf) =
    pi* (pack _ _ Ce) (go Ce)
      (λ c → pack _ _ (Cf c)) (λ c → go (Cf c))
      (λ p → Cext _ _ p)
  go (extId Ce c d) = id* (pack _ _ Ce) (go Ce) c d

U₀-rec₂
  : (P : (A B : U₀) → eqU₀ A B → Set ℓ)
  → P nat₀ nat₀ tt
  → (∀ (A B : U₀) (eAB : eqU₀ A B) → P A B eAB
      → (C : El₀ A → U₀) (D : El₀ B → U₀)
      → (eCD : ∀ a b → eq₀ A B a b → eqU₀ (C a) (D b))
      → (∀ a b (eab : eq₀ A B a b) → P (C a) (D b) (eCD a b eab))
      → (Cext : ∀ {a₀ a₁} → eq₀ A A a₀ a₁ → eqU₀ (C a₀) (C a₁))
      → (Dext : ∀ {b₀ b₁} → eq₀ B B b₀ b₁ → eqU₀ (D b₀) (D b₁))
      → P (sigma₀ A C Cext) (sigma₀ B D Dext) (eAB , eCD))
  → (∀ (A B : U₀) (eAB : eqU₀ A B) → P A B eAB
      → (C : El₀ A → U₀) (D : El₀ B → U₀)
      → (eCD : ∀ a b → eq₀ A B a b → eqU₀ (C a) (D b))
      → (∀ a b (eab : eq₀ A B a b) → P (C a) (D b) (eCD a b eab))
      → (Cext : ∀ {a₀ a₁} → eq₀ A A a₀ a₁ → eqU₀ (C a₀) (C a₁))
      → (Dext : ∀ {b₀ b₁} → eq₀ B B b₀ b₁ → eqU₀ (D b₀) (D b₁))
      → P (pi₀ A C Cext) (pi₀ B D Dext) (eAB , eCD))
  → (∀ (A B : U₀) (eAB : eqU₀ A B) → P A B eAB
      → (a₀ a₁ : El₀ A) (b₀ b₁ : El₀ B)
      → (e₀ : eq₀ A B a₀ b₀) (e₁ : eq₀ A B a₁ b₁)
      → P (id₀ A a₀ a₁) (id₀ B b₀ b₁) (eAB , (e₀ , e₁)))
  → ∀ (A B : U₀) (e : eqU₀ A B) → P A B e
U₀-rec₂ P nat* sigma* pi* id* (pack A Au Ae) (pack B Bu Be) e =
  go Ae Be e
  where
  go
    : ∀ {C D : Set} {Cu : InU C} {Du : InU D}
      (Ce : Ext Cu) (De : Ext Du) (eCD : CodeEq Cu Du)
    → P (pack C Cu Ce) (pack D Du De) eCD
  go extNat extNat eCD = nat*
  go (extΣ Ce Cext Cf) (extΣ De Dext Df) eCD =
    sigma* (pack _ _ Ce) (pack _ _ De) (eCD .₁)
      (go Ce De (eCD .₁))
      (λ c → pack _ _ (Cf c)) (λ d → pack _ _ (Df d))
      (eCD .₂)
      (λ c d cd → go (Cf c) (Df d) (eCD .₂ c d cd))
      (λ p → Cext _ _ p) (λ p → Dext _ _ p)
  go (extΠ Ce Cext Cf) (extΠ De Dext Df) eCD =
    pi* (pack _ _ Ce) (pack _ _ De) (eCD .₁)
      (go Ce De (eCD .₁))
      (λ c → pack _ _ (Cf c)) (λ d → pack _ _ (Df d))
      (eCD .₂)
      (λ c d cd → go (Cf c) (Df d) (eCD .₂ c d cd))
      (λ p → Cext _ _ p) (λ p → Dext _ _ p)
  go (extId Ce c₀ c₁) (extId De d₀ d₁) eCD =
    id* (pack _ _ Ce) (pack _ _ De) (eCD .₁)
      (go Ce De (eCD .₁)) c₀ c₁ d₀ d₁ (eCD .₂ .₁) (eCD .₂ .₂)
