open import Prelude

module Model.Prop.Unit.Core where

open import Model.Prop.CwF

Unit : Ty Γ ℓ
∣ Unit ∣ γ       = Lift _ ⊤
Unit .heq  _ _ _ = ⊤ₚ
Unit .refl*  _   = tt
Unit .sym* _ _ _ = tt
Unit .trans* _ _ _ _ _ = tt
Unit .coe    p t = t
Unit .coh    p t = tt

⋆ : Tm Γ (Unit {ℓ = ℓ})
⋆ = (λ γ → _) , λ p → tt
