open import Prelude

module Model.Prop.IF.Nat.Core where

open Prelude public
  using (ℕ; zero; suc)

private variable
  m n o k : ℕ

module N where
  elim : (P : ℕ → Set ℓ)
    → (pz : P zero) (ps : ∀ n → P n → P (suc n))
    → (m : ℕ) → P m
  elim P pz ps zero    = pz
  elim P pz ps (suc m) = ps m (elim P pz ps m)

  elimp : (P : ℕ → Prop ℓ)
    → (pz : P zero) (ps : ∀ n → P n → P (suc n))
    → (m : ℕ) → P m
  elimp P pz ps zero    = pz
  elimp P pz ps (suc m) = ps m (elimp P pz ps m)

  _~_ : ℕ → ℕ → Prop
  zero  ~ zero  = ⊤ₚ
  suc n ~ suc m = n ~ m
  suc n ~ zero  = ⊥ₚ
  zero  ~ suc m = ⊥ₚ

  -- -- large elimination into Prop
  -- data _≈_ : ℕ → ℕ → Prop where
  --   zero : zero ≈ zero
  --   suc  : m ≈ n → suc m ≈ suc n

  ref : ∀ n → n ~ n
  ref zero    = tt
  ref (suc n) = ref n

  sym : ∀ {m n} → m ~ n → n ~ m
  sym {zero}  {zero}  _ = tt
  sym {suc m} {suc n} p = sym {m} {n} p

  trans : ∀ {m n o} → m ~ n → n ~ o → m ~ o
  trans {zero}  {zero}  {o}     p q = q
  trans {suc m} {suc n} {suc o} p q = trans {m} {n} {o} p q

open import Model.Prop.CwF

Nat : Ty Γ lzero
∣ Nat ∣ γ = ℕ
Nat .heq γ₀₁ = N._~_
Nat .refl*  = N.ref
Nat .sym* m n = N.sym {m} {n}
Nat .trans* m n o = N.trans {m} {n} {o}
Nat .coe γ₀₁ m = m
Nat .coh γ₀₁ = N.ref

z : Tm Γ Nat
∣ z ∣ _ = zero
z .resp γ₀₁ = tt

s : Tm Γ Nat → Tm Γ Nat
∣ s n ∣ γ   = suc (∣ n ∣ γ)
s n .resp p = n .resp p
