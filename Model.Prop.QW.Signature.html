open import Prelude

module Model.Prop.QW.Signature where

open import Model.Prop.CwF
open import Model.Prop.QW.Free.Core

record EqSys
  {Γ : Con ℓΓ} (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero)
  : Set (lsuc ℓΓ) where
  field
    E : Ty Γ lzero
    V : Ty (Γ ▷ E) lzero
    lhs rhs : Tm (Γ ▷ E) (Term (A [ wk ]) (B [ wk ⁺ ]) V)

open EqSys public

_[_]ε
  : ∀ {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
  → EqSys A B → (σ : Sub Δ Γ)
  → EqSys (A [ σ ]) (B [ σ ⁺ ])
E   (ε [ σ ]ε) = E ε [ σ ]
V   (ε [ σ ]ε) = V ε [ σ ⁺ ]
lhs (ε [ σ ]ε) = lhs ε [ σ ⁺ ]t
rhs (ε [ σ ]ε) = rhs ε [ σ ⁺ ]t
