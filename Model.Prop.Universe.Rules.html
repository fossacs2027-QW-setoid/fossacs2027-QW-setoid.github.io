-- Strict Tarski computation.  Decoding agrees judgmentally with the existing
-- semantic formers as complete Ty records.
El-nat≡Nat : El (natᵁ {Γ = Γ}) ≡ Nat
El-nat≡Nat = refl

El-sigma≡Sigma
  : (A : Tm Γ U) (B : Tm (Γ ▷ El A) U)
  → El (sigmaᵁ A B) ≡ Sigma (El A) (El B)
El-sigma≡Sigma A B = refl

El-pi≡Π
  : (A : Tm Γ U) (B : Tm (Γ ▷ El A) U)
  → El (piᵁ A B) ≡ Π (El A) (El B)
El-pi≡Π A B = refl

El-id≡Id
  : (A : Tm Γ U) (a b : Tm Γ (El A))
  → El (idᵁ A a b) ≡ Id A a b
El-id≡Id A a b = refl
