module Model.Prop.Unit.Elim where

open import Model.Prop.CwF
open import Model.Prop.Unit.Core

elim-Unit
  : (P : Ty (Γ ▷ Unit {ℓ = ℓ'}) ℓ)
  → (t : Tm Γ (P ⟨ ⋆ ⟩))
  → Tm (Γ ▷ Unit) P
{-# INJECTIVE_FOR_INFERENCE elim-Unit #-}
elim-Unit P t = (λ (γ , _) → ∣ t ∣ γ) , λ (p , _) → t .resp p
