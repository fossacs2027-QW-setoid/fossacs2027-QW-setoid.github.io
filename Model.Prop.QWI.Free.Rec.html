open import Prelude

module Model.Prop.QWI.Free.Rec where

open import Model.Prop.CwF hiding (A; B)
open import Model.Prop.QWI.Polynomial
open import Model.Prop.QWI.Free.Core
open import Model.Prop.QWI.Free.Elim

module _ {Γ : Con ℓΓ} (S : Poly Γ) (X : Ty (Γ ▷ I S) lzero)
  (C : Ty (Γ ▷ I S) ℓ) where

  Term-rec-var-step : Set _
  Term-rec-var-step = Var-step S X (C [ wk ])

  Term-rec-op-step : Set _
  Term-rec-op-step = Op-step S X (C [ wk ])

  rec-Term : Term-rec-var-step → Term-rec-op-step
    → Tm (Γ ▷ I S ▷ Term S X) (C [ wk ])
  rec-Term = elim-Term S X (C [ wk ])

  rec-Termₑ : Term-rec-var-step → Term-rec-op-step
    → Tm (Γ ▷ I S) (Term S X) → Tm (Γ ▷ I S) C
  rec-Termₑ = elim-Termₑ S X (C [ wk ])
