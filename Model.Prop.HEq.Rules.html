EqS[]
  : ∀ {σ₀ σ₁ : Sub Δ Θ} {τ : Sub Γ Δ}
  → ⇑ EqS Θ σ₀ σ₁ [ τ ]
  ≡ ⇑ EqS Θ (σ₀ ∘ τ) (σ₁ ∘ τ)
EqS[] = refl

EqT[]
  : ∀ {Γ : Con ℓΓ}{Δ : Con ℓΔ} {A : Ty Δ ℓ}{σ₀ σ₁ : Sub Γ Δ}{τ : Sub Θ Γ}
    {σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁)}{t₀ : Tm Γ (A [ σ₀ ])}{t₁ : Tm Γ (A [ σ₁ ])}
  → ⇑ EqT A σ₀ σ₁ σ₀₁ t₀ t₁ [ τ ]
  ≡ ⇑ EqT A (σ₀ ∘ τ) (σ₁ ∘ τ) (σ₀₁ [ τ ]t) (t₀ [ τ ]t) (t₁ [ τ ]t)
EqT[] = refl

Eq[]
  : ∀ {A : Ty Δ ℓ} {σ : Sub Γ Δ} {τ : Sub Θ Γ}
    {t u : Tm Γ (A [ σ ])}
  → ⇑ Eq A σ t u [ τ ]
  ≡ ⇑ Eq A (σ ∘ τ) (t [ τ ]t) (u [ τ ]t)
Eq[] = refl

coeT[]
  : ∀ {A : Ty Δ ℓA}{σ₀ σ₁}{σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁)}{t₀ : Tm Γ (A [ σ₀ ])}
  → (coeT A σ₀ σ₁ σ₀₁ t₀) [ τ ]t
  ≡ coeT A (σ₀ ∘ τ) (σ₁ ∘ τ) (σ₀₁ [ τ ]t) (t₀ [ τ ]t)
coeT[] = refl

eqS-▸[] : ∀ {ℓΓ ℓΔ ℓθ} {Γ : Con ℓΓ} {Δ : Con ℓΔ} {A : Ty Δ ℓA} {Θ : Con ℓθ} {τ : Sub Θ Γ}
  → (σ₀ σ₁ : Sub Γ Δ)(t₀ : Tm Γ (A [ σ₀ ]))(t₁ : Tm Γ (A [ σ₁ ]))
  → (σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁))
  → (t₀₁ : Tm Γ (⇑ EqT A σ₀ σ₁ σ₀₁ t₀ t₁))
  → eqS-▸ A σ₀ σ₁ t₀ t₁ σ₀₁ t₀₁ [ τ ]t
  ≡ eqS-▸ A (σ₀ ∘ τ) (σ₁ ∘ τ) (t₀ [ τ ]t) (t₁ [ τ ]t) (σ₀₁ [ τ ]t) (t₀₁ [ τ ]t)
eqS-▸[] σ₀ σ₁ t₀ t₁ σ₀₁ t₀₁ = refl
