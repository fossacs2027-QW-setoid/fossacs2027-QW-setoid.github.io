data Term₀ (A : Set) (B : A → Set) (X : Set) : Set where
  var₀ : X → Term₀ A B X
  op₀  : (a : A) (f : B a → Term₀ A B X) → Term₀ A B X

map
  : ∀ {A₀ A₁ X₀ X₁ : Set} (h : A₀ → A₁)
  → {B₀ : A₀ → Set} {B₁ : A₁ → Set}
  → (∀ {a} → B₁ (h a) → B₀ a)
  → (X₀ → X₁)
  → Term₀ A₀ B₀ X₀ → Term₀ A₁ B₁ X₁
map h g k (var₀ x)  = var₀ (k x)
map h g k (op₀ a f) = op₀ (h a) (λ b → map h g k (f (g b)))

module Term-rel
  {A₀ A₁ : Set} (_~A_ : A₀ → A₁ → Prop)
  (B₀ : A₀ → Set) (B₁ : A₁ → Set)
  (_~B_ : ∀ {a₀ a₁} → B₀ a₀ → B₁ a₁ → Prop)
  {X₀ X₁ : Set} (_~X_ : X₀ → X₁ → Prop)
  where

  Term₀~ : Term₀ A₀ B₀ X₀ → Term₀ A₁ B₁ X₁ → Prop
  Term₀~ (var₀ x₀)   (var₀ x₁)   = x₀ ~X x₁
  Term₀~ (op₀ a₀ f₀) (op₀ a₁ f₁) =
    ∃[ a₀₁ ∈ a₀ ~A a₁ ]
      ({b₀ : B₀ a₀} {b₁ : B₁ a₁} → b₀ ~B b₁ → Term₀~ (f₀ b₀) (f₁ b₁))
  Term₀~ _ _ = ⊥ₚ

module _
  {A : Set} (_~A_ : A → A → Prop)
  {B : A → Set} (_~B_ : ∀ {a₀ a₁} → B a₀ → B a₁ → Prop)
  {X : Set} (_~X_ : X → X → Prop)
  where
  open Term-rel _~A_ B B _~B_ _~X_

  extTerm : Term₀ A B X → Prop
  extTerm (var₀ _)  = ⊤ₚ
  extTerm (op₀ a f) = ({b₀ b₁ : B a} → b₀ ~B b₁ → Term₀~ (f b₀) (f b₁))
    ∧ ((b : B a) → extTerm (f b))

module _
  {A : Set} (_~A_ : A → A → Prop)
  {B : A → Set} (_~B_ : ∀ {a₀ a₁} → B a₀ → B a₁ → Prop)
  {X : Set} (_~X_ : X → X → Prop)
  (Arefl : (a : A) → a ~A a)
  (Xrefl : (x : X) → x ~X x)
  where
  open Term-rel _~A_ B B _~B_ _~X_

  Term₀~-refl : (t : Term₀ A B X) → extTerm _~A_ _~B_ _~X_ t → Term₀~ t t
  Term₀~-refl (var₀ x)  _        = Xrefl x
  Term₀~-refl (op₀ a f) (f~ , _) = Arefl a , λ b₀₁ → f~ b₀₁

module _
  (A₀ A₁ : Set)
  (A₀₁~ : A₀ → A₁ → Prop) (A₁₀~ : A₁ → A₀ → Prop)
  (Asym : ∀ {a₀ a₁} → A₀₁~ a₀ a₁ → A₁₀~ a₁ a₀)
  (B₀ : A₀ → Set) (B₁ : A₁ → Set)
  (B₀₁~ : ∀ {a₀ a₁} → B₀ a₀ → B₁ a₁ → Prop)
  (B₁₀~ : ∀ {a₁ a₀} → B₁ a₁ → B₀ a₀ → Prop)
  (Bsym : ∀ {a₀ a₁ b₀ b₁} → B₁₀~ {a₁} {a₀} b₁ b₀ → B₀₁~ b₀ b₁)
  (X₀ X₁ : Set)
  (X₀₁~ : X₀ → X₁ → Prop) (X₁₀~ : X₁ → X₀ → Prop)
  (Xsym : ∀ {x₀ x₁} → X₀₁~ x₀ x₁ → X₁₀~ x₁ x₀)
  where
  open Term-rel

  Term₀~-sym
    : (t₀ : Term₀ A₀ B₀ X₀) (t₁ : Term₀ A₁ B₁ X₁)
    → Term₀~ A₀₁~ B₀ B₁ B₀₁~ X₀₁~ t₀ t₁
    → Term₀~ A₁₀~ B₁ B₀ B₁₀~ X₁₀~ t₁ t₀
  Term₀~-sym (var₀ x₀)   (var₀ x₁)   x₀₁         = Xsym x₀₁
  Term₀~-sym (op₀ a₀ f₀) (op₀ a₁ f₁) (a₀₁ , f₀₁) =
    Asym a₀₁ , λ {b₁} {b₀} b₁₀ →
      Term₀~-sym (f₀ b₀) (f₁ b₁) (f₀₁ (Bsym b₁₀))

module _
  (A₀ : Set) (B₀ : A₀ → Set) (X₀ : Set)
  (A₁ : Set) (B₁ : A₁ → Set) (X₁ : Set)
  (A₂ : Set) (B₂ : A₂ → Set) (X₂ : Set)
  (A₀₁~ : A₀ → A₁ → Prop) (A₁₂~ : A₁ → A₂ → Prop) (A₀₂~ : A₀ → A₂ → Prop)
  (B₀₁~ : ∀ {a₀ a₁} → B₀ a₀ → B₁ a₁ → Prop)
  (B₁₂~ : ∀ {a₁ a₂} → B₁ a₁ → B₂ a₂ → Prop)
  (B₀₂~ : ∀ {a₀ a₂} → B₀ a₀ → B₂ a₂ → Prop)
  (X₀₁~ : X₀ → X₁ → Prop) (X₁₂~ : X₁ → X₂ → Prop) (X₀₂~ : X₀ → X₂ → Prop)
  (Atrans : ∀ {a₀ a₁ a₂} → A₀₁~ a₀ a₁ → A₁₂~ a₁ a₂ → A₀₂~ a₀ a₂)
  (Bcoe₀₁ : ∀ {a₀ a₁} → A₀₁~ a₀ a₁ → B₀ a₀ → B₁ a₁)
  (Bcoh₀₁ : ∀ {a₀ a₁} (a₀₁ : A₀₁~ a₀ a₁) (b₀ : B₀ a₀) → B₀₁~ b₀ (Bcoe₀₁ a₀₁ b₀))
  (Btrans : ∀ {a₀ a₁ a₂} (a₀₁ : A₀₁~ a₀ a₁) (a₁₂ : A₁₂~ a₁ a₂)
    {b₀ : B₀ a₀} {b₂ : B₂ a₂} → B₀₂~ b₀ b₂ → B₁₂~ (Bcoe₀₁ a₀₁ b₀) b₂)
  (Xtrans : ∀ {x₀ x₁ x₂} → X₀₁~ x₀ x₁ → X₁₂~ x₁ x₂ → X₀₂~ x₀ x₂)
  where
  open Term-rel

  Term₀~-trans
    : {t₀ : Term₀ A₀ B₀ X₀} {t₁ : Term₀ A₁ B₁ X₁} {t₂ : Term₀ A₂ B₂ X₂}
    → Term₀~ A₀₁~ B₀ B₁ B₀₁~ X₀₁~ t₀ t₁
    → Term₀~ A₁₂~ B₁ B₂ B₁₂~ X₁₂~ t₁ t₂
    → Term₀~ A₀₂~ B₀ B₂ B₀₂~ X₀₂~ t₀ t₂
  Term₀~-trans {var₀ _}    {var₀ _}    {var₀ _}    x₀₁         x₁₂ = Xtrans x₀₁ x₁₂
  Term₀~-trans {op₀ a₀ f₀} {op₀ a₁ f₁} {op₀ a₂ f₂} (a₀₁ , f₀₁) (a₁₂ , f₁₂) =
    Atrans a₀₁ a₁₂ , λ {b₀} {b₂} b₀₂ →
      Term₀~-trans
        {t₀ = f₀ b₀} {t₁ = f₁ (Bcoe₀₁ a₀₁ b₀)} {t₂ = f₂ b₂}
        (f₀₁ (Bcoh₀₁ a₀₁ b₀))
        (f₁₂ (Btrans a₀₁ a₁₂ b₀₂))

module _
  {A₀ A₁ : Set} {_~A₀_ : A₀ → A₀ → Prop} {_~A₁_ : A₁ → A₁ → Prop}
  {B₀ : A₀ → Set} {_~B₀_ : ∀ {a₀ a₀'} → B₀ a₀ → B₀ a₀' → Prop}
  {B₁ : A₁ → Set} {_~B₁_ : ∀ {a₁ a₁'} → B₁ a₁ → B₁ a₁' → Prop}
  {X₀ X₁ : Set} {_~X₀_ : X₀ → X₀ → Prop} {_~X₁_ : X₁ → X₁ → Prop}
  where
  open Term-rel _~A₀_ B₀ B₀ _~B₀_ _~X₀_ using (Term₀~)

  open Term-rel _~A₁_ B₁ B₁ _~B₁_ _~X₁_ using ()
    renaming (Term₀~ to Term₁~)

  map~
    : (h : A₀ → A₁) (h~ : ∀ {a₀ a₀'} → a₀ ~A₀ a₀' → h a₀ ~A₁ h a₀')
    → (g : ∀ {a₀} → B₁ (h a₀) → B₀ a₀)
    → (g~ : ∀ {a₀ a₀'} {b₁ : B₁ (h a₀)} {b₁' : B₁ (h a₀')} → b₁ ~B₁ b₁' → g b₁ ~B₀ g b₁')
    → (k : X₀ → X₁) (k~ : ∀ {x y} → x ~X₀ y → k x ~X₁ k y)
    → {t₀ t₀' : Term₀ A₀ B₀ X₀}
    → Term₀~  t₀ t₀'
    → Term₁~ (map h g k t₀) (map h g k t₀')
  map~ h h~ g g~ k k~ {var₀ x}  {var₀ y} xy = k~ xy
  map~ h h~ g g~ k k~ {op₀ a f} {op₀ a' f'} (aa' , ff') =
    h~ aa' , λ {b₀} {b₁} bb' →
      map~ h h~ g g~ k k~
        {t₀ = f (g b₀)} {t₀' = f' (g b₁)} (ff' (g~ bb'))

  map-ext
    : (h : A₀ → A₁) (h~ : ∀ {a₀ a₀'} → a₀ ~A₀ a₀' → h a₀ ~A₁ h a₀')
    → (g : ∀ {a₀} → B₁ (h a₀) → B₀ a₀)
    → (g~ : ∀ {a₀ a₀'} {b₁ : B₁ (h a₀)} {b₁' : B₁ (h a₀')} → b₁ ~B₁ b₁' → g b₁ ~B₀ g b₁')
    → (k : X₀ → X₁) (k~ : ∀ {x y} → x ~X₀ y → k x ~X₁ k y)
    → (t : Term₀ A₀ B₀ X₀)
    → extTerm _~A₀_ _~B₀_ _~X₀_ t
    → extTerm _~A₁_ _~B₁_ _~X₁_
        (map h {B₀ = B₀} {B₁ = B₁} g k t)
  map-ext h h~ g g~ k k~ (var₀ _)  _ = tt
  map-ext h h~ g g~ k k~ (op₀ a f) (f~ , extf) =
    (λ {b₀} {b₁} bb' → map~ h h~ g g~ k k~
      {t₀ = f (g b₀)} {t₀' = f (g b₁)} (f~ (g~ bb')))
    , λ b → map-ext h h~ g g~ k k~ (f (g b)) (extf (g b))

module _ (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero) where
  B~ : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) {a₀ : ∣ A ∣ γ₀} {a₁ : ∣ A ∣ γ₁}
    → ∣ B ∣ (γ₀ , a₀) → ∣ B ∣ (γ₁ , a₁) → Prop
  B~ γ₀₁ {a₀} {a₁} b₀ b₁ =
    ∃[ a₀₁ ∈ (A ∋ γ₀₁ ⊢ a₀ ~ a₁) ] (B ∋ (γ₀₁ , a₀₁) ⊢ b₀ ~ b₁)

module _ (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero) (X : Ty Γ lzero) where
  open Term-rel

  |Term| : ∣ Γ ∣ → Set
  |Term| γ = Σₚ (Term₀ (∣ A ∣ γ) (λ a → ∣ B ∣ (γ , a)) (∣ X ∣ γ))
    (extTerm (A .heq (Γ .ref γ)) (B~ A B (Γ .ref γ)) (X .heq (Γ .ref γ)))

  Term~ : ∀ {γ₀ γ₁} → Γ ∋ γ₀ ~ γ₁ → |Term| γ₀ → |Term| γ₁ → Prop
  Term~ {γ₀} {γ₁} γ₀₁ (t₀ , _) (t₁ , _) =
    Term₀~ (A ∋ γ₀₁ ⊢_~_)
      (λ a₀ → ∣ B ∣ (γ₀ , a₀)) (λ a₁ → ∣ B ∣ (γ₁ , a₁))
      (B~ A B γ₀₁) (X ∋ γ₀₁ ⊢_~_) t₀ t₁

  Term-refl : ∀ {γ} (t : |Term| γ) → Term~ (Γ .ref γ) t t
  Term-refl (t , extt) =
    Term₀~-refl (A ∋ Γ .ref _ ⊢_~_) (B~ A B (Γ .ref _)) (X ∋ Γ .ref _ ⊢_~_)
      (A .refl*) (X .refl*) t extt

  Term-sym
    : ∀ {γ₀ γ₁} {γ₀₁ : Γ ∋ γ₀ ~ γ₁} (t₀ : |Term| γ₀) (t₁ : |Term| γ₁)
    → Term~ γ₀₁ t₀ t₁ → Term~ (Γ .sym γ₀₁) t₁ t₀
  Term-sym {γ₀} {γ₁} {γ₀₁} (t₀ , _) (t₁ , _) =
    Term₀~-sym
      (∣ A ∣ γ₀) (∣ A ∣ γ₁)
      (A ∋ γ₀₁ ⊢_~_) (A ∋ Γ .sym γ₀₁ ⊢_~_) (λ {x} {y} → A .sym* x y)
      (λ a₀ → ∣ B ∣ (γ₀ , a₀)) (λ a₁ → ∣ B ∣ (γ₁ , a₁))
      (B~ A B γ₀₁) (B~ A B (Γ .sym γ₀₁))
      (λ (a₁₀ , b₁₀) → A .sym* _ _ a₁₀ , B .sym* _ _ b₁₀)
      (∣ X ∣ γ₀) (∣ X ∣ γ₁) (X ∋ γ₀₁ ⊢_~_) (X ∋ Γ .sym γ₀₁ ⊢_~_) (λ {x} {y} → X .sym* x y) t₀ t₁

  Term-trans
    : ∀ {γ₀ γ₁ γ₂} {γ₀₁ : Γ ∋ γ₀ ~ γ₁} {γ₁₂ : Γ ∋ γ₁ ~ γ₂}
      (t₀ : |Term| γ₀) (t₁ : |Term| γ₁) (t₂ : |Term| γ₂)
    → Term~ γ₀₁ t₀ t₁ → Term~ γ₁₂ t₁ t₂
    → Term~ (Γ .trans γ₀₁ γ₁₂) t₀ t₂
  Term-trans {γ₀} {γ₁} {γ₂} {γ₀₁} {γ₁₂} (t₀ , _) (t₁ , _) (t₂ , _) =
    Term₀~-trans
      (∣ A ∣ γ₀) (λ a → ∣ B ∣ (γ₀ , a)) (∣ X ∣ γ₀)
      (∣ A ∣ γ₁) (λ a → ∣ B ∣ (γ₁ , a)) (∣ X ∣ γ₁)
      (∣ A ∣ γ₂) (λ a → ∣ B ∣ (γ₂ , a)) (∣ X ∣ γ₂)
      (A ∋ γ₀₁ ⊢_~_) (A ∋ γ₁₂ ⊢_~_) (A ∋ Γ .trans γ₀₁ γ₁₂ ⊢_~_)
      (B~ A B γ₀₁) (B~ A B γ₁₂) (B~ A B (Γ .trans γ₀₁ γ₁₂))
      (X ∋ γ₀₁ ⊢_~_) (X ∋ γ₁₂ ⊢_~_) (X ∋ Γ .trans γ₀₁ γ₁₂ ⊢_~_) (λ {x} {y} {z} → A .trans* x y z)
      (λ a₀₁ b₀ → B .coe (γ₀₁ , a₀₁) b₀)
      (λ a₀₁ b₀ → a₀₁ , B .coh (γ₀₁ , a₀₁) b₀)
      (λ a₀₁ a₁₂ (a₀₂ , b₀₂) →
        a₁₂ , B .trans* _ _ _ (B .sym* _ _ (B .coh (γ₀₁ , a₀₁) _)) b₀₂)
      (λ {x} {y} {z} → X .trans* x y z) {t₀ = t₀} {t₁ = t₁} {t₂ = t₂}

  Term-coe : ∀ {γ₀ γ₁} → Γ ∋ γ₀ ~ γ₁ → |Term| γ₀ → |Term| γ₁
  Term-coe γ₀₁ (t , extt) =
    map (A .coe γ₀₁)
      {B₀ = λ a₀ → ∣ B ∣ (_ , a₀)} {B₁ = λ a₁ → ∣ B ∣ (_ , a₁)}
      (λ {a₀} b₁ → B .coe (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a₀)) b₁) (X .coe γ₀₁) t
    , map-ext
        (A .coe γ₀₁)
        (λ {a₀} {a₀'} a₀₀' → A .trans* _ _ _ (A .sym* _ _ (A .coh γ₀₁ a₀)) (A .trans* _ _ _ a₀₀' (A .coh γ₀₁ a₀')))
        (λ {a₀} b₁ → B .coe (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a₀)) b₁)
        (λ {a₀} {a₀'} {b₁} {b₁'} (a₁₁' , b₁₁') →
          A .trans* _ _ _ (A .coh γ₀₁ a₀) (A .trans* _ _ _ a₁₁' (A .sym* _ _ (A .coh γ₀₁ a₀')))
          , B .trans* _ _ _ (B .sym* _ _ (B .coh (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a₀)) b₁))
              (B .trans* _ _ _ b₁₁' (B .coh (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a₀')) b₁')))
        (X .coe γ₀₁)
        (λ {x} {y} xy → X .trans* _ _ _ (X .sym* _ _ (X .coh γ₀₁ x))
          (X .trans* _ _ _ xy (X .coh γ₀₁ y)))
        t extt

  Term-coh
    : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (t : |Term| γ₀)
    → Term~ γ₀₁ t (Term-coe γ₀₁ t)
  Term-coh γ₀₁ (var₀ x , _) = X .coh γ₀₁ x
  Term-coh γ₀₁ (op₀ a f , f~ , extf) =
    A .coh γ₀₁ a , λ {b₀} {b₂} (a₀₂ , b₀₂) →
      let b₁ = B .coe (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a)) b₂
          t₁ = f b₁ , extf b₁
      in Term-trans (f b₀ , extf b₀) t₁ (Term-coe γ₀₁ t₁)
           (f~
             (A .sym* _ _ (A .trans* _ _ _ (A .coh γ₀₁ a) (A .sym* _ _ a₀₂))
             , B .trans* _ _ _ b₀₂ (B .coh (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a)) b₂)))
           (Term-coh γ₀₁ t₁)

  Term : Ty Γ lzero
  ∣ Term ∣     = |Term|
  Term .heq    = Term~
  Term .refl*  = Term-refl
  Term .sym*   = Term-sym
  Term .trans* = Term-trans
  Term .coe    = Term-coe
  Term .coh    = Term-coh
  {-# INJECTIVE_FOR_INFERENCE Term #-}

  var : (x : Tm Γ X) → Tm Γ Term
  ∣ var x ∣ γ = var₀ (∣ x ∣ γ) , tt
  var x .resp = x .resp

  op
    : (a : Tm Γ A) (f : Tm Γ (Π (B ⟨ a ⟩) (Term [ wk ])))
    → Tm Γ Term
  ∣ op a f ∣ γ =
    op₀ (∣ a ∣ γ) (λ b → ∣ f ∣ γ .₁ b .₁)
    , (λ (_ , b₀₁) → ∣ f ∣ γ .₂ b₀₁)
    , λ b → ∣ f ∣ γ .₁ b .₂
  op a f .resp γ₀₁ =
    a .resp γ₀₁ , λ (_ , b₀₁) → f .resp γ₀₁ _ _ b₀₁
