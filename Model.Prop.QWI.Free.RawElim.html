open import Prelude

module Model.Prop.QWI.Free.RawElim where

open import Model.Prop.QWI.Free.Core
open ℒ using (extent; value)

module _ {I : Set} {A : I → Set} {B : ∀ {i} → A i → Set}
  {ar : ∀ {i a} → B {i} a → I} {X : I → Set}
  (A~ : ∀ {i j} → A i → A j → Prop)
  (B~ : ∀ {i j a a'} → B {i} a → B {j} a' → Prop)
  (X~ : ∀ {i j} → X i → X j → Prop) where

  open Term-rel {I₀ = I} {I₁ = I} {A₀ = A} {A₁ = A} A~
    {B₀ = B} {B₁ = B} B~ {ar₀ = ar} {ar₁ = ar} {X₀ = X} {X₁ = X} X~

  private
    T : I → Set
    T i = Σₚ (Term₀ A B ar X i) (extTerm A~ B~ X~)

    Ext : ∀ {i} {a : A i} → ((b : B a) → T (ar b)) → Prop
    Ext f = ∀ {b b'} → B~ b b' → Term₀~ (f b .₁) (f b' .₁)

  module Elim₀ (P : ∀ {i} → T i → Set ℓ)
    (P~ : ∀ {i j} {t : T i} {t' : T j} → Term₀~ (t .₁) (t' .₁) → P t → P t' → Prop ℓ)
    (v : ∀ {i} (x : X i) → P (var₀ x , tt))
    (v~ : ∀ {i j} {x : X i} {y : X j} (xy : X~ x y) → P~ xy (v x) (v y))
    (p : ∀ {i} (a : A i) (f : (b : B a) → T (ar b)) (f~ : Ext f)
      → (ih : (b : B a) → P (f b))
      → (∀ {b b'} (bb : B~ b b') → P~ (f~ bb) (ih b) (ih b'))
      → P (op₀ a (λ b → f b .₁) , f~ , λ b → f b .₂))
    (p~ : ∀ {i j} {a : A i} {a' : A j}
      {f : (b : B a) → T (ar b)} {f' : (b : B a') → T (ar b)}
      {f~ : Ext f} {f'~ : Ext f'}
      {ih : (b : B a) → P (f b)} {ih' : (b : B a') → P (f' b)}
      → (ih~ : ∀ {b b'} (bb : B~ b b') → P~ (f~ bb) (ih b) (ih b'))
      → (ih'~ : ∀ {b b'} (bb : B~ b b') → P~ (f'~ bb) (ih' b) (ih' b'))
      → (aa : A~ a a')
      → (ff : ∀ {b b'} → B~ b b' → Term₀~ (f b .₁) (f' b' .₁))
      → (∀ {b b'} (bb : B~ b b') → P~ (ff bb) (ih b) (ih' b'))
      → P~ (aa , ff) (p a f f~ ih ih~) (p a' f' f'~ ih' ih'~)) where

    elim-par : ∀ {i} (t : Term₀ A B ar X i) (extt : extTerm A~ B~ X~ t) → ℒ (P (t , extt))
    elim-par (var₀ x) _ = ⊤ₚ , λ _ → v x
    elim-par (op₀ a f) (f~ , extf) =
      (∃[ ds ∈ ((b : B a) → elim-par (f b) (extf b) .extent) ]
        (∀ {b b'} (bb : B~ b b') → P~ (f~ bb)
          (elim-par (f b) (extf b) .value (ds b))
          (elim-par (f b') (extf b') .value (ds b'))))
      , λ { (ds , coh) → p a (λ b → f b , extf b) f~
          (partial-values (λ b → elim-par (f b) (extf b)) ds) coh }

    elim-par~ : ∀ {i j} (t : Term₀ A B ar X i) (t' : Term₀ A B ar X j)
      → (extt : extTerm A~ B~ X~ t) (extt' : extTerm A~ B~ X~ t')
      → (tt' : Term₀~ t t')
      → (d : elim-par t extt .extent) (d' : elim-par t' extt' .extent)
      → P~ tt' (elim-par t extt .value d) (elim-par t' extt' .value d')
    elim-par~ (var₀ x) (var₀ y) _ _ xy _ _ = v~ xy
    elim-par~ (op₀ a f) (op₀ a' f') (f~ , extf) (f'~ , extf')
      (aa , ff) (ds , coh) (ds' , coh') = p~ coh coh' aa ff
        (λ {b} {b'} bb → elim-par~ (f b) (f' b') (extf b) (extf' b') (ff bb) (ds b) (ds' b'))

    totality : ∀ {i} (t : Term₀ A B ar X i) (extt : extTerm A~ B~ X~ t) → elim-par t extt .extent
    totality (var₀ x) _ = tt
    totality (op₀ a f) (f~ , extf) = (λ b → totality (f b) (extf b))
      , λ {b} {b'} bb → elim-par~ (f b) (f b') (extf b) (extf b') (f~ bb)
          (totality (f b) (extf b)) (totality (f b') (extf b'))

    val : ∀ {i} (t : Term₀ A B ar X i) (extt : extTerm A~ B~ X~ t) → P (t , extt)
    val t extt = elim-par t extt .value (totality t extt)

    val~ : ∀ {i j} (t : Term₀ A B ar X i) (t' : Term₀ A B ar X j)
      → (extt : extTerm A~ B~ X~ t) (extt' : extTerm A~ B~ X~ t')
      → (tt' : Term₀~ t t') → P~ tt' (val t extt) (val t' extt')
    val~ t t' extt extt' tt' = elim-par~ t t' extt extt' tt' (totality t extt) (totality t' extt')
