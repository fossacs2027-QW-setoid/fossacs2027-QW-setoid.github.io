data 𝒲I {I : Set} (A : I → Set) (B : ∀ {i} → A i → Set)
  (ar : ∀ {i a} → B {i} a → I) : I → Set where
  supᵢ₀ : ∀ {i} (a : A i) → ((b : B a) → 𝒲I A B ar (ar b)) → 𝒲I A B ar i

record EqSys₀ {I : Set} (A : I → Set) (B : ∀ {i} → A i → Set)
  (ar : ∀ {i a} → B {i} a → I) : Set₁ where
  field
    E : I → Set
    V : ∀ {i} → E i → I → Set
    V~ : ∀ {i} (e : E i) → ∀ {j k} → V e j → V e k → Prop
    lhs rhs : ∀ {i} (e : E i) → Term₀ A B ar (V e) i
open EqSys₀ public

eval : ∀ {I : Set} {A : I → Set} {B : ∀ {i} → A i → Set}
  {ar : ∀ {i a} → B {i} a → I} {X : I → Set} {i}
  → Term₀ A B ar X i → (∀ {j} → X j → 𝒲I A B ar j) → 𝒲I A B ar i
eval (var₀ x) ρ = ρ x
eval (op₀ a f) ρ = supᵢ₀ a (λ b → eval (f b) ρ)

module QWI-rel
  {I : Set} {A : I → Set} {B : ∀ {i} → A i → Set}
  {ar : ∀ {i a} → B {i} a → I}
  (A~ : ∀ {i j} → A i → A j → Prop)
  (B~ : ∀ {i j a a'} → B {i} a → B {j} a' → Prop)
  (ε : EqSys₀ A B ar) where

  open Term-rel {I₀ = I} {I₁ = I} {A₀ = A} {A₁ = A} A~
    {B₀ = B} {B₁ = B} B~ {ar₀ = ar} {ar₁ = ar} using (Term₀~)

  mutual
    data QWI₀~ : ∀ {i j} → 𝒲I A B ar i → 𝒲I A B ar j → Prop where
      qwi-sym
        : ∀ {i j} {w₀ : 𝒲I A B ar i} {w₁ : 𝒲I A B ar j} → QWI₀~ w₀ w₁ → QWI₀~ w₁ w₀
      qwi-trans
        : ∀ {i j k} {w₀ : 𝒲I A B ar i} {w₁ : 𝒲I A B ar j} {w₂ : 𝒲I A B ar k}
        → Ext w₁ → QWI₀~ w₀ w₁ → QWI₀~ w₁ w₂
        → QWI₀~ w₀ w₂
      qwi-sup
        : ∀ {i j} {a₀ : A i} {a₁ : A j}
          {f₀ : (b : B a₀) → 𝒲I A B ar (ar b)} {f₁ : (b : B a₁) → 𝒲I A B ar (ar b)}
        → A~ a₀ a₁
        → (∀ {b₀ b₁} → B~ b₀ b₁ → QWI₀~ (f₀ b₀) (f₁ b₁))
        → QWI₀~ (supᵢ₀ a₀ f₀) (supᵢ₀ a₁ f₁)
      qwi-equ : ∀ {i} (e : ε .E i)
        → (ρ : ∀ {j} → ε .V e j → Σₚ (𝒲I A B ar j) Ext)
        → (ρ~ : ∀ {j k} {x : ε .V e j} {y : ε .V e k} → ε .V~ e x y → QWI₀~ (ρ x .₁) (ρ y .₁))
        → QWI₀~
            (eval (ε .lhs e) (λ v → ρ v .₁))
            (eval (ε .rhs e) (λ v → ρ v .₁))

    Ext : ∀ {i} → 𝒲I A B ar i → Prop
    Ext (supᵢ₀ a f) =
      ({y₀ y₁ : B a} (y₀₁ : B~ y₀ y₁) → QWI₀~ (f y₀) (f y₁))
      ∧ ((y : B a) → Ext (f y))

  |𝒲| : I → Set
  |𝒲| i = Σₚ (𝒲I A B ar i) Ext

  qwi-refl : (Arefl : ∀ {i} (a : A i) → A~ a a) → ∀ {i} {w : 𝒲I A B ar i} → Ext w → QWI₀~ w w
  qwi-refl Arefl {w = supᵢ₀ a f} (f~ , _) = qwi-sup (Arefl a) f~

  eval~ : ∀ {X₀ X₁ : I → Set} (X~ : ∀ {i j} → X₀ i → X₁ j → Prop)
    {i j} {t₀ : Term₀ A B ar X₀ i} {t₁ : Term₀ A B ar X₁ j}
    → Term₀~ X~ t₀ t₁
    → (ρ₀ : ∀ {i} → X₀ i → |𝒲| i) (ρ₁ : ∀ {i} → X₁ i → |𝒲| i)
    → (∀ {i j} {x : X₀ i} {y : X₁ j} → X~ x y → QWI₀~ (ρ₀ x .₁) (ρ₁ y .₁))
    → QWI₀~ (eval t₀ (λ x → ρ₀ x .₁)) (eval t₁ (λ x → ρ₁ x .₁))
  eval~ X~ {t₀ = var₀ x₀} {var₀ x₁} x₀₁ ρ₀ ρ₁ ρ~ = ρ~ x₀₁
  eval~ X~ {t₀ = op₀ a₀ f₀} {op₀ a₁ f₁} (a₀₁ , f₀₁) ρ₀ ρ₁ ρ~ =
    qwi-sup a₀₁ (λ {b₀} {b₁} b₀₁ →
      eval~ X~ {t₀ = f₀ b₀} {t₁ = f₁ b₁} (f₀₁ b₀₁) ρ₀ ρ₁ ρ~)

  eval-ext : ∀ {X : I → Set} (X~ : ∀ {i j} → X i → X j → Prop) {i} (t : Term₀ A B ar X i) → extTerm A~ B~ X~ t
    → (ρ : ∀ {i} → X i → |𝒲| i)
    → (∀ {i j} {x : X i} {y : X j} → X~ x y → QWI₀~ (ρ x .₁) (ρ y .₁))
    → Ext (eval t (λ x → ρ x .₁))
  eval-ext X~ (var₀ x) _ ρ ρ~ = ρ x .₂
  eval-ext X~ (op₀ a f) (f~ , extf) ρ ρ~ =
    (λ {b₀} {b₁} b₀₁ → eval~ X~ {t₀ = f b₀} {t₁ = f b₁} (f~ b₀₁) ρ ρ ρ~)
    , λ b → eval-ext X~ (f b) (extf b) ρ ρ~

  eval-QExt : ∀ {X : I → Set} (X~ : ∀ {i j} → X i → X j → Prop) {i} → Σₚ (Term₀ A B ar X i) (extTerm A~ B~ X~)
    → (ρ : ∀ {i} → X i → |𝒲| i)
    → (∀ {i j} {x : X i} {y : X j} → X~ x y → QWI₀~ (ρ x .₁) (ρ y .₁))
    → |𝒲| i
  eval-QExt X~ (t , extt) ρ ρ~ = eval t (λ x → ρ x .₁) , eval-ext X~ t extt ρ ρ~

  module Evaluation (Arefl : ∀ {i} (a : A i) → A~ a a)
    (Q~ : ∀ {i j} → |𝒲| i → |𝒲| j → Prop)
    (to-QEq : ∀ {i j} {u : |𝒲| i} {v : |𝒲| j} → Q~ u v → QWI₀~ (u .₁) (v .₁))
    (from-QEq : ∀ {i j} {u : |𝒲| i} {v : |𝒲| j} → QWI₀~ (u .₁) (v .₁) → Q~ u v)
    {X : I → Set} (X~ : ∀ {i j} → X i → X j → Prop)
    (ρ : ∀ {i} → X i → |𝒲| i)
    (ρ~ : ∀ {i j} {x : X i} {y : X j} → X~ x y → QWI₀~ (ρ x .₁) (ρ y .₁)) where

    open FreeRaw.Elim₀ A~ B~ X~ (λ {i} _ → |𝒲| i) (λ _ → Q~)
      ρ (λ xy → from-QEq (ρ~ xy))
      (λ a f f~ ih ih~ → supᵢ₀ a (λ b → ih b .₁)
        , (λ bb → to-QEq (ih~ bb)) , λ b → ih b .₂)
      (λ ih~ ih'~ aa ff ii → from-QEq (qwi-sup aa (λ bb → to-QEq (ii bb))))
      using (val; val~)

    eval-W : ∀ {i} (t : Term₀ A B ar X i) → extTerm A~ B~ X~ t → |𝒲| i
    eval-W = val

    eval-W~ : ∀ {i j} {t₀ : Term₀ A B ar X i} {t₁ : Term₀ A B ar X j}
      {ext₀ : extTerm A~ B~ X~ t₀} {ext₁ : extTerm A~ B~ X~ t₁}
      → Term₀~ X~ t₀ t₁ → QWI₀~ (eval-W t₀ ext₀ .₁) (eval-W t₁ ext₁ .₁)
    eval-W~ {t₀ = t₀} {t₁} {ext₀} {ext₁} tt' =
      to-QEq (val~ t₀ t₁ ext₀ ext₁ tt')

    rec-raw : ∀ {i} (t : Term₀ A B ar X i) (extt : extTerm A~ B~ X~ t)
      → QWI₀~ (eval-W t extt .₁) (eval t (λ x → ρ x .₁))
    rec-raw (var₀ x) _ = qwi-refl Arefl (ρ x .₂)
    rec-raw (op₀ a f) (f~ , extf) = qwi-sup (Arefl a) λ {b} {b'} bb →
      qwi-trans (eval-QExt X~ (f b , extf b) ρ ρ~ .₂)
        (rec-raw (f b) (extf b))
        (eval~ X~ {t₀ = f b} {t₁ = f b'} (f~ bb) ρ ρ ρ~)

    rec-eq : ∀ {i j} (t : Term₀ A B ar X i) (u : Term₀ A B ar X j)
      (extt : extTerm A~ B~ X~ t) (extu : extTerm A~ B~ X~ u)
      → QWI₀~ (eval t (λ x → ρ x .₁)) (eval u (λ x → ρ x .₁))
      → QWI₀~ (eval-W t extt .₁) (eval-W u extu .₁)
    rec-eq t u extt extu eq =
      qwi-trans (eval-QExt X~ (t , extt) ρ ρ~ .₂) (rec-raw t extt)
        (qwi-trans (eval-QExt X~ (u , extu) ρ ρ~ .₂) eq
          (qwi-sym (rec-raw u extu)))

module _ {I₀ I₁ : Set} {A₀ : I₀ → Set} {A₁ : I₁ → Set}
  {B₀ : ∀ {i} → A₀ i → Set} {B₁ : ∀ {i} → A₁ i → Set}
  {ar₀ : ∀ {i a} → B₀ {i} a → I₀} {ar₁ : ∀ {i a} → B₁ {i} a → I₁} where

  map-W : (R : I₀ → I₁ → Prop)
    → (h : ∀ {i j} → R i j → A₀ i → A₁ j)
    → (g : ∀ {i j} (r : R i j) {a : A₀ i} → B₁ (h r a) → B₀ a)
    → (comm : ∀ {i j} (r : R i j) {a : A₀ i} (b : B₁ (h r a)) → R (ar₀ (g r b)) (ar₁ b))
    → ∀ {i j} → R i j → 𝒲I A₀ B₀ ar₀ i → 𝒲I A₁ B₁ ar₁ j
  map-W R h g comm r (supᵢ₀ a f) = supᵢ₀ (h r a)
    (λ b → map-W R h g comm (comm r b) (f (g r b)))

module QuotientConstruction {Γ : Con ℓΓ} (S : Poly Γ) (ε : EqSys S) where
  open At S

  QWI₀ : (γ : ∣ Γ ∣) → I₀ γ → Set
  QWI₀ γ = 𝒲I (A₀ {γ}) B₀ ar₀

  SigAt : (γ : ∣ Γ ∣) → EqSys₀ (A₀ {γ}) B₀ ar₀
  SigAt γ .E i = ∣ E ε ∣ (γ , i)
  SigAt γ .V {i} e j = ∣ V ε ∣ (((γ , i) , e) , j)
  SigAt γ .V~ {i} e {j} {k} x y = ∃[ jk ∈ (I S ∋ Γ .ref γ ⊢ j ~ k) ]
    (V ε ∋ (((Γ .ref γ , I S .refl* i) , E ε .refl* e) , jk) ⊢ x ~ y)
  SigAt γ .lhs {i} e = ∣ lhs ε ∣ ((γ , i) , e) .₁
  SigAt γ .rhs {i} e = ∣ rhs ε ∣ ((γ , i) , e) .₁

  module RelAt (γ : ∣ Γ ∣) = QWI-rel (A~ (Γ .ref γ)) (B~ (Γ .ref γ)) (SigAt γ)
  open RelAt

  qwi-index : ∀ {γ i j} {w : QWI₀ γ i} {v : QWI₀ γ j}
    → QWI₀~ γ w v → I S ∋ Γ .ref γ ⊢ i ~ j
  qwi-index (qwi-sym eq) = I S .sym* _ _ (qwi-index eq)
  qwi-index (qwi-trans extw eq eq') = I S .trans* _ _ _ (qwi-index eq) (qwi-index eq')
  qwi-index (qwi-sup aa ff) = aa .₁
  qwi-index (qwi-equ e ρ ρ~) = I S .refl* _

  QWI-coe : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) → I S ∋ q ⊢ i ~ j → QWI₀ γ i → QWI₀ δ j
  QWI-coe q = map-W (I S .heq q) (λ ij → A S .coe (q , ij))
    (pull q) (λ ij b → ar~ (pull~ q ij b))

  |QWI| : (γ : ∣ Γ ∣) → I₀ γ → Set
  |QWI| γ i = Σₚ (QWI₀ γ i) (Ext γ)

  eval-map : ∀ {γ δ} (q : Γ ∋ γ ~ δ)
    {X₀ : I₀ γ → Set} {X₁ : I₀ δ → Set}
    (X~ : ∀ {i j} → X₀ i → X₁ j → Prop)
    {i j k} (ij : I S ∋ q ⊢ i ~ j)
    {t₀ : Term₀ (A₀ {γ}) B₀ ar₀ X₀ i} {t₁ : Term₀ (A₀ {δ}) B₀ ar₀ X₁ k}
    → Term-rel.Term₀~ (A~ q) (B~ q) X~ t₀ t₁
    → (ρ₀ : ∀ {i} → X₀ i → QWI₀ γ i) (ρ₁ : ∀ {i} → X₁ i → QWI₀ δ i)
    → (∀ {i j k} (ij : I S ∋ q ⊢ i ~ j) {x : X₀ i} {y : X₁ k}
      → X~ x y → QWI₀~ δ (QWI-coe q ij (ρ₀ x)) (ρ₁ y))
    → QWI₀~ δ (QWI-coe q ij (eval t₀ ρ₀)) (eval t₁ ρ₁)
  eval-map q X~ ij {var₀ x} {var₀ y} xy ρ₀ ρ₁ ρ~ = ρ~ ij xy
  eval-map q X~ ij {op₀ a f} {op₀ a' f'} (aa , ff) ρ₀ ρ₁ ρ~ =
    qwi-sup (A-trans (A-sym (ij , A S .coh (q , ij) a)) aa)
      (λ {b} {b'} bb → eval-map q X~ (ar~ (pull~ q ij b))
        {t₀ = f (pull q ij b)} {t₁ = f' b'}
        (ff (B-trans (pull~ q ij b) bb)) ρ₀ ρ₁ ρ~)

  mutual
    QWI-coe~ : ∀ {γ δ i i' j j'} (q : Γ ∋ γ ~ δ)
      (ij : I S ∋ q ⊢ i ~ j) (ij' : I S ∋ q ⊢ i' ~ j')
      {w : QWI₀ γ i} {w' : QWI₀ γ i'}
      → QWI₀~ γ w w' → QWI₀~ δ (QWI-coe q ij w) (QWI-coe q ij' w')
    QWI-coe~ q ij ij' (qwi-sym eq) = qwi-sym (QWI-coe~ q ij' ij eq)
    QWI-coe~ {γ} {δ} q ij ij' (qwi-trans {j = k} {w₁ = w} extw eq eq') =
      qwi-trans (ext-coe q (I S .coh q k) w extw)
        (QWI-coe~ q ij (I S .coh q k) eq) (QWI-coe~ q (I S .coh q k) ij' eq')
    QWI-coe~ q ij ij' (qwi-sup {a₀ = a} {a₁ = a'} aa ff) =
      qwi-sup (A-trans (A-sym (ij , A S .coh (q , ij) a))
        (A-trans aa (ij' , A S .coh (q , ij') a')))
        (λ {b} {b'} bb → QWI-coe~ q (ar~ (pull~ q ij b)) (ar~ (pull~ q ij' b'))
          (ff (B-trans (pull~ q ij b) (B-trans bb (B-sym (pull~ q ij' b'))))))
    QWI-coe~ {γ} {δ} {i} {j = j} q ij ij' (qwi-equ e ρ ρ~) =
      qwi-trans (eval-QExt δ (SigAt δ .V~ e') (∣ lhs ε ∣ ((δ , j) , e')) ρ' ρ'~ .₂)
        (eval-map q VX ij {t₀ = ∣ lhs ε ∣ ((γ , i) , e) .₁} {t₁ = ∣ lhs ε ∣ ((δ , j) , e') .₁} (lhs ε .resp ((q , ij) , ee)) (λ x → ρ x .₁) (λ x → ρ' x .₁) cross)
        (qwi-trans (eval-QExt δ (SigAt δ .V~ e') (∣ rhs ε ∣ ((δ , j) , e')) ρ' ρ'~ .₂)
          (qwi-equ e' ρ' ρ'~)
          (qwi-sym (eval-map q VX ij' {t₀ = ∣ rhs ε ∣ ((γ , i) , e) .₁} {t₁ = ∣ rhs ε ∣ ((δ , j) , e') .₁} (rhs ε .resp ((q , ij) , ee))
            (λ x → ρ x .₁) (λ x → ρ' x .₁) cross)))
      where
        e' = E ε .coe (q , ij) e
        ee = E ε .coh (q , ij) e
        back = (Γ .sym q , I S .sym* _ _ ij) , E ε .sym* _ _ ee
        back-i : (k : I₀ δ) → I₀ γ
        back-i k = I S .coe (Γ .sym q) k
        back-ij : (k : I₀ δ) → I S ∋ q ⊢ back-i k ~ k
        back-ij k = I S .sym* _ _ (I S .coh (Γ .sym q) k)
        pull-v : ∀ {k} → SigAt δ .V e' k → SigAt γ .V e (back-i k)
        pull-v {k} = V ε .coe (back , I S .coh (Γ .sym q) k)
        ρ' : ∀ {k} → SigAt δ .V e' k → |QWI| δ k
        ρ' {k} v = QWI-coe q (back-ij k) (ρ (pull-v v) .₁)
          , ext-coe q (back-ij k) (ρ (pull-v v) .₁) (ρ (pull-v v) .₂)
        ρ'~ : ∀ {k l} {x : SigAt δ .V e' k} {y : SigAt δ .V e' l}
          → SigAt δ .V~ e' x y → QWI₀~ δ (ρ' x .₁) (ρ' y .₁)
        ρ'~ {k} {l} {x} {y} (kl , xy) = QWI-coe~ q (back-ij k) (back-ij l)
          (ρ~ (I S .trans* _ _ _ (back-ij k) (I S .trans* _ _ _ kl (I S .coh (Γ .sym q) l))
            , V ε .trans* _ _ _ (V ε .sym* _ _ (V ε .coh (back , I S .coh (Γ .sym q) k) x))
              (V ε .trans* _ _ _ xy (V ε .coh (back , I S .coh (Γ .sym q) l) y))))
        VX : ∀ {k l} → SigAt γ .V e k → SigAt δ .V e' l → Prop
        VX {k} {l} x y = ∃[ kl ∈ (I S ∋ q ⊢ k ~ l) ]
          (V ε ∋ (((q , ij) , ee) , kl) ⊢ x ~ y)
        cross : ∀ {k l m} (kl : I S ∋ q ⊢ k ~ l)
          {x : SigAt γ .V e k} {y : SigAt δ .V e' m}
          → VX x y → QWI₀~ δ (QWI-coe q kl (ρ x .₁)) (ρ' y .₁)
        cross {m = m} kl {y = y} (km , xy) = QWI-coe~ q kl (back-ij m)
          (ρ~ (I S .trans* _ _ _ km (I S .coh (Γ .sym q) m)
            , V ε .trans* _ _ _ xy (V ε .coh (back , I S .coh (Γ .sym q) m) y)))

    ext-coe : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) (ij : I S ∋ q ⊢ i ~ j)
      (w : QWI₀ γ i) → Ext γ w → Ext δ (QWI-coe q ij w)
    ext-coe q ij (supᵢ₀ a f) (f~ , extf) =
      (λ {b} {b'} bb → QWI-coe~ q (ar~ (pull~ q ij b)) (ar~ (pull~ q ij b'))
        (f~ (B-trans (pull~ q ij b) (B-trans bb (B-sym (pull~ q ij b'))))))
      , λ b → ext-coe q (ar~ (pull~ q ij b)) (f (pull q ij b)) (extf (pull q ij b))

  QWI-coe-id : ∀ {γ i j} (ij : I S ∋ Γ .ref γ ⊢ i ~ j)
    (w : QWI₀ γ i) → Ext γ w → QWI₀~ γ w (QWI-coe (Γ .ref γ) ij w)
  QWI-coe-id {γ} ij (supᵢ₀ a f) (f~ , extf) = qwi-sup (ij , A S .coh (Γ .ref γ , ij) a)
    λ {b} {b'} bb → qwi-trans (extf (pull (Γ .ref γ) ij b'))
      (f~ (B-trans bb (B-sym (pull~ (Γ .ref γ) ij b'))))
      (QWI-coe-id (ar~ (pull~ (Γ .ref γ) ij b')) (f (pull (Γ .ref γ) ij b')) (extf (pull (Γ .ref γ) ij b')))

  QWI-coe-∘ : ∀ {γ δ θ i j k} (q : Γ ∋ γ ~ δ) (r : Γ ∋ δ ~ θ)
    (ij : I S ∋ q ⊢ i ~ j) (jk : I S ∋ r ⊢ j ~ k) (w : QWI₀ γ i) → Ext γ w
    → QWI₀~ θ (QWI-coe r jk (QWI-coe q ij w)) (QWI-coe (Γ .trans q r) (I S .trans* _ _ _ ij jk) w)
  QWI-coe-∘ q r ij jk (supᵢ₀ a f) (f~ , extf) =
    qwi-sup (A-trans (A-sym (jk , A S .coh (r , jk) _))
      (A-trans (A-sym (ij , A S .coh (q , ij) a))
        (I S .trans* _ _ _ ij jk , A S .coh (Γ .trans q r , I S .trans* _ _ _ ij jk) a)))
      λ {b} {b'} bb →
        let b₀ = pull q ij (pull r jk b)
            ik = I S .trans* _ _ _ ij jk
            b₁ = pull (Γ .trans q r) ik b'
            ij₀ = ar~ (pull~ q ij (pull r jk b))
            jk₀ = ar~ (pull~ r jk b)
        in qwi-trans (ext-coe (Γ .trans q r) _ (f b₀) (extf b₀))
          (QWI-coe-∘ q r ij₀ jk₀ (f b₀) (extf b₀))
          (QWI-coe~ (Γ .trans q r) _ _ (f~
            (B-trans (pull~ q ij (pull r jk b))
              (B-trans (pull~ r jk b) (B-trans bb (B-sym (pull~ (Γ .trans q r) ik b')))))))

  QWI~ : ∀ {γ δ i j} → (q : Γ ∋ γ ~ δ) → I S ∋ q ⊢ i ~ j → |QWI| γ i → |QWI| δ j → Prop
  QWI~ {δ = δ} q ij (w , _) (w' , _) = QWI₀~ δ (QWI-coe q ij w) w'

  QWI : Ty (Γ ▷ I S) lzero
  ∣ QWI ∣ (γ , i) = |QWI| γ i
  QWI .heq (q , ij) = QWI~ q ij
  QWI .refl* (w , extw) = qwi-sym (QWI-coe-id (I S .refl* _) w extw)
  QWI .sym* {γ , i} {δ , j} {q , ij} (w , extw) (w' , extw') eq =
    qwi-trans (ext-coe (Γ .sym q) (I S .sym* _ _ ij) _ (ext-coe q ij w extw))
      (qwi-sym (QWI-coe~ (Γ .sym q) (I S .sym* _ _ ij) (I S .sym* _ _ ij) eq))
      (qwi-trans (ext-coe (Γ .trans q (Γ .sym q)) _ w extw)
        (QWI-coe-∘ q (Γ .sym q) ij (I S .sym* _ _ ij) w extw)
        (qwi-sym (QWI-coe-id _ w extw)))
  QWI .trans* {γ , i} {δ , j} {θ , k} {q , ij} {r , jk}
    (w , extw) (w' , extw') (w'' , extw'') eq eq' =
    qwi-trans (ext-coe r jk _ (ext-coe q ij w extw))
      (qwi-sym (QWI-coe-∘ q r ij jk w extw))
      (qwi-trans (ext-coe r jk w' extw') (QWI-coe~ r jk jk eq) eq')
  QWI .coe (q , ij) (w , extw) = QWI-coe q ij w , ext-coe q ij w extw
  QWI .coh (q , ij) (w , extw) = qwi-refl _ A-refl (ext-coe q ij w extw)

  uncoe-QEq : ∀ {γ i j} {ij : I S ∋ Γ .ref γ ⊢ i ~ j} {u : |QWI| γ i} {v : |QWI| γ j}
    → QWI~ (Γ .ref γ) ij u v → QWI₀~ γ (u .₁) (v .₁)
  uncoe-QEq {γ} {ij = ij} {u} eq = qwi-trans (ext-coe (Γ .ref γ) ij (u .₁) (u .₂))
    (QWI-coe-id ij (u .₁) (u .₂)) eq

  coe-QEq : ∀ {γ i j} {ij : I S ∋ Γ .ref γ ⊢ i ~ j} {u : |QWI| γ i} {v : |QWI| γ j}
    → QWI₀~ γ (u .₁) (v .₁) → QWI~ (Γ .ref γ) ij u v
  coe-QEq {ij = ij} {u} eq = qwi-trans (u .₂) (qwi-sym (QWI-coe-id ij (u .₁) (u .₂))) eq

  retarget : ∀ {γ δ i j k l} (q : Γ ∋ γ ~ δ)
    (ij : I S ∋ q ⊢ i ~ j) (ik : I S ∋ q ⊢ i ~ k)
    {w : QWI₀ γ i} {v : QWI₀ δ l} → Ext γ w
    → QWI₀~ δ (QWI-coe q ik w) v → QWI₀~ δ (QWI-coe q ij w) v
  retarget q ij ik {w} extw eq = qwi-trans (ext-coe q ik w extw)
    (QWI-coe~ q ij ik (qwi-refl _ A-refl extw)) eq

  sup-QWI : Tm (Γ ▷ I S ▷ A S ▷ Branch S QWI) (QWI [ wk∘wk ])
  ∣ sup-QWI ∣ (((γ , i) , a) , f) = supᵢ₀ a (λ b → f .₁ b .₁)
    , (λ {b} {b'} bb → uncoe-QEq {u = f .₁ b} {v = f .₁ b'} (f .₂ (bb .₂)))
    , λ b → f .₁ b .₂
  sup-QWI .resp {(((γ , i) , a) , f)} (((q , ij) , aa) , ff) = qwi-sup
    (A-trans (A-sym (ij , A S .coh (q , ij) _)) (ij , aa))
    λ {b} {b'} bb → retarget q (ar~ (pull~ q ij b)) (ar~ (B-trans (pull~ q ij b) bb))
      (f .₁ (pull q ij b) .₂) (ff (pull q ij b) b' (B-trans (pull~ q ij b) bb .₂))

  sup-QWIₑ : (a : Tm (Γ ▷ I S) (A S))
    → Tm (Γ ▷ I S) (Branch S QWI ⟨ a ⟩) → Tm (Γ ▷ I S) QWI
  sup-QWIₑ a f = sup-QWI [ idS ▸ a ▸ f ]t

  eval-QWI : {X : Ty (Γ ▷ I S) lzero}
    → Tm (Γ ▷ I S ▷ X) (QWI [ wk ])
    → Tm (Γ ▷ I S ▷ Term S X) (QWI [ wk ])
  ∣ eval-QWI {X} ρ ∣ ((γ , i) , t) = eval-QExt γ (Construction.X~ S X (Γ .ref γ)) t
    (λ {j} x → ∣ ρ ∣ ((γ , j) , x))
    (λ {i} {j} {x} {y} xy → uncoe-QEq
      {u = ∣ ρ ∣ ((γ , i) , x)} {v = ∣ ρ ∣ ((γ , j) , y)}
      (ρ .resp ((Γ .ref γ , xy .₁) , xy .₂)))
  eval-QWI {X} ρ .resp {((γ , i) , t)} {((δ , j) , u)} ((q , ij) , tu) =
    eval-map q (Construction.X~ S X q) ij {t₀ = t .₁} {t₁ = u .₁} tu
      (λ {i} x → ∣ ρ ∣ ((γ , i) , x) .₁) (λ {i} x → ∣ ρ ∣ ((δ , i) , x) .₁)
      (λ {i} {j} {k} ik {x} {y} xy → retarget q ik (xy .₁)
        (∣ ρ ∣ ((γ , i) , x) .₂) (ρ .resp ((q , xy .₁) , xy .₂)))

  eval-QWIₑ : {X : Ty (Γ ▷ I S) lzero}
    → Tm (Γ ▷ I S) (Term S X)
    → Tm (Γ ▷ I S ▷ X) (QWI [ wk ]) → Tm (Γ ▷ I S) QWI
  eval-QWIₑ t ρ = eval-QWI ρ ⟨ t ⟩t

  variable-sort : Sub (Γ ▷ I S ▷ E ε ▷ I S [ wk∘wk ]) (Γ ▷ I S)
  variable-sort = wk∘wk∘wk ▸ vz

  QWI-eq-env : Ty (Γ ▷ I S ▷ E ε) lzero
  QWI-eq-env = Π (I S [ wk∘wk ]) (Π (V ε) (QWI [ variable-sort ∘ wk ]))

  EqContext : Con ℓΓ
  EqContext = Γ ▷ I S ▷ E ε ▷ QWI-eq-env

  equation-sort : Sub EqContext (Γ ▷ I S)
  equation-sort = wk∘wk∘wk ▸ vs (vs vz)

  assignment : ∀ {γ i} (e : ∣ E ε ∣ (γ , i))
    → ∣ QWI-eq-env ∣ ((γ , i) , e) → ∀ {j} → SigAt γ .V e j → |QWI| γ j
  assignment e ρ {j} x = ρ .₁ j .₁ x

  assignment~ : ∀ {γ i} (e : ∣ E ε ∣ (γ , i)) (ρ : ∣ QWI-eq-env ∣ ((γ , i) , e))
    {j k} {x : SigAt γ .V e j} {y : SigAt γ .V e k}
    → SigAt γ .V~ e x y → QWI₀~ γ (assignment e ρ x .₁) (assignment e ρ y .₁)
  assignment~ e ρ {j} {k} {x} {y} (jk , xy) = uncoe-QEq
    {u = assignment e ρ x} {v = assignment e ρ y} (ρ .₂ jk x y xy)

  eval-eq : Tm (Γ ▷ I S ▷ E ε) (Term (S [ wk∘wk ]s) (V ε) [ idS ▸ vs vz ])
    → Tm EqContext (QWI [ equation-sort ])
  ∣ eval-eq t ∣ (((γ , i) , e) , ρ) = eval-QExt γ (SigAt γ .V~ e) (∣ t ∣ ((γ , i) , e))
    (assignment e ρ) (assignment~ e ρ)
  eval-eq t .resp {(((γ , i) , e) , ρ)} {(((δ , j) , e') , ρ')} (((q , ij) , ee) , rr) =
    eval-map q VX ij {t₀ = ∣ t ∣ ((γ , i) , e) .₁} {t₁ = ∣ t ∣ ((δ , j) , e') .₁}
      (t .resp ((q , ij) , ee))
      (λ x → assignment e ρ x .₁) (λ x → assignment e' ρ' x .₁)
      (λ {k} {l} {m} kl {x} {y} (km , xy) → retarget q kl km
        (assignment e ρ x .₂) (rr k m km x y xy))
    where
      VX : ∀ {k l} → SigAt γ .V e k → SigAt δ .V e' l → Prop
      VX {k} {l} x y = ∃[ kl ∈ (I S ∋ q ⊢ k ~ l) ]
        (V ε ∋ (((q , ij) , ee) , kl) ⊢ x ~ y)

  qwi-eq : Tm EqContext (⇑ Eq QWI equation-sort (eval-eq (lhs ε)) (eval-eq (rhs ε)))
  ∣ qwi-eq ∣ (((γ , i) , e) , ρ) = ↑ (coe-QEq
    {u = ∣ eval-eq (lhs ε) ∣ (((γ , i) , e) , ρ)}
    {v = ∣ eval-eq (rhs ε) ∣ (((γ , i) , e) , ρ)}
    (qwi-equ e (assignment e ρ) (assignment~ e ρ)))
  qwi-eq .resp _ = tt

  qwi-eq-at : (e : Tm (Γ ▷ I S) (E ε))
    (ρ : Tm (Γ ▷ I S) (QWI-eq-env ⟨ e ⟩))
    → Tm (Γ ▷ I S) (⇑ Eq QWI idS
      (eval-eq (lhs ε) [ idS ▸ e ▸ ρ ]t) (eval-eq (rhs ε) [ idS ▸ e ▸ ρ ]t))
  qwi-eq-at e ρ = qwi-eq [ idS ▸ e ▸ ρ ]t
