open import Prelude

module Model.Prop.Universe.Lemmas where

open import Model.Prop.Universe.Core
open import Model.Prop.IF.Nat.Core

private
  fromExt : ∀ {A : Set} {Au : InU A} → Ext Au → U₀
  fromExt {A} {Au} Ae = pack A Au Ae

  refl-ext : ∀ {A : Set} {Au : InU A} → Ext Au → (a : A) → ElEq Au Au a a
  refl-ext extNat = N.ref
  refl-ext (extΣ Ae Bext Be) (a , b) =
    refl-ext Ae a , refl-ext (Be a) b
  refl-ext (extΠ Ae Bext Be) (f , fe) = λ _ _ p → fe p
  refl-ext (extId Ae a b) p = tt

  reflU-ext : ∀ {A : Set} {Au : InU A} → (Ae : Ext Au) → CodeEq Au Au
  reflU-ext extNat = tt
  reflU-ext (extΣ Ae Bext Be) =
    reflU-ext Ae , λ _ _ p → Bext _ _ p
  reflU-ext (extΠ Ae Bext Be) =
    reflU-ext Ae , λ _ _ p → Bext _ _ p
  reflU-ext (extId Ae a b) =
    reflU-ext Ae , (refl-ext Ae a , refl-ext Ae b)

  sym-ext
    : ∀ {A B : Set} {Au : InU A} {Bu : InU B}
    → Ext Au → Ext Bu → ∀ {a b}
    → ElEq Au Bu a b → ElEq Bu Au b a
  sym-ext extNat extNat {n} {m} = N.sym {n}
  sym-ext (extΣ Ae Aext Af) (extΣ Be Bext Bf) (ab , pq) =
    sym-ext Ae Be ab , sym-ext (Af _) (Bf _) pq
  sym-ext (extΠ Ae Aext Af) (extΠ Be Bext Bf) fg =
    λ b a ba →
      sym-ext (Af a) (Bf b) (fg a b (sym-ext Be Ae ba))
  sym-ext (extId Ae a₀ a₁) (extId Be b₀ b₁) p = tt

  symU-ext
    : ∀ {A B : Set} {Au : InU A} {Bu : InU B}
    → Ext Au → Ext Bu → CodeEq Au Bu → CodeEq Bu Au
  symU-ext extNat extNat p = tt
  symU-ext (extΣ Ae Aext Af) (extΣ Be Bext Bf) p =
    symU-ext Ae Be (p .₁) ,
    λ b a ba → symU-ext (Af a) (Bf b) (p .₂ a b (sym-ext Be Ae ba))
  symU-ext (extΠ Ae Aext Af) (extΠ Be Bext Bf) p =
    symU-ext Ae Be (p .₁) ,
    λ b a ba → symU-ext (Af a) (Bf b) (p .₂ a b (sym-ext Be Ae ba))
  symU-ext (extId Ae a₀ a₁) (extId Be b₀ b₁) p =
    symU-ext Ae Be (p .₁) ,
    (sym-ext Ae Be (p .₂ .₁) , sym-ext Ae Be (p .₂ .₂))

refl₀ : (A : U₀) → (a : El₀ A) → eq₀ A A a a
refl₀ A = refl-ext (ext₀ A)

reflU₀ : (A : U₀) → eqU₀ A A
reflU₀ A = reflU-ext (ext₀ A)

sym₀
  : (A B : U₀) → ∀ {a b}
  → eq₀ A B a b → eq₀ B A b a
sym₀ A B = sym-ext (ext₀ A) (ext₀ B)

symU₀ : (A B : U₀) → eqU₀ A B → eqU₀ B A
symU₀ A B = symU-ext (ext₀ A) (ext₀ B)

-- Transitivity and coercion have to be constructed together.  The two
-- transitivity fields let the recursive coercions compose heterogeneous
-- equalities in either orientation without ever inspecting a Prop proof to
-- produce data.
record CastKit
  {A B : Set} {Au : InU A} {Bu : InU B}
  (Ae : Ext Au) (Be : Ext Bu) (e : CodeEq Au Bu) : Set₁ where
  field
    transᶠ
      : ∀ {C : Set} {Cu : InU C} → Ext Cu
      → (a : A) (b : B) (c : C)
      → ElEq Au Bu a b → ElEq Bu Cu b c → ElEq Au Cu a c

    transᵇ
      : ∀ {C : Set} {Cu : InU C} → Ext Cu
      → (a : A) (b : B) (c : C)
      → ElEq Bu Au b a → ElEq Au Cu a c → ElEq Bu Cu b c

    castᶠ : A → B
    castᵇ : B → A

    cohᶠ : (a : A) → ElEq Au Bu a (castᶠ a)
    cohᵇ : (b : B) → ElEq Bu Au b (castᵇ b)

open CastKit public

private
  kit
    : ∀ {A B : Set} {Au : InU A} {Bu : InU B}
    → (Ae : Ext Au) (Be : Ext Bu) (e : CodeEq Au Bu)
    → CastKit Ae Be e

  -- Natural numbers
  kit extNat extNat e .transᶠ extNat m n o mn no = N.trans {m} {n} mn no
  kit extNat extNat e .transᵇ extNat m n o nm mo = N.trans {n} nm mo
  kit extNat extNat e .castᶠ n = n
  kit extNat extNat e .castᵇ n = n
  kit extNat extNat e .cohᶠ = N.ref
  kit extNat extNat e .cohᵇ = N.ref

  -- Dependent sums
  kit (extΣ Ae Aext Af) (extΣ Be Bext Bf) e .transᶠ
      (extΣ Ce Cext Cf) (a , p) (b , q) (c , r) (ab , pq) (bc , qr) =
    transᶠ (kit Ae Be (e .₁)) Ce a b c ab bc ,
    transᶠ (kit (Af a) (Bf b) (e .₂ a b ab)) (Cf c) p q r pq qr

  kit (extΣ Ae Aext Af) (extΣ Be Bext Bf) e .transᵇ
      (extΣ Ce Cext Cf) (a , p) (b , q) (c , r) (ba , qp) (ac , pr) =
    transᵇ (kit Ae Be (e .₁)) Ce a b c ba ac ,
    transᵇ
      (kit (Af a) (Bf b) (e .₂ a b (sym-ext Be Ae ba)))
      (Cf c) p q r qp pr

  kit (extΣ Ae Aext Af) (extΣ Be Bext Bf) e .castᶠ (a , p) =
    let K  = kit Ae Be (e .₁)
        b  = castᶠ K a
        ab = cohᶠ K a
    in b , castᶠ (kit (Af a) (Bf b) (e .₂ a b ab)) p
  kit (extΣ Ae Aext Af) (extΣ Be Bext Bf) e .castᵇ (b , q) =
    let K  = kit Ae Be (e .₁)
        a  = castᵇ K b
        ba = cohᵇ K b
        ab = sym-ext Be Ae ba
    in a , castᵇ (kit (Af a) (Bf b) (e .₂ a b ab)) q
  kit (extΣ Ae Aext Af) (extΣ Be Bext Bf) e .cohᶠ (a , p) =
    let K  = kit Ae Be (e .₁)
        b  = castᶠ K a
        ab = cohᶠ K a
        L  = kit (Af a) (Bf b) (e .₂ a b ab)
    in ab , cohᶠ L p
  kit (extΣ Ae Aext Af) (extΣ Be Bext Bf) e .cohᵇ (b , q) =
    let K  = kit Ae Be (e .₁)
        a  = castᵇ K b
        ba = cohᵇ K b
        ab = sym-ext Be Ae ba
        L  = kit (Af a) (Bf b) (e .₂ a b ab)
    in ba , cohᵇ L q

  -- Dependent functions
  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .transᶠ
      (extΠ Ce Cext Cf) (f , fe) (g , ge) (h , he) fg gh = λ a c ac →
    let K   = kit Ae Be (e .₁)
        b   = castᶠ K a
        ab  = cohᶠ K a
        bc  = transᵇ K Ce a b c (sym-ext Ae Be ab) ac
        L   = kit (Af a) (Bf b) (e .₂ a b ab)
    in transᶠ L (Cf c) (f a) (g b) (h c) (fg a b ab) (gh b c bc)
  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .transᶠ extNat x y z fg ()
  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .transᶠ (extΣ Ce Cext Cf) x y z fg ()
  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .transᶠ (extId Ce c₀ c₁) x y z fg ()

  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .transᵇ
      (extΠ Ce Cext Cf) (f , fe) (g , ge) (h , he) gf fh = λ b c bc →
    let K   = kit Ae Be (e .₁)
        a   = castᵇ K b
        ba  = cohᵇ K b
        ab  = sym-ext Be Ae ba
        ac  = transᶠ K Ce a b c ab bc
        L   = kit (Af a) (Bf b) (e .₂ a b ab)
    in transᵇ L (Cf c) (f a) (g b) (h c) (gf b a ba) (fh a c ac)
  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .transᵇ extNat x y z gf ()
  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .transᵇ (extΣ Ce Cext Cf) x y z gf ()
  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .transᵇ (extId Ce c₀ c₁) x y z gf ()

  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .castᶠ (f , fe) =
    (λ b →
      let K⁻ = kit Be Ae (symU-ext Ae Be (e .₁))
          a  = castᶠ K⁻ b
          ba = cohᶠ K⁻ b
          ab = sym-ext Be Ae ba
      in castᶠ (kit (Af a) (Bf b) (e .₂ a b ab)) (f a)) ,
    λ {b₀} {b₁} bb →
      let K     = kit Ae Be (e .₁)
          K⁻    = kit Be Ae (symU-ext Ae Be (e .₁))
          a₀    = castᶠ K⁻ b₀
          a₁    = castᶠ K⁻ b₁
          b₀a₀  = cohᶠ K⁻ b₀
          b₁a₁  = cohᶠ K⁻ b₁
          a₀b₀  = sym-ext Be Ae b₀a₀
          a₁b₁  = sym-ext Be Ae b₁a₁
          a₁b₀  = transᶠ K Be a₁ b₁ b₀ a₁b₁ (sym-ext Be Be bb)
          a₀a₁  = transᶠ K Ae a₀ b₀ a₁ a₀b₀ (sym-ext Ae Be a₁b₀)
          ff   = fe a₀a₁
          L₀   = kit (Af a₀) (Bf b₀) (e .₂ a₀ b₀ a₀b₀)
          L₁   = kit (Af a₁) (Bf b₁) (e .₂ a₁ b₁ a₁b₁)
          L₁₀  = kit (Af a₁) (Bf b₀) (e .₂ a₁ b₀ a₁b₀)
          g₀    = castᶠ L₀ (f a₀)
          g₁    = castᶠ L₁ (f a₁)
          b₀a₁f = transᵇ L₀ (Af a₁) (f a₀) g₀ (f a₁)
                     (sym-ext (Af a₀) (Bf b₀) (cohᶠ L₀ (f a₀))) ff
      in transᵇ L₁₀ (Bf b₁) (f a₁) g₀ g₁
           b₀a₁f (cohᶠ L₁ (f a₁))

  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .castᵇ (g , ge) =
    (λ a →
      let K  = kit Ae Be (e .₁)
          b  = castᶠ K a
          ab = cohᶠ K a
      in castᵇ (kit (Af a) (Bf b) (e .₂ a b ab)) (g b)) ,
    λ {a₀} {a₁} aa →
      let K    = kit Ae Be (e .₁)
          b₀   = castᶠ K a₀
          b₁   = castᶠ K a₁
          a₀b₀  = cohᶠ K a₀
          a₁b₁  = cohᶠ K a₁
          b₀a₀  = sym-ext Ae Be a₀b₀
          b₀a₁  = transᵇ K Ae a₀ b₀ a₁ b₀a₀ aa
          b₀b₁  = transᵇ K Be a₁ b₀ b₁ b₀a₁ a₁b₁
          gg   = ge b₀b₁
          L₀   = kit (Af a₀) (Bf b₀) (e .₂ a₀ b₀ a₀b₀)
          L₁   = kit (Af a₁) (Bf b₁) (e .₂ a₁ b₁ a₁b₁)
          L₁₀  = kit (Af a₁) (Bf b₀)
                    (e .₂ a₁ b₀ (sym-ext Be Ae b₀a₁))
          f₀    = castᵇ L₀ (g b₀)
          f₁    = castᵇ L₁ (g b₁)
          a₁b₀g = transᶠ L₁ (Bf b₀) f₁ (g b₁) (g b₀)
                     (sym-ext (Bf b₁) (Af a₁) (cohᵇ L₁ (g b₁)))
                     (sym-ext (Bf b₀) (Bf b₁) gg)
      in sym-ext (Af a₁) (Af a₀)
           (transᶠ L₁₀ (Af a₀) f₁ (g b₀) f₀
             a₁b₀g (cohᵇ L₀ (g b₀)))

  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .cohᶠ (f , fe) = λ a b ab →
    let K    = kit Ae Be (e .₁)
        K⁻   = kit Be Ae (symU-ext Ae Be (e .₁))
        a′   = castᶠ K⁻ b
        ba′  = cohᶠ K⁻ b
        a′b  = sym-ext Be Ae ba′
        aa′  = transᶠ K Ae a b a′ ab ba′
        ff   = fe aa′
        L′   = kit (Af a′) (Bf b) (e .₂ a′ b a′b)
    in sym-ext (Bf b) (Af a)
         (transᵇ L′ (Af a) (f a′) (castᶠ L′ (f a′)) (f a)
           (sym-ext (Af a′) (Bf b) (cohᶠ L′ (f a′)))
           (sym-ext (Af a) (Af a′) ff))
  kit (extΠ Ae Aext Af) (extΠ Be Bext Bf) e .cohᵇ (g , ge) = λ b a ba →
    let K    = kit Ae Be (e .₁)
        b′   = castᶠ K a
        ab′  = cohᶠ K a
        b′a  = sym-ext Ae Be ab′
        bb′  = transᵇ K Be a b b′ ba ab′
        gg   = ge bb′
        L′   = kit (Af a) (Bf b′) (e .₂ a b′ ab′)
    in sym-ext (Af a) (Bf b)
         (transᶠ L′ (Bf b) (castᵇ L′ (g b′)) (g b′) (g b)
           (sym-ext (Bf b′) (Af a) (cohᵇ L′ (g b′)))
           (sym-ext (Bf b) (Bf b′) gg))

  -- Identity types.  Their element equality is proof-irrelevant, so only
  -- coercion has computational content.
  kit (extId Ae a₀ a₁) (extId Be b₀ b₁) e .transᶠ (extId Ce c₀ c₁) x y z p q = tt
  kit (extId Ae a₀ a₁) (extId Be b₀ b₁) e .transᵇ (extId Ce c₀ c₁) x y z p q = tt
  kit (extId Ae a₀ a₁) (extId Be b₀ b₁) e .castᶠ p =
    let K    = kit Ae Be (e .₁)
        a₀b₀ = e .₂ .₁
        a₁b₁ = e .₂ .₂
        a₀a₁ = FId→R (refl-ext Ae) p
        b₁a₀ = transᵇ K Ae a₁ b₁ a₀
                 (sym-ext Ae Be a₁b₁) (sym-ext Ae Ae a₀a₁)
        a₀b₁ = sym-ext Be Ae b₁a₀
        b₀b₁ = transᵇ K Be a₀ b₀ b₁ (sym-ext Ae Be a₀b₀) a₀b₁
    in ford b₀b₁
  kit (extId Ae a₀ a₁) (extId Be b₀ b₁) e .castᵇ p =
    let K    = kit Ae Be (e .₁)
        a₀b₀ = e .₂ .₁
        a₁b₁ = e .₂ .₂
        b₀b₁ = FId→R (refl-ext Be) p
        a₀b₁ = transᶠ K Be a₀ b₀ b₁ a₀b₀ b₀b₁
        b₁a₁ = sym-ext Ae Be a₁b₁
        a₀a₁ = transᶠ K Ae a₀ b₁ a₁ a₀b₁ b₁a₁
    in ford a₀a₁
  kit (extId Ae a₀ a₁) (extId Be b₀ b₁) e .cohᶠ p = tt
  kit (extId Ae a₀ a₁) (extId Be b₀ b₁) e .cohᵇ p = tt

  transU-ext
    : ∀ {A B C : Set} {Au : InU A} {Bu : InU B} {Cu : InU C}
    → Ext Au → Ext Bu → Ext Cu
    → CodeEq Au Bu → CodeEq Bu Cu → CodeEq Au Cu
  transU-ext extNat extNat extNat eAB eBC = tt
  transU-ext (extΣ Ae Aext Af) (extΣ Be Bext Bf) (extΣ Ce Cext Cf) eAB eBC =
    let K = kit Ae Be (eAB .₁)
    in transU-ext Ae Be Ce (eAB .₁) (eBC .₁) , λ a c ac →
      let b  = castᶠ K a
          ab = cohᶠ K a
          bc = transᵇ K Ce a b c (sym-ext Ae Be ab) ac
      in transU-ext (Af a) (Bf b) (Cf c)
           (eAB .₂ a b ab) (eBC .₂ b c bc)
  transU-ext (extΠ Ae Aext Af) (extΠ Be Bext Bf) (extΠ Ce Cext Cf) eAB eBC =
    let K = kit Ae Be (eAB .₁)
    in transU-ext Ae Be Ce (eAB .₁) (eBC .₁) , λ a c ac →
      let b  = castᶠ K a
          ab = cohᶠ K a
          bc = transᵇ K Ce a b c (sym-ext Ae Be ab) ac
      in transU-ext (Af a) (Bf b) (Cf c)
           (eAB .₂ a b ab) (eBC .₂ b c bc)
  transU-ext (extId Ae a₀ a₁) (extId Be b₀ b₁) (extId Ce c₀ c₁) eAB eBC =
    let K = kit Ae Be (eAB .₁)
    in transU-ext Ae Be Ce (eAB .₁) (eBC .₁) ,
       (transᶠ K Ce a₀ b₀ c₀ (eAB .₂ .₁) (eBC .₂ .₁) ,
        transᶠ K Ce a₁ b₁ c₁ (eAB .₂ .₂) (eBC .₂ .₂))

castKit₀ : (A B : U₀) → (e : eqU₀ A B) → CastKit (ext₀ A) (ext₀ B) e
castKit₀ A B = kit (ext₀ A) (ext₀ B)

trans₀
  : (A B : U₀) → eqU₀ A B → (C : U₀)
  → ∀ {a b c} → eq₀ A B a b → eq₀ B C b c → eq₀ A C a c
trans₀ A B e C {a} {b} {c} =
  transᶠ (castKit₀ A B e) (ext₀ C) a b c

cast₀ : (A B : U₀) → eqU₀ A B → El₀ A → El₀ B
cast₀ A B e = castᶠ (castKit₀ A B e)

cast₀-eq
  : (A B : U₀) → (e : eqU₀ A B) → (a : El₀ A)
  → eq₀ A B a (cast₀ A B e a)
cast₀-eq A B e = cohᶠ (castKit₀ A B e)

cast₀-back : (A B : U₀) → eqU₀ A B → El₀ B → El₀ A
cast₀-back A B e = castᵇ (castKit₀ A B e)

cast₀-back-eq
  : (A B : U₀) → (e : eqU₀ A B) → (b : El₀ B)
  → eq₀ B A b (cast₀-back A B e b)
cast₀-back-eq A B e = cohᵇ (castKit₀ A B e)

transU₀ : ∀ {A B C : U₀} → eqU₀ A B → eqU₀ B C → eqU₀ A C
transU₀ {A} {B} {C} = transU-ext (ext₀ A) (ext₀ B) (ext₀ C)
