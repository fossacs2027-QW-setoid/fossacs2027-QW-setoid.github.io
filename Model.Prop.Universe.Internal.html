-- The universe is a constant type.  Its equality has to be lifted once
-- because U₀ : Set₁ while its structural equality is deliberately small.
U : Ty Γ (lsuc lzero)
∣ U ∣ _ = U₀
U .heq _ A B = LiftP (lsuc lzero) (eqU₀ A B)
U .refl* A = ↑ (reflU₀ A)
U .sym* A B p = ↑ (symU₀ A B (p .↓))
U .trans* A B C p q =
  ↑ (transU₀ {A = A} {B = B} {C = C} (p .↓) (q .↓))
U .coe _ A = A
U .coh _ A = ↑ (reflU₀ A)

U[] : U [ σ ] ≡ U
U[] = refl

El : Tm Γ U → Ty Γ lzero
∣ El A ∣ γ = El₀ (∣ A ∣ γ)
El A .heq {γ₀} {γ₁} γ₀₁ = eq₀ (∣ A ∣ γ₀) (∣ A ∣ γ₁)
El A .refl* {γ} a = refl₀ (∣ A ∣ γ) a
El A .sym* {γ₀} {γ₁} _ _ p = sym₀ (∣ A ∣ γ₀) (∣ A ∣ γ₁) p
El A .trans* {γ₀} {γ₁} {γ₂} {γ₀₁} _ _ _ p q =
  trans₀ (∣ A ∣ γ₀) (∣ A ∣ γ₁) (A .resp γ₀₁ .↓) (∣ A ∣ γ₂) p q
El A .coe {γ₀} {γ₁} γ₀₁ a =
  cast₀ (∣ A ∣ γ₀) (∣ A ∣ γ₁) (A .resp γ₀₁ .↓) a
El A .coh {γ₀} {γ₁} γ₀₁ a =
  cast₀-eq (∣ A ∣ γ₀) (∣ A ∣ γ₁) (A .resp γ₀₁ .↓) a

El[] : {A : Tm Γ U} → El A [ σ ] ≡ El (A [ σ ]t)
El[] = refl

natᵁ : Tm Γ U
∣ natᵁ ∣ _ = nat₀
natᵁ .resp _ = ↑ (reflU₀ nat₀)

sigmaᵁ : (A : Tm Γ U) → Tm (Γ ▷ El A) U → Tm Γ U
∣ sigmaᵁ { Γ = Γ } A B ∣ γ =
  sigma₀ (∣ A ∣ γ) (λ a → ∣ B ∣ (γ , a))
    (λ p → B .resp (Γ .ref γ , p) .↓)
sigmaᵁ A B .resp γ₀₁ = ↑
  (A .resp γ₀₁ .↓ , λ a₀ a₁ a₀₁ → B .resp (γ₀₁ , a₀₁) .↓)

piᵁ : (A : Tm Γ U) → Tm (Γ ▷ El A) U → Tm Γ U
∣ piᵁ { Γ = Γ } A B ∣ γ =
  pi₀ (∣ A ∣ γ) (λ a → ∣ B ∣ (γ , a))
    (λ p → B .resp (Γ .ref γ , p) .↓)
piᵁ A B .resp γ₀₁ = ↑
  (A .resp γ₀₁ .↓ , λ a₀ a₁ a₀₁ → B .resp (γ₀₁ , a₀₁) .↓)

idᵁ : (A : Tm Γ U) → Tm Γ (El A) → Tm Γ (El A) → Tm Γ U
∣ idᵁ A a b ∣ γ = id₀ (∣ A ∣ γ) (∣ a ∣ γ) (∣ b ∣ γ)
idᵁ A a b .resp γ₀₁ = ↑
  (A .resp γ₀₁ .↓ , (a .resp γ₀₁ , b .resp γ₀₁))

natᵁ[] : natᵁ [ σ ]t ≡ natᵁ
natᵁ[] = refl

sigmaᵁ[]
  : {A : Tm Γ U} {B : Tm (Γ ▷ El A) U}
  → sigmaᵁ A B [ σ ]t
  ≡ sigmaᵁ (A [ σ ]t) (B [ σ ⁺ ]t)
sigmaᵁ[] = refl

piᵁ[]
  : {A : Tm Γ U} {B : Tm (Γ ▷ El A) U}
  → piᵁ A B [ σ ]t
  ≡ piᵁ (A [ σ ]t) (B [ σ ⁺ ]t)
piᵁ[] = refl

idᵁ[]
  : {A : Tm Γ U} {a b : Tm Γ (El A)}
  → idᵁ A a b [ σ ]t
  ≡ idᵁ (A [ σ ]t) (a [ σ ]t) (b [ σ ]t)
idᵁ[] = refl
