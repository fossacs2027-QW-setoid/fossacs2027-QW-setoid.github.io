QW-step : {Γ : Con ℓΓ} {A : Ty Γ lzero} (B : Ty (Γ ▷ A) lzero)
  (ε : EqSys A B) (P : Ty (Γ ▷ QW A B ε) ℓ) → Set _
QW-step {Γ = Γ} {A = A} B ε P =
  let Γ₀ = Γ ▷ A ▷ B ⇒ QW A B ε [ wk ]
  in Tm (Γ₀ ▷ Π (B [ wk ])
      (P [ wk∘wk∘wk ▸ app {Γ = Γ₀ ▷ B [ wk ]}
        {A = B [ wk∘wk ]} {B = QW A B ε [ wk∘wk∘wk∘wk ]}
        (vs vz) ⟨ vz ⟩t ]))
    (P [ wk∘wk∘wk ▸
      (sup-QW {Γ = Γ₀} (A [ wk∘wk ]) (B [ wk∘wk ⁺ ]) (ε [ wk∘wk ]ε)
        (vs vz) vz) [ wk ]t ])

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero}
         (B : Ty (Γ ▷ A) lzero)
         (ε : EqSys A B)
         (P : Ty (Γ ▷ QW A B ε) ℓ)
  where

  assignment-base : {X : Ty Γ lzero} → Tm (Γ ▷ X) (Sigma (QW A B ε) P [ wk ])
    → Tm (Γ ▷ X) (QW A B ε [ wk ])
  assignment-base ρ = fstₑ (QW A B ε [ wk ]) (P [ wk ⁺ ]) ρ

  QW-elim-env : Ty (Γ ▷ E ε) ℓ
  QW-elim-env = V ε ⇒ Sigma (QW A B ε) P [ wk ]

module FiberConstruction {Γ : Con ℓΓ} {A : Ty Γ lzero}
  (B : Ty (Γ ▷ A) lzero) (ε : EqSys A B)
  (P : Ty (Γ ▷ QW A B ε) ℓ) (p : QW-step B ε P) where

  P-at : (γ : ∣ Γ ∣) (w : QW₀ A B γ)
    → Ext A B ε γ w → Set ℓ
  P-at γ w extw = ∣ P ∣ (γ , (w , extw))

  P~-at
    : (γ : ∣ Γ ∣) {w₀ w₁ : QW₀ A B γ}
    → QW₀~ A B ε γ w₀ w₁
    → {extw₀ : Ext A B ε γ w₀} {extw₁ : Ext A B ε γ w₁}
    → P-at γ w₀ extw₀ → P-at γ w₁ extw₁ → Prop ℓ
  P~-at γ w₀₁ {extw₀} {extw₁} = P .heq (Γ .ref γ , coe-QEq A B ε {u = _ , extw₀} {v = _ , extw₁} w₀₁)

  private
    module Steps (γ : ∣ Γ ∣) = StepTypes
      (A .heq (Γ .ref γ)) (B~ A B (Γ .ref γ)) (SigAt A B ε γ)
      (P-at γ) (P~-at γ)

  alg-at : (γ : ∣ Γ ∣) → Steps.Step γ
  alg-at γ {a} {f} extf f~ ih ih~ = ∣ p ∣
    (((γ , a) , ((λ b → f b , extf b)
      , λ {b₀} {b₁} b₀₁ →
          coe-QEq A B ε {u = _ , extf b₀} {v = _ , extf b₁} (f~ (A .refl* a , b₀₁))))
      , (ih , λ {b₀} {b₁} b₀₁ → ih~ (A .refl* a , b₀₁)))

  alg-at~ : (γ : ∣ Γ ∣) → Steps.Step~ γ (alg-at γ)
  alg-at~ γ f₀~ extf₀ ih₀ ih₀~ f₁~ extf₁ ih₁ ih₁~ a₀₁ f₀₁ ih₀₁ =
    p .resp (((Γ .ref γ , a₀₁)
             , λ b₀ b₁ b₀₁ →
                 coe-QEq A B ε {u = _ , extf₀ b₀} {v = _ , extf₁ b₁} (f₀₁ (a₀₁ , b₀₁)))
             , λ b₀ b₁ b₀₁ → ih₀₁ (a₀₁ , b₀₁))

  module Fiber (γ : ∣ Γ ∣) =
    QW-elim₀
      (A .heq (Γ .ref γ)) (A .refl*) (λ {a₀} {a₁} → A .sym* a₀ a₁)
      (B~ A B (Γ .ref γ))
      (λ (aa , bb) → A .sym* _ _ aa , B .sym* _ _ bb)
      (λ (aa , bb) (aa' , bb') → A .trans* _ _ _ aa aa' , B .trans* _ _ _ bb bb')
      (λ aa b → B .coe (Γ .ref γ , aa) b)
      (λ aa b → aa , B .coh (Γ .ref γ , aa) b)
      (SigAt A B ε γ)
      (λ e x → V ε .refl* {γ , e} x)
      (λ e → ∣ lhs ε ∣ (γ , e) .₂) (λ e → ∣ rhs ε ∣ (γ , e) .₂)
      (QW A B ε ∋ Γ .ref γ ⊢_~_)
      (λ {u} {v} → uncoe-QEq A B ε {u = u} {v = v})
      (λ {u} {v} → coe-QEq A B ε {u = u} {v = v})
      (P-at γ) (P~-at γ)
      (λ w extw v → P .refl* v)
      (λ {v₀ = x} {v₁ = y} → P .sym* x y) (λ {v₀ = x} {v₁ = y} {v₂ = z} → P .trans* x y z)
      (alg-at γ) (alg-at~ γ)

deval : {Γ : Con ℓΓ} {A : Ty Γ lzero}
  (B : Ty (Γ ▷ A) lzero) (ε : EqSys A B)
  (P : Ty (Γ ▷ QW A B ε) ℓ) (p : QW-step B ε P)
  {X : Ty Γ lzero} (t : Tm Γ (Term A B X))
  → (ρ : Tm (Γ ▷ X) (Sigma (QW A B ε) P [ wk ]))
  → Tm Γ (P ⟨ eval-QW A B ε t (assignment-base B ε P ρ) ⟩)
deval {Γ = Γ} {A} B ε P p {X} t ρ = Free.elim-Termₑ A B X
  (P [ wk ▸ eval-QW (A [ wk ]) (B [ wk ⁺ ]) (ε [ wk ]ε)
    {X = X [ wk ]} vz (assignment-base B ε P ρ [ wk ⁺ ]t) ])
  (sndₑ (QW A B ε [ wk ]) (P [ wk ⁺ ]) ρ)
  (p [ wk∘wk∘wk ▸ vs (vs vz) ▸ subterm [ wk ]t ▸ vz ]t) t
  where
    subterm : Tm (Γ ▷ A ▷ B ⇒ Term A B X [ wk ]) (B [ wk ] ⇒ QW A B ε [ wk∘wk ])
    subterm = ƛ: (B [ wk ])
      (eval-QW (A [ wk∘wk∘wk ]) (B [ wk∘wk∘wk ⁺ ])
        (ε [ wk∘wk∘wk ]ε) {X = X [ wk∘wk∘wk ]}
        (appₑ' {A = B [ wk∘wk ]} {B = Term A B X [ wk∘wk∘wk ]} (vs vz) vz)
        (assignment-base B ε P ρ [ wk∘wk∘wk ⁺ ]t))

{-# NOINLINE deval #-}

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero}
         (B : Ty (Γ ▷ A) lzero)
         (ε : EqSys A B)
         (P : Ty (Γ ▷ QW A B ε) ℓ)
         (p : QW-step B ε P)
  where

  private
    QW-elim-eq-model : Prp (Γ ▷ E ε ▷ QW-elim-env B ε P) ℓ
    QW-elim-eq-model = EqT {Γ = Γeq} P (wk∘wk ▸ left) (wk∘wk ▸ right)
      (eqS-▸ {Γ = Γeq} (QW A B ε) wk∘wk wk∘wk left right (reflS wk∘wk)
        (qw-eq A B ε [ wk ▸ ƛ: X {B = QW A B ε [ wk∘wk∘wk ]} ρ₀ ]t))
      (deval {Γ = Γeq} (B [ wk∘wk ⁺ ]) (ε [ wk∘wk ]ε) (P [ wk∘wk ⁺ ])
        (p [ wk∘wk ⁺ ⁺ ⁺ ]t) {X = X} (lhs ε [ wk ]t) ρ)
      (deval {Γ = Γeq} (B [ wk∘wk ⁺ ]) (ε [ wk∘wk ]ε) (P [ wk∘wk ⁺ ])
        (p [ wk∘wk ⁺ ⁺ ⁺ ]t) {X = X} (rhs ε [ wk ]t) ρ)
      where
        Γeq : Con _
        Γeq = Γ ▷ E ε ▷ QW-elim-env B ε P

        X : Ty Γeq lzero
        X = V ε [ wk ]

        ρ : Tm (Γeq ▷ X) (Sigma (QW A B ε) P [ wk∘wk∘wk ])
        ρ = app vz

        ρ₀ : Tm (Γeq ▷ X) (QW A B ε [ wk∘wk∘wk ])
        ρ₀ = assignment-base {Γ = Γeq} (B [ wk∘wk ⁺ ]) (ε [ wk∘wk ]ε) (P [ wk∘wk ⁺ ]) ρ

        left right : Tm Γeq (QW A B ε [ wk∘wk ])
        left = eval-QW {Γ = Γeq} (A [ wk∘wk ]) (B [ wk∘wk ⁺ ]) (ε [ wk∘wk ]ε)
          {X = X} (lhs ε [ wk ]t) ρ₀
        right = eval-QW {Γ = Γeq} (A [ wk∘wk ]) (B [ wk∘wk ⁺ ]) (ε [ wk∘wk ]ε)
          {X = X} (rhs ε [ wk ]t) ρ₀

  -- Use the context-free predicate for conversion under substitution; the
  -- contextual expression above supplies its transport proof.
  QW-elim-eq : Prp (Γ ▷ E ε ▷ QW-elim-env B ε P) ℓ
  ∣ QW-elim-eq ∣ ((γ , e) , ρ) =
    FiberConstruction.Fiber.QW-elim₀-eq B ε P p γ e
      (λ x → ρ .₁ x .₁)
      (λ {x} {y} xy → uncoe-QEq A B ε
        {u = ρ .₁ x .₁} {v = ρ .₁ y .₁} (ρ .₂ xy .₁))
      (λ x → ρ .₁ x .₂) (λ xy → ρ .₂ xy .₂)
  QW-elim-eq .coe {γ₀} {γ₁} q eq = QW-elim-eq-model .coe {γ₀} {γ₁} q eq

  QW-elim-coh : Set _
  QW-elim-coh = Tm Γ (Π (E ε) (Π (QW-elim-env B ε P) (⇑ QW-elim-eq))) -- QW-coherence
  {-# NOINLINE QW-elim-coh #-}

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero}
         (B : Ty (Γ ▷ A) lzero)
         (ε : EqSys A B)
         (P : Ty (Γ ▷ QW A B ε) ℓ)
         (p : QW-step B ε P)
         (equ-p : QW-elim-coh B ε P p)
  where

  open FiberConstruction B ε P p public

  private
    fiber-equ-p : (γ : ∣ Γ ∣) → Fiber.QW-elim₀-coh γ
    fiber-equ-p γ e ρ ρ~ ρP ρP~ =
      LiftS.↓ (∣ equ-p ∣ γ .₁ e .₁
        ((λ x → ρ x , ρP x)
          , λ {x} {y} xy → coe-QEq A B ε {u = ρ x} {v = ρ y} (ρ~ xy) , ρP~ xy))

  elim-at
    : (γ : ∣ Γ ∣) (w : QW₀ A B γ)
    → (extw : Ext A B ε γ w)
    → ∣ P ∣ (γ , (w , extw))
  elim-at γ = Fiber.elim-QW₀ γ (fiber-equ-p γ)

  elim-at~
    : (γ : ∣ Γ ∣) {w₀ w₁ : QW₀ A B γ}
    → (w₀₁ : QW₀~ A B ε γ w₀ w₁)
    → (extw₀ : Ext A B ε γ w₀)
    → (extw₁ : Ext A B ε γ w₁)
    → P ∋ (Γ .ref γ , coe-QEq A B ε {u = _ , extw₀} {v = _ , extw₁} w₀₁)
        ⊢ elim-at γ w₀ extw₀ ~ elim-at γ w₁ extw₁
  elim-at~ γ = Fiber.elim-QW₀~ γ (fiber-equ-p γ)

  elim-at-coe
    : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → (w : QW₀ A B γ₀) (extw : Ext A B ε γ₀ w)
    → P ∋ (γ₀₁ , QW A B ε .coh γ₀₁ (w , extw))
        ⊢ elim-at γ₀ w extw
        ~ elim-at γ₁ (QW-coe A B ε γ₀₁ w)
            (ext-coe A B ε γ₀₁ w extw)
  elim-at-coe {γ₁ = γ₁} γ₀₁ (sup₀ a f) (f~ , extf) =
    p .resp
      (((γ₀₁ , A .coh γ₀₁ a)
        , λ b₀ b₁ b₀₁ → QW-coe~ A B ε γ₀₁ (f~ (branch-eq b₀₁)))
       , λ b₀ b₁ b₀₁ → P .trans* _ _ _
           (elim-at-coe γ₀₁ (f b₀) (extf b₀))
           (elim-at~ γ₁
             (QW-coe~ A B ε γ₀₁ (f~ (branch-eq b₀₁)))
             (transported-ext b₀) (transported-ext (pull b₁))))
    where
      back = Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a)
      pull = B .coe back
      branch-eq : ∀ {b₀ b₁} → B ∋ (γ₀₁ , A .coh γ₀₁ a) ⊢ b₀ ~ b₁
        → B~ A B (Γ .ref _) b₀ (pull b₁)
      branch-eq b₀₁ = A .refl* a , B .trans* _ _ _ b₀₁ (B .coh back _)
      transported-ext = λ b → ext-coe A B ε γ₀₁ (f b) (extf b)

  elim-QW : Tm (Γ ▷ QW A B ε) P
  ∣ elim-QW ∣ (γ , (w , extw)) = elim-at γ w extw
  elim-QW .resp {γ₀ , (w₀ , extw₀)} {γ₁ , (w₁ , extw₁)}
    (γ₀₁ , w₀₁) =
    P .trans* _ _ _
      (elim-at-coe γ₀₁ w₀ extw₀)
      (elim-at~ γ₁ w₀₁
        (ext-coe A B ε γ₀₁ w₀ extw₀) extw₁)

  elim-QWᵢ : Tm Γ (Π (QW A B ε) P)
  elim-QWᵢ = ƛ elim-QW
