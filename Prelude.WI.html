module Prelude.WI where

open import Prelude.Base
open import Prelude.Proof-Irrelevant-Identity

module _ {I : Set} (A : I → Set) (B : ∀ {i} → A i → Set) (ar : ∀ {i n} → B {i} n → I) where

  data 𝒟𝒲 : I → Set where
    dsup : ∀ {i} (x : A i) (f : ∀ {j} (y : B {i} x) → j ≡ ar y → 𝒟𝒲 j)
      → 𝒟𝒲 i

  module _
    (P : ∀ {i} → 𝒟𝒲 i → Set ℓ)
    (dsupᵖ : ∀ {i} {x : A i} {f : ∀ {j} (y : B x) → j ≡ ar y → 𝒟𝒲 j}
      → (∀ {j} (y : B x) (p : j ≡ ar y) → P (f y p)) → P (dsup x f))
    where

    rec : ∀ {i} (t : 𝒟𝒲 i) → P t
    rec (dsup x f) = dsupᵖ λ y p → rec (f y p)

module _ {I₀ I₁ : Set} {A₀ : I₀ → Set} {A₁ : I₁ → Set}
  {B₀ : ∀ {i} → A₀ i → Set} {B₁ : ∀ {i} → A₁ i → Set}
  {ar₀ : ∀ {i n} → B₀ {i} n → I₀} {ar₁ : ∀ {i n} → B₁ {i} n → I₁}
  where

  map
    : (u : I₀ → I₁)
    → (h : ∀ {i₀ i₁} → i₁ ≡ u i₀ → A₀ i₀ → A₁ i₁)
    → (g : ∀ {i₀ i₁} (p : i₁ ≡ u i₀) {x : A₀ i₀}
         → B₁ (h p x) → B₀ x)
    → (comm : ∀ {i₀ i₁} (p : i₁ ≡ u i₀) {x : A₀ i₀}
         → {y : B₁ (h p x)} → ar₁ y ≡ u (ar₀ (g p y)))
    → ∀ {i₀ i₁} → i₁ ≡ u i₀
    → 𝒟𝒲 A₀ B₀ ar₀ i₀ → 𝒟𝒲 A₁ B₁ ar₁ i₁
  map u h g comm p (dsup x f) =
    dsup (h p x) λ y q →
      map u h g comm (q ∙ (comm p)) (f (g p y) refl)
