open import Prelude

module Model.Prop.QW.Free.Rec where

open import Model.Prop.CwF
open import Model.Prop.Pi
open import Model.Prop.QW.Free.Core
import Model.Prop.QW.Free.RawElim as Raw
import Model.Prop.QW.Free.Elim as Free

open Term-rel using (Term₀~)

-- Steps retain both the original children and their recursive results.
Term-rec-var-step : (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero)
  (X : Ty Γ lzero) (C : Ty Γ ℓ) → Set _
Term-rec-var-step A B X C = Free.Var-step A B X (C [ wk ])

Term-rec-op-step : (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero)
  (X : Ty Γ lzero) (C : Ty Γ ℓ) → Set _
Term-rec-op-step A B X C = Free.Op-step A B X (C [ wk ])

module _ {Γ : Con ℓΓ} (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero)
  (X : Ty Γ lzero) (C : Ty Γ ℓ)
  (v : Term-rec-var-step A B X C)
  (p : Term-rec-op-step A B X C) where

  open Term-rel using (Term₀~)

  rec-Term₀ : (γ : ∣ Γ ∣)
    → (t : Term₀ (∣ A ∣ γ) (λ a → ∣ B ∣ (γ , a)) (∣ X ∣ γ))
    → (extt : extTerm (A ∋ Γ .ref γ ⊢_~_) (B~ A B (Γ .ref γ))
        (X ∋ Γ .ref γ ⊢_~_) t)
    → ∣ C ∣ γ
  rec-Term₀ γ = Raw.Elim₀.val
    (A ∋ Γ .ref γ ⊢_~_) (B~ A B (Γ .ref γ)) (X ∋ Γ .ref γ ⊢_~_)
    (λ t → ∣ C ∣ γ) (λ tt' → C .heq (Γ .ref γ))
    (λ x → ∣ v ∣ (γ , x)) (λ xy → v .resp (Γ .ref γ , xy))
    (λ a f f~ ih ih~ → ∣ p ∣
      (((γ , a) , (f , λ bb → f~ (A .refl* a , bb)))
        , ih , λ bb → ih~ (A .refl* a , bb)))
    (λ ih~ ih'~ aa ff ii → p .resp
      (((Γ .ref γ , aa) , λ b b' bb → ff (aa , bb))
        , λ b b' bb → ii (aa , bb)))

  rec-Term~ : ∀ {γ₀ γ₁} (q : Γ ∋ γ₀ ~ γ₁)
    → (t₀ : Term₀ (∣ A ∣ γ₀) (λ a → ∣ B ∣ (γ₀ , a)) (∣ X ∣ γ₀))
    → (t₁ : Term₀ (∣ A ∣ γ₁) (λ a → ∣ B ∣ (γ₁ , a)) (∣ X ∣ γ₁))
    → (ext₀ : extTerm (A ∋ Γ .ref γ₀ ⊢_~_) (B~ A B (Γ .ref γ₀))
        (X ∋ Γ .ref γ₀ ⊢_~_) t₀)
    → (ext₁ : extTerm (A ∋ Γ .ref γ₁ ⊢_~_) (B~ A B (Γ .ref γ₁))
        (X ∋ Γ .ref γ₁ ⊢_~_) t₁)
    → (tt' : Term₀~ (A ∋ q ⊢_~_) _ _ (B~ A B q) (X ∋ q ⊢_~_) t₀ t₁)
    → C ∋ q ⊢ rec-Term₀ γ₀ t₀ ext₀ ~ rec-Term₀ γ₁ t₁ ext₁
  rec-Term~ q (var₀ x₀) (var₀ x₁) _ _ xx = v .resp (q , xx)
  rec-Term~ q (op₀ a₀ f₀) (op₀ a₁ f₁) (f₀~ , extf₀) (f₁~ , extf₁) (aa , ff) =
    p .resp (((q , aa) , λ b₀ b₁ bb → ff (aa , bb))
      , λ b₀ b₁ bb → rec-Term~ q (f₀ b₀) (f₁ b₁) (extf₀ b₀) (extf₁ b₁)
          (ff (aa , bb)))

  rec-Term : Tm (Γ ▷ (Term A B X)) (C [ wk ])
  ∣ rec-Term ∣ (γ , t) = rec-Term₀ γ (t .₁) (t .₂)
  rec-Term .resp {γ₀ , t₀ , extt₀} {γ₁ , t₁ , extt₁} (γ₀₁ , t₀₁) =
    rec-Term~ γ₀₁ t₀ t₁ extt₀ extt₁ t₀₁
  {-# INJECTIVE_FOR_INFERENCE rec-Term #-}

  rec-Termₑ : (t : Tm Γ (Term A B X)) → Tm Γ C
  rec-Termₑ t = rec-Term [ idS ▸ t ]t
  {-# INJECTIVE_FOR_INFERENCE rec-Termₑ #-}
