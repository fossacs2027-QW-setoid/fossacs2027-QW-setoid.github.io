module W-rel {A₀ A₁ : Set} (_~A_ : A₀ → A₁ → Prop)
  (B₀ : A₀ → Set) (B₁ : A₁ → Set) (_~B_ : ∀ {x₀ x₁} → B₀ x₀ → B₁ x₁ → Prop) where

  𝒲~ : (w₀ : 𝒲 A₀ B₀) (w₁ : 𝒲 A₁ B₁) → Prop
  𝒲~ (sup₀ x₀ f₀) (sup₀ x₁ f₁) =
    (x₀ ~A x₁) ∧ ({y₀ : B₀ x₀} {y₁ : B₁ x₁} → y₀ ~B y₁ → 𝒲~ (f₀ y₀) (f₁ y₁))

module _
  {A : Set}    (_~A_ : A → A → Prop)
  {B : A → Set}(_~B_ : ∀ {x₀ x₁} → B x₀ → B x₁ → Prop) where
  open W-rel _~A_ B B _~B_

  extW : 𝒲 A B → Prop
  extW (sup₀ x f) = ({y₀ y₁ : B x} (y₀₁ : y₀ ~B y₁) → 𝒲~ (f y₀) (f y₁))
    ∧ ((y : B x) → extW (f y))

module _
    {A : Set} (_~A_ : A → A → Prop)
    {B : A → Set} (_~B_ : ∀ {x₀ x₁} → B x₀ → B x₁ → Prop)
    (Arefl : (x : A) → x ~A x)
  where
  open W-rel _~A_ B B _~B_

  𝒲~-refl : (w : 𝒲 A B) → extW _~A_ _~B_ w → 𝒲~ w w
  𝒲~-refl (sup₀ x f) (f~ , extf) =
    Arefl x , (λ y₀₁ → f~ y₀₁)

module _
    (A₀ A₁ : Set)
    (A₀₁~ : A₀ → A₁ → Prop) (A₁₀~ : A₁ → A₀ → Prop)
    (Asym : ∀ {x₀ x₁} → A₀₁~ x₀ x₁ → A₁₀~ x₁ x₀)
    (B₀ : A₀ → Set) (B₁ : A₁ → Set)
    (B₀₁~ : ∀ {x₀ x₁}  → B₀ x₀ → B₁ x₁  → Prop)
    (B₁₀~ : ∀ {x₁ x₀}  → B₁ x₁ → B₀ x₀  → Prop)
    (Bsym : ∀ {x₀ x₁ y₀ y₁} → B₁₀~ {x₁} {x₀} y₁ y₀ → B₀₁~ y₀ y₁)
  where
  open W-rel

  𝒲~-sym : (w₀ : 𝒲 A₀ B₀) (w₁ : 𝒲 A₁ B₁)
    → 𝒲~ A₀₁~ B₀ B₁ B₀₁~ w₀ w₁ → 𝒲~ A₁₀~ B₁ B₀ B₁₀~ w₁ w₀
  𝒲~-sym (sup₀ x₀ f₀) (sup₀ x₁ f₁)
    (x₀₁ , f₀₁) = (Asym x₀₁) , λ y₀₁ → 𝒲~-sym _ _ (f₀₁ (Bsym y₀₁))

module _
   (A₀ : Set) (B₀ : A₀ → Set)
   (A₁ : Set) (B₁ : A₁ → Set)
   (A₂ : Set) (B₂ : A₂ → Set)
   (A₀₁~ : A₀ → A₁ → Prop) (A₁₂~ : A₁ → A₂ → Prop) (A₀₂~ : A₀ → A₂ → Prop)
   (B₀₁~ : {x₀ : A₀} {x₁ : A₁} → B₀ x₀ → B₁ x₁ → Prop)
   (B₁₂~ : {x₁ : A₁} {x₂ : A₂} → B₁ x₁ → B₂ x₂ → Prop)
   (B₀₂~ : {x₀ : A₀} {x₂ : A₂} → B₀ x₀ → B₂ x₂ → Prop)
   (Atrans : ∀ {x₀ x₁ x₂} → A₀₁~ x₀ x₁ → A₁₂~ x₁ x₂ → A₀₂~ x₀ x₂)
   (Bcoe₀₁ : ∀ {x₀ x₁} → A₀₁~ x₀ x₁ → B₀ x₀ → B₁ x₁)
   (Bcoh₀₁  : ∀ {x₀ x₁} (x₀₁ : A₀₁~ x₀ x₁) (y₀ : B₀ x₀) → B₀₁~ y₀ (Bcoe₀₁ x₀₁ y₀))
   (Btrans : ∀ {x₀ x₁ x₂}
     (x₀₁ : A₀₁~ x₀ x₁) (x₁₂ : A₁₂~ x₁ x₂)
     {y₀ : B₀ x₀} {y₂ : B₂ x₂} → B₀₂~ y₀ y₂
     → B₁₂~ (Bcoe₀₁ x₀₁ y₀) y₂)
  where
  open W-rel
  𝒲~-trans
    : {w₀ : 𝒲 A₀ B₀} {w₁ : 𝒲 A₁ B₁} {w₂ : 𝒲 A₂ B₂}
    → 𝒲~ A₀₁~ B₀ B₁ B₀₁~ w₀ w₁ → 𝒲~ A₁₂~ B₁ B₂ B₁₂~ w₁ w₂
    → 𝒲~ A₀₂~ B₀ B₂ B₀₂~ w₀ w₂
  𝒲~-trans {sup₀ x₀ f₀} {sup₀ x₁ f₁} {sup₀ x₂ f₂} (x₀₁ , f₀₁) (x₁₂ , f₁₂) =
    Atrans x₀₁ x₁₂ , λ {y₀} {y₂} y₀₂ →
      𝒲~-trans (f₀₁ (Bcoh₀₁ x₀₁ y₀)) (f₁₂ (Btrans x₀₁ x₁₂ y₀₂))

module _
    {A₀ : Set} {_~A₀_ : A₀ → A₀ → Prop}
    {B₀ : A₀ → Set} {_~B₀_ : ∀ {x₀ x₀'} → B₀ x₀ → B₀ x₀' → Prop}
    {A₁ : Set} {_~A₁_ : A₁ → A₁ → Prop}
    {B₁ : A₁ → Set} {_~B₁_ : ∀ {x₁ x₁'} → B₁ x₁ → B₁ x₁' → Prop}
  where

  open W-rel _~A₀_ B₀ B₀ _~B₀_ using () renaming (𝒲~ to W₀~)
  open W-rel _~A₁_ B₁ B₁ _~B₁_ using () renaming (𝒲~ to W₁~)

  map~
    : (h : A₀ → A₁) (h~ : ∀ {x₀ x₀'} → x₀ ~A₀ x₀' → h x₀ ~A₁ h x₀')
    → (g : ∀ {x₀} → B₁ (h x₀) → B₀ x₀)
      (g~ : ∀ {x₀ x₁} {y₁ : B₁ (h x₀)} {y₁' : B₁ (h x₁)} → y₁ ~B₁ y₁' → g y₁ ~B₀ g y₁')
    → {w₀ w₀' : 𝒲 A₀ B₀}
    → W₀~ w₀ w₀' → W₁~ (W.map h g w₀) (W.map h g w₀')
  map~ h h~ g g~ {sup₀ _ _} {sup₀ _ _} (x₀₀' , f₀₀') =
    h~ x₀₀' , (λ {y₁} {y₁'} y₁₁' →
      map~ h h~ g g~ (f₀₀' (g~ y₁₁')))

  map-ext
    : (h : A₀ → A₁) (h~ : ∀ {x₀ x₀'} → x₀ ~A₀ x₀' → h x₀ ~A₁ h x₀')
    → (g : ∀ {x₀} → B₁ (h x₀) → B₀ x₀)
      (g~ : ∀ {x₀ x₁} {y₁ : B₁ (h x₀)} {y₁' : B₁ (h x₁)} → y₁ ~B₁ y₁' → g y₁ ~B₀ g y₁')
    → (w₀ : 𝒲 A₀ B₀) (extw : extW _~A₀_ _~B₀_ w₀)
    → extW _~A₁_ _~B₁_ (W.map h {B₀} {B₁} g w₀)
  map-ext h h~ g g~ (sup₀ x₀ f₀) (fext₀ , pf₀) =
    (λ y₀₀' → map~ h h~ g g~ (fext₀ (g~ y₀₀')))
    , λ y₁ → map-ext h h~ g g~ (f₀ (g y₁)) (pf₀ (g y₁))
module _ (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero) where
  open W-rel

  B~ : {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) {x₀ : ∣ A ∣ γ₀} {x₁ : ∣ A ∣ γ₁}
    → ∣ B ∣ (γ₀ , x₀) → ∣ B ∣ (γ₁ , x₁) → Prop
  B~ γ₀₁ {x₀} {x₁} y₀ y₁ = ∃[ x₀₁ ∈ (A ∋ γ₀₁ ⊢ x₀ ~ x₁) ] (B ∋ (γ₀₁ , x₀₁) ⊢ y₀ ~ y₁)

  |W| : ∣ Γ ∣ → Set
  |W| γ = Σₚ (𝒲 (∣ A ∣ γ) (λ x → ∣ B ∣ (γ , x)))
    (extW (A ∋ γ̇ ⊢_~_) (B~ γ̇))
    where γ̇ = Γ .ref γ

  W~ : {γ₀ γ₁ : ∣ Γ ∣} → Γ ∋ γ₀ ~ γ₁ → |W| γ₀ → |W| γ₁ → Prop
  W~ {γ₀} {γ₁} γ₀₁ (w₀ , _) (w₁ , _) = 𝒲~ (A ∋ γ₀₁ ⊢_~_)
    (λ x₀ → ∣ B ∣ (γ₀ , x₀)) (λ x₁ → ∣ B ∣ (γ₁ , x₁))
    (B~ γ₀₁) w₀ w₁

  W-refl
    : ∀ {γ} (w : |W| γ) → W~ (Γ .ref γ) w w
  W-refl (sup₀ x f , f~ , _) =
    A .refl* x , f~

  W-sym
    : ∀ {γ₀ γ₁} {γ₀₁ : Γ ∋ γ₀ ~ γ₁}
    → (w₀ : |W| γ₀) (w₁ : |W| γ₁)
    → W~ γ₀₁ w₀ w₁ → W~ (Γ .sym γ₀₁) w₁ w₀
  W-sym (sup₀ x₀ f₀ , _ , extf₀) (sup₀ x₁ f₁ , _ , extf₁) (x₀₁ , f₀₁) =
    A .sym* x₀ x₁ x₀₁ , λ {y₁} {y₀} (x₁₀ , y₁₀) →
      W-sym (f₀ y₀ , extf₀ y₀) (f₁ y₁ , extf₁ y₁)
        (f₀₁ (A .sym* x₁ x₀ x₁₀ , B .sym* y₁ y₀ y₁₀))

  W-trans
    : ∀ {γ₀ γ₁ γ₂} {γ₀₁ : Γ ∋ γ₀ ~ γ₁} {γ₁₂ : Γ ∋ γ₁ ~ γ₂}
    → (w₀ : |W| γ₀) (w₁ : |W| γ₁) (w₂ : |W| γ₂)
    → W~ γ₀₁ w₀ w₁ → W~ γ₁₂ w₁ w₂
    → W~ (Γ .trans γ₀₁ γ₁₂) w₀ w₂
  W-trans
    (sup₀ x₀ f₀ , _ , extf₀) (sup₀ x₁ f₁ , _ , extf₁) (sup₀ x₂ f₂ , _ , extf₂)
    (x₀₁ , f₀₁) (x₁₂ , f₁₂) =
    A .trans* x₀ x₁ x₂ x₀₁ x₁₂ , λ {y₀} {y₂} (x₀₂ , y₀₂) →
      W-trans
        (f₀ y₀ , extf₀ y₀)
        (f₁ (B .coe (_ , x₀₁) y₀) , extf₁ (B .coe (_ , x₀₁) y₀))
        (f₂ y₂ , extf₂ y₂)
        (f₀₁ (x₀₁ , B .coh (_ , x₀₁) y₀))
        (f₁₂ (x₁₂ , B .trans* _ _ _ (B .sym* _ _ (B .coh (_ , x₀₁) y₀)) y₀₂))

  W-coe
    : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) → |W| γ₀ → |W| γ₁
  W-coe γ₀₁ (w , extw) =
    W.map (A .coe γ₀₁) (λ {x₀} b₁ → B .coe (_ , A .sym* _ _ (A .coh γ₀₁ x₀)) b₁) w ,
    map-ext (A .coe γ₀₁)
            (λ {x₀} {x₀'} x₀₀' → A .trans* _ _ _ (A .sym* _ _ (A .coh γ₀₁ x₀))
                             (A .trans* _ _ _ x₀₀' (A .coh γ₀₁ x₀')))
            (λ {x₀} b₁ → B .coe (_ , A .sym* _ _ (A .coh γ₀₁ x₀)) b₁)
            (λ {x₀} {x₀'} {y₁} {y₁'} (x₁₁' , y₁₁') →
              A .trans* _ _ _ (A .coh γ₀₁ x₀) (A .trans* _ _ _ x₁₁' (A .sym* _ _ (A .coh γ₀₁ x₀'))) ,
              B .trans* _ _ _ (B .sym* _ _ (B .coh (_ , A .sym* _ _ (A .coh γ₀₁ x₀)) y₁))
                (B .trans* _ _ _ y₁₁'
                  (B .coh (_ , A .sym* _ _ (A .coh γ₀₁ x₀')) y₁')))
            w extw

  W-coh
    : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → (w : |W| γ₀) → W~ γ₀₁ w (W-coe γ₀₁ w)
  W-coh γ₀₁ (sup₀ x f , f~ , extf) =
    A .coh γ₀₁ x , λ {y₀} {y₂} (x₀₂ , y₀₂) →
      let y₁ = B .coe (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ x)) y₂
          w₁ = f y₁ , extf y₁
      in W-trans (f y₀ , extf y₀) w₁ (W-coe γ₀₁ w₁)
           (f~
             (A .sym* _ _ (A .trans* _ _ _ (A .coh γ₀₁ x) (A .sym* _ _ x₀₂)) ,
              B .trans* _ _ _ y₀₂ (B .coh (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ x)) y₂)))
           (W-coh γ₀₁ w₁)

  W : Ty Γ lzero
  ∣ W ∣     = |W|
  W .heq    = W~
  W .refl*  = W-refl
  W .sym*   = W-sym
  W .trans* = W-trans
  W .coe    = W-coe
  W .coh    = W-coh
  {-# INJECTIVE_FOR_INFERENCE W #-}

  sup-W
    : (x : Tm Γ A) (f : Tm Γ (Π (B ⟨ x ⟩) (W [ wk ])))
    → Tm Γ W
  ∣ sup-W x f ∣ γ = sup₀ (∣ x ∣ γ) (λ x → ∣ f ∣ γ .₁ x .₁)
    , (λ (γ₀₁ , y₀₁) → ∣ f ∣ γ .₂ y₀₁) , λ y → ∣ f ∣ γ .₁ y .₂
  sup-W x f .resp γ₀₁ = (x .resp γ₀₁) , λ (x₀₁ , y₀₁) → f .resp γ₀₁ _ _ y₀₁
