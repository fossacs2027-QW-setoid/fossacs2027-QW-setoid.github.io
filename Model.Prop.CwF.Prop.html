open import Prelude

module Model.Prop.CwF.Prop where

open import Model.Prop.CwF.Core

record Prp (Γ : Con ℓΓ) (ℓP : Level) : Set (lsuc (ℓP ⊔ ℓΓ)) where
  field
    P
      : (γ : ∣ Γ ∣) → Prop ℓP
    coe
      : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) → P γ₀ → P γ₁
open Prp renaming (P to ∣_∣) public

_[_]ₚ : Prp Γ ℓ → Sub Δ Γ → Prp Δ ℓ
∣ P [ σ ]ₚ ∣ γ = ∣ P ∣ (∣ σ ∣ γ)
(P [ σ ]ₚ) .coe γ₀₁ p₀ = P .coe (σ .resp γ₀₁) p₀

Props : (ℓ : Level) → Ty Γ (lsuc ℓ)
∣ Props ℓ ∣ _   = Prop ℓ
Props ℓ .heq _ x₀ x₁ = LiftP (lsuc ℓ) ((x₀ → x₁) ∧ (x₁ → x₀))
Props ℓ .refl* _ = ↑ ((λ  x → x) , (λ x → x)) -- ↑ ((λ x → x) , (λ x → x))
Props ℓ .sym* _ _ (↑ (p , q)) = ↑ (q , p)
Props ℓ .trans* _ _ _ (↑ (p , q)) (↑ (r , s)) = ↑ ((λ z → r (p z)) , λ  z → q (s z))
Props ℓ .coe _ x₀ = x₀
Props ℓ .coh _ x₀ = ↑ ((λ z → z) , (λ z → z))

Props[]
  : Props ℓ [ σ ] ≡ Props ℓ
Props[] = refl

ElP : Tm Γ (Props ℓ) → Ty Γ ℓ
ElP p .∣_∣ γ = LiftS lzero (∣ p ∣ γ)
ElP p .heq γ₀₁ _ _ = ⊤ₚ
ElP p .refl* _ = tt
ElP p .sym* _ _ _ = tt
ElP p .trans* _ _ _ _ _ = tt
ElP p .coe γ₀₁ (↑ a₀) .↓ = (p .resp γ₀₁ .↓ .₁ a₀) 
ElP p .coh _ _ = tt

ElP[]
  : {a : Tm Γ (Props ℓ)}
  → ElP a [ σ ] ≡ ElP (a [ σ ]t)
ElP[] = refl

irr : {a : Tm Γ (Props ℓ)} {u v : Tm Γ (ElP a)} → u ≡ v
irr = refl

codeP : Prp Γ ℓ → Tm Γ (Props ℓ)
∣ codeP P ∣ γ = ∣ P ∣ γ
codeP {Γ = Γ} P .resp γ₀₁ .↓ =
  (P .coe γ₀₁) , P .coe (Γ .sym γ₀₁)

decodeP : Tm Γ (Props ℓ) → Prp Γ ℓ
∣ decodeP p ∣ γ = ∣ p ∣ γ
decodeP p .coe γ₀₁ = p .resp γ₀₁ .↓ .₁

-- Lift
⇑_ : Prp Γ ℓP → Ty Γ ℓP
∣ ⇑ P ∣ γ         = LiftS lzero (∣ P ∣ γ) -- ↑ (∣ P ∣ γ)
⇑_ P .heq _ _ _   = ⊤ₚ
⇑_ P .refl* _     = tt
⇑_ P .sym* _ _ _ = tt
⇑_ P .trans* _ _ _ _ _ = tt
⇑_ P .coe p (↑ x) = ↑ (P .coe p x)
⇑_ P .coh _ _     = tt
{-# INJECTIVE_FOR_INFERENCE ⇑_ #-}

⇑[] : (P : Prp Γ ℓP) → (⇑ P) [ σ ] ≡ ⇑ (P [ σ ]ₚ)
⇑[] P = refl
 
ElP=⇑
  : {P : Prp Γ ℓ}
  → ElP (codeP P) ≡ (⇑ P)
ElP=⇑ = refl

⇓_ : Ty Γ ℓA → Prp Γ ℓA
∣ ⇓ A ∣    _   = ⊤ₚ
(⇓ A) .coe _ _ = tt
 
-- Closure under standard prop formers

πₚ' : (A : Ty Γ ℓA) → (b : Prp (Γ ▷ A) ℓA) → Prp Γ ℓA
∣ πₚ' a b ∣ γ = (x : ∣ a ∣ γ) → ∣ b ∣ (γ , x)
πₚ' {Γ = Γ} a b .coe γ₀₁ f x =
 b .coe (γ₀₁ , a .sym* _ _ (a .coh (Γ .sym γ₀₁) x) ) (f (a .coe (Γ .sym γ₀₁) x))

πₚ : (A : Ty Γ ℓA) → (b : Tm (Γ ▷ A) (Props ℓA)) → Tm Γ (Props ℓA)
πₚ A b = codeP (πₚ' A (decodeP b))

appₚ : {A : Ty Γ ℓA}{b : Tm (Γ ▷ A) (Props ℓA)}
  → Tm Γ (ElP (πₚ A b)) → (a : Tm Γ A) → Tm Γ (ElP (b [ idS ▸ a ]t))
∣ appₚ f x ∣ γ  = ↑ (∣ f ∣ γ .↓ (∣ x ∣ γ))
appₚ f x .resp _ = tt

lamₚ : {A : Ty Γ ℓA}{b : Tm (Γ ▷ A) (Props ℓA)}
  → Tm (Γ ▷ A) (ElP b) → Tm Γ (ElP (πₚ A b))
∣ lamₚ t ∣ γ = ↑ (λ x → ∣ t ∣ (γ , x) .↓)
lamₚ t .resp _ = tt

impₚ' : (p q : Prp Γ ℓ) → Prp Γ ℓ
∣ impₚ' p q ∣ γ = ∣ p ∣ γ → ∣ q ∣ γ
impₚ' {Γ = Γ} p q .coe γ₀₁ p→q x = q .coe γ₀₁ (p→q (p .coe (Γ .sym γ₀₁) x))

impₚ : (p q : Tm Γ (Props ℓ)) → Tm Γ (Props ℓ)
impₚ p q = codeP (impₚ' (decodeP p) (decodeP q))
