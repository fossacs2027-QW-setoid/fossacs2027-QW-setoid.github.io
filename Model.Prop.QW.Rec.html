QW-rec-step : {Γ : Con ℓΓ} {A : Ty Γ lzero} (B : Ty (Γ ▷ A) lzero)
  (ε : EqSys A B) (C : Ty Γ ℓ) → Set _
QW-rec-step B ε C = QW-step B ε (C [ wk ])

-- Paired assignments retain the original QW values as well as the results.
QW-rec-coh : {Γ : Con ℓΓ} {A : Ty Γ lzero} (B : Ty (Γ ▷ A) lzero)
  (ε : EqSys A B) (C : Ty Γ ℓ) (p : QW-rec-step B ε C) → Set _
QW-rec-coh B ε C p = QW-elim-coh B ε (C [ wk ]) p

open RelAt using (Ext; QW₀~)

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero}
         (B : Ty (Γ ▷ A) lzero)
         (ε : EqSys A B)
         (C : Ty Γ ℓ)
         (p : QW-rec-step B ε C)
         (equ-p : QW-rec-coh B ε C p)
  where

  private
    C-at : (γ : ∣ Γ ∣) (w : QW₀ A B γ)
      → Ext A B ε γ w → Set ℓ
    C-at γ w extw = ∣ C ∣ γ

    C~-at
      : (γ : ∣ Γ ∣) {w₀ w₁ : QW₀ A B γ}
      → QW₀~ A B ε γ w₀ w₁
      → {extw₀ : Ext A B ε γ w₀} {extw₁ : Ext A B ε γ w₁}
      → C-at γ w₀ extw₀ → C-at γ w₁ extw₁ → Prop ℓ
    C~-at γ w₀₁ {extw₀} {extw₁} = C .heq (Γ .ref γ)

    private
      module Steps (γ : ∣ Γ ∣) = Raw.StepTypes
        (A .heq (Γ .ref γ)) (B~ A B (Γ .ref γ)) (SigAt A B ε γ)
        (C-at γ) (C~-at γ)

    alg-at : (γ : ∣ Γ ∣) → Steps.Step γ
    alg-at γ {a} {f} extf f~ ih ih~ = ∣ p ∣
      (((γ , a) , ((λ b → f b , extf b)
        , λ {b₀} {b₁} b₀₁ →
            coe-QEq A B ε {u = _ , extf b₀} {v = _ , extf b₁} (f~ (A .refl* a , b₀₁))))
        , (ih , λ {b₀} {b₁} b₀₁ → ih~ (A .refl* a , b₀₁)))

    alg-at~ : (γ : ∣ Γ ∣) → Steps.Step~ γ (alg-at γ)
    alg-at~ γ f₀~ extf₀ ih₀ ih₀~ f₁~ extf₁ ih₁ ih₁~ a₀₁ f₀₁ ih₀₁ =
      p .resp (((Γ .ref γ , a₀₁)
               , λ b₀ b₁ b₀₁ →
                   coe-QEq A B ε {u = _ , extf₀ b₀} {v = _ , extf₁ b₁} (f₀₁ (a₀₁ , b₀₁)))
               , λ b₀ b₁ b₀₁ → ih₀₁ (a₀₁ , b₀₁))

    module Fiber (γ : ∣ Γ ∣) =
      Raw.QW-elim₀
        (A .heq (Γ .ref γ)) (A .refl*) (λ {a₀} {a₁} → A .sym* a₀ a₁)
        (B~ A B (Γ .ref γ))
        (λ (aa , bb) → A .sym* _ _ aa , B .sym* _ _ bb)
        (λ (aa , bb) (aa' , bb') → A .trans* _ _ _ aa aa' , B .trans* _ _ _ bb bb')
        (λ aa b → B .coe (Γ .ref γ , aa) b)
        (λ aa b → aa , B .coh (Γ .ref γ , aa) b)
        (SigAt A B ε γ)
        (λ e x → V ε .refl* {γ , e} x)
        (λ e → ∣ lhs ε ∣ (γ , e) .₂) (λ e → ∣ rhs ε ∣ (γ , e) .₂)
        (QW A B ε ∋ Γ .ref γ ⊢_~_)
        (λ {u} {v} → uncoe-QEq A B ε {u = u} {v = v})
        (λ {u} {v} → coe-QEq A B ε {u = u} {v = v})
        (C-at γ) (C~-at γ)
        (λ w extw v → C .refl* v)
        (λ {v₀ = x} {v₁ = y} → C .sym* x y) (λ {v₀ = x} {v₁ = y} {v₂ = z} → C .trans* x y z)
        (alg-at γ) (alg-at~ γ)

    private
      fiber-equ-p : (γ : ∣ Γ ∣) → Fiber.QW-elim₀-coh γ
      fiber-equ-p γ e ρ ρ~ ρP ρP~ =
        LiftS.↓ (∣ equ-p ∣ γ .₁ e .₁
          ((λ x → ρ x , ρP x)
            , λ {x} {y} xy → coe-QEq A B ε {u = ρ x} {v = ρ y} (ρ~ xy) , ρP~ xy))

    rec-at
      : (γ : ∣ Γ ∣) (w : QW₀ A B γ)
      → (extw : Ext A B ε γ w)
      → ∣ C ∣ γ
    rec-at γ = Fiber.elim-QW₀ γ (fiber-equ-p γ)

    rec-at~
      : (γ : ∣ Γ ∣) {w₀ w₁ : QW₀ A B γ}
      → (w₀₁ : QW₀~ A B ε γ w₀ w₁)
      → (extw₀ : Ext A B ε γ w₀)
      → (extw₁ : Ext A B ε γ w₁)
      → C ∋ Γ .ref γ ⊢ rec-at γ w₀ extw₀ ~ rec-at γ w₁ extw₁
    rec-at~ γ = Fiber.elim-QW₀~ γ (fiber-equ-p γ)

    rec-at-coe
      : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      → (w : QW₀ A B γ₀) (extw : Ext A B ε γ₀ w)
      → C ∋ γ₀₁ ⊢ rec-at γ₀ w extw
          ~ rec-at γ₁ (QW-coe A B ε γ₀₁ w)
              (ext-coe A B ε γ₀₁ w extw)
    rec-at-coe {γ₁ = γ₁} γ₀₁ (sup₀ a f) (f~ , extf) =
      p .resp
        (((γ₀₁ , A .coh γ₀₁ a)
          , λ b₀ b₁ b₀₁ → QW-coe~ A B ε γ₀₁ (f~ (branch-eq b₀₁)))
         , λ b₀ b₁ b₀₁ → C .trans* _ _ _
             (rec-at-coe γ₀₁ (f b₀) (extf b₀))
             (rec-at~ γ₁
               (QW-coe~ A B ε γ₀₁ (f~ (branch-eq b₀₁)))
               (transported-ext b₀) (transported-ext (pull b₁))))
      where
        back = Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a)
        pull = B .coe back
        branch-eq : ∀ {b₀ b₁} → B ∋ (γ₀₁ , A .coh γ₀₁ a) ⊢ b₀ ~ b₁
          → B~ A B (Γ .ref _) b₀ (pull b₁)
        branch-eq b₀₁ = A .refl* a , B .trans* _ _ _ b₀₁ (B .coh back _)
        transported-ext = λ b → ext-coe A B ε γ₀₁ (f b) (extf b)

  rec-QW : Tm (Γ ▷ QW A B ε) (C [ wk ])
  ∣ rec-QW ∣ (γ , (w , extw)) = rec-at γ w extw
  rec-QW .resp {γ₀ , (w₀ , extw₀)} {γ₁ , (w₁ , extw₁)}
    (γ₀₁ , w₀₁) =
    C .trans* _ _ _
      (rec-at-coe γ₀₁ w₀ extw₀)
      (rec-at~ γ₁ w₀₁
        (ext-coe A B ε γ₀₁ w₀ extw₀) extw₁)

  rec-QWᵢ : Tm Γ (QW A B ε ⇒ C)
  rec-QWᵢ = ƛ rec-QW
