-- External constructor of dependent pairs
module _ {ℓΓ ℓA ℓB} {Γ : Con ℓΓ} {A : Ty Γ ℓA} (B : Ty (Γ ▷ A) ℓB) where
  dpair : (a : Tm Γ A) → (b : Tm Γ (B ⟨ a ⟩)) → Tm Γ (Sigma A B)
  dpair a b = dpairⁱ {B = B} [ idS ▸ a ▸ b ]t
  {-# INJECTIVE_FOR_INFERENCE dpair #-}

module _ {Γ : Con ℓΓ} {A : Ty Γ ℓA} (B : Ty (Γ ▷ A) ℓB) where

  fst : Tm (Γ ▷ Sigma A B) (A [ wk ])
  fst = elim-Sigma (A [ wk ]) (vs vz)

  snd : Tm (Γ ▷ Sigma A B) (B [ wk ▸ fst ])
  snd = elim-Sigma (B [ wk ▸ fst ]) vz

module _ {Γ : Con ℓΓ} (A : Ty Γ ℓA) (B : Ty (Γ ▷ A) ℓB) where
  fstₑ : Tm Γ (Sigma A B) → Tm Γ A
  fstₑ t = fst B ⟨ t ⟩t
  
  sndₑ : (t : Tm Γ (Sigma A B)) → Tm Γ (B ⟨ fstₑ t ⟩)
  sndₑ t = snd B ⟨ t ⟩t

-- Constructor of non-dependent pairs
module _ {ℓΓ ℓA ℓB} {Γ : Con ℓΓ} (A : Ty Γ ℓA) (B : Ty Γ ℓB) where
  pair : (a : Tm Γ A) (b : Tm Γ B) → Tm Γ (A ẋ B)
  pair a b = dpairⁱ {A = A} {B [ wk ]} [ idS ▸ a ▸ b ]t
  {-# INJECTIVE_FOR_INFERENCE pair #-}

-- Eliminators of non-dependent pairs
module _ {Γ : Con ℓΓ} (A : Ty Γ ℓA) (B : Ty Γ ℓB) where
  fst' : Tm (Γ ▷ A ẋ B) (A [ wk ])
  fst' = fst {A = A} (B [ wk ])

  snd' : Tm (Γ ▷ A ẋ B) (B [ wk ])
  snd' = snd {A = A} (B [ wk ])

module _ {Γ : Con ℓΓ} (A : Ty Γ ℓA) (B : Ty Γ ℓB) where
  fstₑ' : Tm Γ (A ẋ B) → Tm Γ A
  fstₑ' t = fst {A = A} (B [ wk ]) ⟨ t ⟩t
  
  sndₑ' : (t : Tm Γ (A ẋ B)) → Tm Γ B
  sndₑ' t = snd {A = A} (B [ wk ]) ⟨ t ⟩t

-- (p : Σ A B) → C (p .₁) → Σ A (λ a → B a × C a)
module _ {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB} (C : Ty (Γ ▷ A) ℓC) where
  assoc-Sigma
    : Tm (Γ ▷ Sigma A B ▷ C [ wk ▸ fst B ]) (Sigma A (B ẋ C) [ wk∘wk ])
  assoc-Sigma = dpair (B [ wk∘wk ⁺ ] ẋ C [ wk∘wk ⁺ ]) (fst B [ wk ]t)
                      (pair (B [ wk∘wk ▸ fst B [ wk ]t ]) (C [ wk∘wk ▸ fst B [ wk ]t ])
                        (snd B [ wk ]t) vz)
