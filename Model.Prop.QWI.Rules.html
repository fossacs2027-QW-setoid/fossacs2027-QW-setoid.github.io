module _ {Γ : Con ℓΓ} {Δ : Con ℓΔ} {S : Poly Γ} {ε : EqSys S} {σ : Sub Δ Γ} where
  QWI[] : QWI S ε [ σ ⁺ ] ≡ QWI (S [ σ ]s) (ε [ σ ]ε)
  QWI[] = refl

  sup-QWI[] : sup-QWI S ε [ σ ⁺ ⁺ ⁺ ]t ≡ sup-QWI (S [ σ ]s) (ε [ σ ]ε)
  sup-QWI[] = refl

  qwi-eq[] : qwi-eq S ε [ σ ⁺ ⁺ ⁺ ]t ≡ qwi-eq (S [ σ ]s) (ε [ σ ]ε)
  qwi-eq[] = refl

  elim-QWI[] : (P : Ty (Γ ▷ I S ▷ QWI S ε) ℓ) (p : QWI-step S ε P)
    (coh : QWI-elim-coh S ε P p)
    → elim-QWI S ε P p coh [ σ ⁺ ⁺ ]t
      ≡ elim-QWI (S [ σ ]s) (ε [ σ ]ε) (P [ σ ⁺ ⁺ ]) (p [ σ ⁺ ⁺ ⁺ ⁺ ]t)
          (coh [ σ ⁺ ]t)
  elim-QWI[] P p coh = refl

module _ {Γ : Con ℓΓ} {S : Poly Γ} {ε : EqSys S}
  (P : Ty (Γ ▷ I S ▷ QWI S ε) ℓ) (p : QWI-step S ε P) (coh : QWI-elim-coh S ε P p) where

  elim-sup-QWI : elim-QWI S ε P p coh [ QWI-head S ε P ]t
    ≡ p [ idS ▸ ƛ (elim-QWI S ε P p coh [ QWI-child S ε P ]t) ]t
  elim-sup-QWI = refl

module _ {Γ : Con ℓΓ} {S : Poly Γ} {ε : EqSys S}
  (C : Ty (Γ ▷ I S) ℓ) (p : QWI-rec-step S ε C) (coh : QWI-rec-coh S ε C p) where

  rec-QWI-elim : rec-QWI S ε C p coh ≡ elim-QWI S ε (C [ wk ]) p coh
  rec-QWI-elim = refl

  rec-QWI[] : {Δ : Con ℓΔ} {σ : Sub Δ Γ}
    → rec-QWI S ε C p coh [ σ ⁺ ⁺ ]t
      ≡ rec-QWI (S [ σ ]s) (ε [ σ ]ε) (C [ σ ⁺ ]) (p [ σ ⁺ ⁺ ⁺ ⁺ ]t)
          (coh [ σ ⁺ ]t)
  rec-QWI[] = refl

  rec-sup-QWI : rec-QWI S ε C p coh [ QWI-head S ε (C [ wk ]) ]t
    ≡ p [ idS ▸ ƛ (rec-QWI S ε C p coh [ QWI-child S ε (C [ wk ]) ]t) ]t
  rec-sup-QWI = refl
