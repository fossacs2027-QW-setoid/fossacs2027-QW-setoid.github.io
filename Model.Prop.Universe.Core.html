open import Prelude

module Model.Prop.Universe.Core where

open import Model.Prop.IF.Nat.Core

-- A Martin-Löf identity family enlarged with the ambient setoid equality.
-- The second constructor is what makes the family invariant under setoid
-- equality while keeping the reflexivity computation rule judgmental.
data FId (A : Set) (R : A → A → Prop) (a : A) : A → Set where
  frefl : FId A R a a
  ford  : ∀ {b} → R a b → FId A R a b

FId→R
  : ∀ {A : Set} {R : A → A → Prop}
  → ((a : A) → R a a)
  → ∀ {a b} → FId A R a b → R a b
FId→R reflR frefl    = reflR _
FId→R reflR (ford p) = p

-- The over-approximating universe.  The equality parameters of functions
-- and identity types are deliberately arbitrary at this stage.
data InU : Set → Set₁ where
  inNat
    : InU ℕ

  inΣ
    : ∀ {A : Set} (Au : InU A)
      {B : A → Set} (Bu : (a : A) → InU (B a))
    → InU (Σ A B)

  inΠ
    : ∀ {A : Set} (Au : InU A) (A≈ : A → A → Prop)
      {B : A → Set} (Bu : (a : A) → InU (B a))
      (B≈ : (a₀ a₁ : A) → B a₀ → B a₁ → Prop)
    → InU
        (Σₚ ((a : A) → B a)
          (λ f → ∀ {a₀ a₁} → A≈ a₀ a₁ → B≈ a₀ a₁ (f a₀) (f a₁)))

  inId
    : ∀ {A : Set} (Au : InU A) (A≈ : A → A → Prop) (a b : A)
    → InU (FId A A≈ a b)

-- Heterogeneous equality of decoded elements.  Equality witnesses stored in
-- the over-approximation are ignored; only the shapes and recursive codes
-- determine equality.
ElEq : ∀ {A B : Set} → InU A → InU B → A → B → Prop
ElEq inNat inNat m n = N._~_ m n
ElEq (inΣ Au Bu) (inΣ Cu Du) (a , b) (c , d) =
  ∃[ ac ∈ ElEq Au Cu a c ] ElEq (Bu a) (Du c) b d
ElEq (inΠ Au A≈ Bu B≈) (inΠ Cu C≈ Du D≈) (f , _) (g , _) =
  ∀ a c → ElEq Au Cu a c → ElEq (Bu a) (Du c) (f a) (g c)
ElEq (inId Au A≈ a b) (inId Cu C≈ c d) p q = ⊤ₚ
ElEq _ _ _ _ = ⊥ₚ

-- Structural equality of codes.  Codes with different outer constructors
-- are disjoint even when their carrier Sets happen to coincide.
CodeEq : ∀ {A B : Set} → InU A → InU B → Prop
CodeEq inNat inNat = ⊤ₚ
CodeEq (inΣ Au Bu) (inΣ Cu Du) =
  CodeEq Au Cu ∧
  (∀ a c → ElEq Au Cu a c → CodeEq (Bu a) (Du c))
CodeEq (inΠ Au A≈ Bu B≈) (inΠ Cu C≈ Du D≈) =
  CodeEq Au Cu ∧
  (∀ a c → ElEq Au Cu a c → CodeEq (Bu a) (Du c))
CodeEq (inId Au A≈ a b) (inId Cu C≈ c d) =
  CodeEq Au Cu ∧ (ElEq Au Cu a c ∧ ElEq Au Cu b d)
CodeEq _ _ = ⊥ₚ

-- Extensional codes are the well-formed fragment of InU.  In particular,
-- the arbitrary equality parameters in InU are forced to be ElEq.
data Ext : ∀ {A : Set} (Au : InU A) → Set₁ where
  extNat : Ext inNat

  extΣ
    : ∀ {A : Set} {Au : InU A}
      (Ae : Ext Au)
      {B : A → Set} {Bu : (a : A) → InU (B a)}
      (Bext : ∀ a₀ a₁ → ElEq Au Au a₀ a₁ → CodeEq (Bu a₀) (Bu a₁))
      (Be : (a : A) → Ext (Bu a))
    → Ext (inΣ Au Bu)

  extΠ
    : ∀ {A : Set} {Au : InU A}
      (Ae : Ext Au)
      {B : A → Set} {Bu : (a : A) → InU (B a)}
      (Bext : ∀ a₀ a₁ → ElEq Au Au a₀ a₁ → CodeEq (Bu a₀) (Bu a₁))
      (Be : (a : A) → Ext (Bu a))
    → Ext
        (inΠ Au (ElEq Au Au) Bu
          (λ a₀ a₁ → ElEq (Bu a₀) (Bu a₁)))

  extId
    : ∀ {A : Set} {Au : InU A}
    → Ext Au → (a b : A)
    → Ext (inId Au (ElEq Au Au) a b)

record U₀ : Set₁ where
  constructor pack
  field
    El₀  : Set
    in₀  : InU El₀
    ext₀ : Ext in₀

open U₀ public

eq₀ : (A B : U₀) → El₀ A → El₀ B → Prop
eq₀ A B = ElEq (in₀ A) (in₀ B)

eqU₀ : U₀ → U₀ → Prop
eqU₀ A B = CodeEq (in₀ A) (in₀ B)

nat₀ : U₀
nat₀ = pack ℕ inNat extNat

sigma₀
  : (A : U₀) (B : El₀ A → U₀)
  → (∀ {a₀ a₁} → eq₀ A A a₀ a₁ → eqU₀ (B a₀) (B a₁))
  → U₀
sigma₀ A B Bext =
  pack
    (Σ (El₀ A) (λ a → El₀ (B a)))
    (inΣ (in₀ A) (λ a → in₀ (B a)))
    (extΣ (ext₀ A) (λ _ _ p → Bext p) (λ a → ext₀ (B a)))

pi₀
  : (A : U₀) (B : El₀ A → U₀)
  → (∀ {a₀ a₁} → eq₀ A A a₀ a₁ → eqU₀ (B a₀) (B a₁))
  → U₀
pi₀ A B Bext =
  pack
    (Σₚ ((a : El₀ A) → El₀ (B a))
      (λ f → ∀ {a₀ a₁} → eq₀ A A a₀ a₁ →
        eq₀ (B a₀) (B a₁) (f a₀) (f a₁)))
    (inΠ (in₀ A) (eq₀ A A) (λ a → in₀ (B a))
      (λ a₀ a₁ → eq₀ (B a₀) (B a₁)))
    (extΠ (ext₀ A) (λ _ _ p → Bext p) (λ a → ext₀ (B a)))

id₀ : (A : U₀) → El₀ A → El₀ A → U₀
id₀ A a b =
  pack
    (FId (El₀ A) (eq₀ A A) a b)
    (inId (in₀ A) (eq₀ A A) a b)
    (extId (ext₀ A) a b)
