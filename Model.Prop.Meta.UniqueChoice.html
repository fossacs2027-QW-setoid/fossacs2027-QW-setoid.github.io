uc : (A : Ty Γ ℓA)
  → Tm Γ (Exists A (Π (A [ wk ]) (⇑ Eq A wk∘wk (vs vz) vz))
    ⇒ A)
∣ uc A ∣ γ = (λ (x , _) → x) , (λ { {(x , f , _)} {(x' , _)} _ → f x' .↓})
uc A .resp γ₀₁ (x₀ , f₀ , p₀) (x₁ , f₁ , p₁) _ =
  A .trans* _ _ _ (A .coh γ₀₁ x₀) (A .sym* _ _ ((f₁ (A .coe γ₀₁ x₀)) .↓))
