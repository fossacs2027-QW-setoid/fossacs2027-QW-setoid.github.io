rec-Unit : ∀ {ℓU} → (B : Ty Γ ℓB) → Tm Γ B → Tm Γ (Unit {ℓ = ℓU} ⇒ B)
rec-Unit B t = ƛ (t [ wk ]t)

rec-Unitₑ : ∀ {ℓU} → (B : Ty Γ ℓB) → Tm Γ B → Tm Γ (Unit {ℓ = ℓU}) → Tm Γ B
rec-Unitₑ B t u = appₑ (B [ wk ]) (rec-Unit B t) u
