suc-inj : {n m : ℕ} → suc n N.~ suc m → n N.~ m
suc-inj p = p

Nat-discrete : {x₀ x₁ : ℕ} (q : x₀ N.~ x₁) → x₀ ≡ x₁
Nat-discrete {zero}   {zero}   q  = refl
Nat-discrete {suc x₀} {suc x₁} q =
  cong suc (Nat-discrete {x₀} {x₁} (suc-inj {x₀} {x₁} q))

cc : (B : Ty Γ ℓB)(R : Ty (Γ ▷ Nat ▷ B [ wk ]) ℓ)
   → Tm Γ (Π Nat (Exists (B [ wk ]) R) ⇒ Exists (Nat ⇒ B) (Π Nat (R [ wk∘wk ▸ vz ▸ app vz ])))
∣ cc {Γ = Γ} B R ∣ γ = (λ h → ((λ n → ₁ (₁ h n)) , fact₀ h) , ((λ n → ₂ (₁ h n)) , fact₁ h)) , λ _ → tt
 where
  fact₀ : (h : |Π| Nat (Exists (B [ wk ]) R) γ)
        → {x₀ x₁ : ℕ} (q : x₀ N.~ x₁) → B ∋ Γ .ref γ ⊢ ₁ (₁ h x₀) ~ ₁ (₁ h x₁)
  fact₀ h {x₀} {x₁} q with refl ← Nat-discrete {x₀} {x₁} q = B .refl* (₁ (₁ h x₀))

  fact₁ : (h : |Π| Nat (Exists (B [ wk ]) R) γ)
        → {x₀ x₁ : ℕ} (q : x₀ N.~ x₁)
        → R ∋ (Γ .ref γ , q) , fact₀ h q ⊢ ₂ (₁ h x₀) ~ ₂ (₁ h x₁)
  fact₁ h {x₀} {x₁} q with refl ← Nat-discrete {x₀} {x₁} q = R .refl* (₂ (₁ h x₀))
cc B R .resp γ₀₁ x y q = tt
