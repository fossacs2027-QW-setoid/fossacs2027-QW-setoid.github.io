open import Prelude

module Model.Prop.QWI.Rec where

open import Model.Prop.CwF hiding (A; B)
open import Model.Prop.Pi
open import Model.Prop.QWI.Signature
open import Model.Prop.QWI.Core using (QWI)
open import Model.Prop.QWI.Elim

module _ {Γ : Con ℓΓ} (S : Poly Γ) (ε : EqSys S) (C : Ty (Γ ▷ I S) ℓ) where

  QWI-rec-step : Set _
  QWI-rec-step = QWI-step S ε (C [ wk ])

  QWI-rec-coh : QWI-rec-step → Set (ℓΓ ⊔ ℓ)
  QWI-rec-coh = QWI-elim-coh S ε (C [ wk ])

  rec-QWI : (p : QWI-rec-step) → QWI-rec-coh p
    → Tm (Γ ▷ I S ▷ QWI S ε) (C [ wk ])
  rec-QWI = elim-QWI S ε (C [ wk ])

  rec-QWIᵢ : (p : QWI-rec-step) → QWI-rec-coh p
    → Tm (Γ ▷ I S) (Π (QWI S ε) (C [ wk ]))
  rec-QWIᵢ = elim-QWIᵢ S ε (C [ wk ])
