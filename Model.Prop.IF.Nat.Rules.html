open import Prelude

module Model.Prop.IF.Nat.Rules where

open import Model.Prop.CwF
open import Model.Prop.IF.Nat.Core
open import Model.Prop.IF.Nat.Elim
open import Model.Prop.IF.Nat.Derived

Nat[] : Nat [ σ ] ≡ Nat
Nat[] = refl

z[] : z [ σ ]t ≡ z
z[] = refl

s[]
  : {t : Tm Γ Nat}
  → (s t) [ σ ]t ≡ s (t [ σ ]t)
s[] = refl

module _ (P : Ty (Γ ▷ Nat) ℓ) (pz : Tm Γ (P ⟨ z ⟩)) (ps : Tm (Γ ▷ Nat ▷ P) (P [ wk∘wk ▸ s (vs vz) ])) where
  elim-z : elim-Nat P pz ps ⟨ z ⟩t ≡ pz
  elim-z = refl

  elim-s : (n : Tm Γ Nat) → elim-Nat P pz ps ⟨ s n ⟩t ≡ ps [ idS ▸ n ▸ elim-Nat P pz ps ⟨ n ⟩t ]t
  elim-s n = refl

  elim-sⁱ : elim-Nat P pz ps [ wk ▸ sⁱ ]t ≡ ps [ idS ▸ elim-Nat P pz ps ]t
  elim-sⁱ = refl

  elim-Nat[]
    : {σ : Sub Δ Γ}
    → elim-Nat P pz ps [ σ ⁺ ]t ≡ elim-Nat (P [ σ ⁺ ]) (pz [ σ ]t) (ps [ σ ⁺ ⁺ ]t)
  elim-Nat[] = refl
