open import Prelude

module Model.Prop.W.Rules where

open import Model.Prop.CwF
open import Model.Prop.Pi

open import Model.Prop.W.Core
open import Model.Prop.W.Elim
open import Model.Prop.W.Rec

W[]
  : {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero} {σ : Sub Δ Γ}
  → W A B [ σ ] ≡ W (A [ σ ]) (B [ σ ⁺ ])
W[] = refl

sup-W[]
  : {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero} {σ : Sub Δ Γ}
  → (x : Tm Γ A) (f : Tm Γ (Π (B ⟨ x ⟩) (W A B [ wk ])))
  → sup-W A B x f [ σ ]t
  ≡ sup-W (A [ σ ]) (B [ σ ⁺ ]) (x [ σ ]t) (f [ σ ]t)
sup-W[] x f = refl

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero} {σ : Sub Δ Γ}
  where

  elim-W[]
    : (P : Ty (Γ ▷ W A B) ℓ)
    → (p : _)
    → elim-W B P p [ σ ⁺ ]t ≡
      elim-W (B [ σ ⁺ ]) (P [ σ ⁺ ]) (p [ σ ⁺ ⁺ ⁺ ]t)
  elim-W[] P p = refl

  elim-sup-W
    : {P : Ty (Γ ▷ W A B) ℓ}
    → {p : _}
    → {x : Tm Γ A} {f : Tm Γ ((B ⟨ x ⟩) ⇒ W A B)}
    → elim-W B P p ⟨ sup-W A B x f ⟩t
    ≡ p [ idS ▸ x ▸ f ▸
      (ƛ (elim-W B P p [ wk {A = A} ∘ wk {A = Π B (W A B [ wk ∘ wk ])} ∘ wk {A = B [ wk ]} ▸
      (app {Γ = Γ ▷ A ▷ Π B (W A B [ wk ∘ wk ]) ▷ B [ wk ]} {B = W A B [ wk ∘ wk ∘ wk ∘ wk ]} (vs vz) ⟨ vz ⟩t) ]t)) [ idS ▸ x ▸ f ]t ]t
  elim-sup-W = refl

-- These equalities check sharing on arbitrary terms, not just constructors.
module _ {Γ : Con ℓΓ} {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
  (C : Ty Γ ℓ) (p : W-rec-step B C) where

  rec-W-elim : rec-W B C p ≡ elim-W B (C [ wk ]) p
  rec-W-elim = refl

  rec-W[] : {σ : Sub Δ Γ}
    → rec-W B C p [ σ ⁺ ]t
    ≡ rec-W (B [ σ ⁺ ]) (C [ σ ]) (p [ σ ⁺ ⁺ ⁺ ]t)
  rec-W[] = refl

  rec-sup-W : (a : Tm Γ A) (f : Tm Γ ((B ⟨ a ⟩) ⇒ W A B))
    → rec-Wₑ B C p (sup-W A B a f)
    ≡ p [ idS ▸ a ▸ f ▸
      ƛ: (B ⟨ a ⟩) (rec-Wₑ (B [ wk ⁺ ]) (C [ wk ]) (p [ wk ⁺ ⁺ ⁺ ]t) (app f)) ]t
  rec-sup-W a f = refl
