module Model.Prop.Product.Core where

open import Model.Prop.CwF

module _ (A : Ty Γ ℓA) (B : Ty (Γ ▷ A) ℓB) where
  Sigma : Ty Γ (ℓA ⊔ ℓB)
  ∣ Sigma ∣ γ = Σ (∣ A ∣ γ) λ a → ∣ B ∣ (γ , a)
  Sigma .heq    γ₀₁ (a₀ , b₀) (a₁ , b₁) = ∃ (A ∋ γ₀₁ ⊢ a₀ ~ a₁) (λ a₀₁ → B ∋ (γ₀₁ , a₀₁) ⊢ b₀ ~ b₁)
  Sigma .refl*      (a₀ , b₀)       = A .refl* a₀ , B .refl* b₀
  Sigma .sym* (a₀ , b₀) (a₁ , b₁) (a₀₁ , b₀₁) =
    A .sym* a₀ a₁ a₀₁ , B .sym* b₀ b₁ b₀₁
  Sigma .trans* (a₀ , b₀) (a₁ , b₁) (a₂ , b₂) (a₀₁ , b₀₁) (a₁₂ , b₁₂) =
    A .trans* a₀ a₁ a₂ a₀₁ a₁₂ , B .trans* b₀ b₁ b₂ b₀₁ b₁₂
  Sigma .coe    γ₀₁ (a₀ , b₀)        = A .coe γ₀₁ a₀ , B .coe (γ₀₁ , A .coh γ₀₁ a₀) b₀
  Sigma .coh    γ₀₁ (a₀ , b₀)        = A .coh γ₀₁ a₀ , B .coh (γ₀₁ , A .coh γ₀₁ a₀) b₀
  {-# INJECTIVE_FOR_INFERENCE Sigma #-}

  Exists : Ty Γ (ℓA ⊔ ℓB)
  ∣ Exists ∣ γ = Σ (∣ A ∣ γ) λ a → ∣ B ∣ (γ , a)
  Exists .heq γ₀₁ _ _ = ⊤ₚ
  Exists .refl* x = tt
  Exists .sym* _ _ _ = tt
  Exists .trans* _ _ _ _ _ = tt
  Exists .coe γ₀₁ (a₀ , b₀) = A .coe γ₀₁ a₀ , B .coe (γ₀₁ , A .coh γ₀₁ a₀) b₀
  Exists .coh γ₀₁ _ = tt
  {-# INJECTIVE_FOR_INFERENCE Sigma #-}

module _ (A : Ty Γ ℓA) (B : Ty Γ ℓB) where
  infixr 6 _ẋ_
  _ẋ_ : Ty Γ (ℓA ⊔ ℓB)
  _ẋ_ .∣_∣ γ = Σ (∣ A ∣ γ) λ a → ∣ B ∣ γ
  _ẋ_ .heq γ₀₁ (a₀ , b₀) (a₁ , b₁) = (A ∋ γ₀₁ ⊢ a₀ ~ a₁) ∧ (B ∋ γ₀₁ ⊢ b₀ ~ b₁)
  _ẋ_ .refl* (a₀ , b₀) = A .refl* a₀ , B .refl* b₀
  _ẋ_ .sym* (a₀ , b₀) (a₁ , b₁) x₀₁ =
    A .sym* a₀ a₁ (x₀₁ .₁) , B .sym* b₀ b₁ (x₀₁ .₂)
  _ẋ_ .trans* (a₀ , b₀) (a₁ , b₁) (a₂ , b₂) x₀₁ x₁₂ =
    A .trans* a₀ a₁ a₂ (x₀₁ .₁) (x₁₂ .₁) , B .trans* b₀ b₁ b₂ (x₀₁ .₂) (x₁₂ .₂)
  _ẋ_ .coe = λ γ₀₁ x₀ → A .coe γ₀₁ (x₀ .₁) , B .coe γ₀₁ (x₀ .₂)
  _ẋ_ .coh = λ γ₀₁ x₀ → A .coh γ₀₁ (x₀ .₁) , B .coh γ₀₁ (x₀ .₂)
{-# INJECTIVE_FOR_INFERENCE _ẋ_ #-}

module _ {ℓΓ ℓA ℓB} {Γ : Con ℓΓ} {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB} where
  dpairⁱ : Tm (Γ ▷ A ▷ B) (Sigma A B [ wk∘wk ])
  ∣ dpairⁱ ∣ ((γ , a) , b) = a , b
  dpairⁱ .resp ((γ₀₁ , a₀₁) , b₀₁) = a₀₁ , b₀₁
  {-# INJECTIVE_FOR_INFERENCE dpairⁱ #-}
