module _ (P : Ty (Γ ▷ Nat) ℓ) (pz : Tm Γ (P ⟨ zⁱ ⟩)) (ps : Tm (Γ ▷ Nat ▷ P) (P [ wk ▸ sⁱ ] [ wk ])) where

  elim-N-at : (γ : ∣ Γ ∣) (m : ℕ) → ∣ P ∣ (γ , m)
  elim-N-at γ = N.elim (λ n → ∣ P ∣ (γ , n)) (∣ pz ∣ γ) (λ n pn → ∣ ps ∣ ((γ , n) , pn))

  elim-Nat : Tm (Γ ▷ Nat) P
  ∣ elim-Nat ∣   (γ , m) = elim-N-at γ m
  elim-Nat .resp {γ₀ , zero}   {γ₁ , zero}   (γ₀₁ , n₀₁) = pz .resp γ₀₁
  elim-Nat .resp {γ₀ , suc n₀} {γ₁ , suc n₁} (γ₀₁ , n₀₁) =
    ps .resp ((γ₀₁ , n₀₁) , elim-Nat .resp (γ₀₁ , n₀₁))
