open import Prelude
  hiding (↑)

module Model.Prop.CwF where

open import Model.Prop.CwF.Core public
open import Model.Prop.CwF.Prop public

-- Shorthands
wk : Sub (Γ ▷ A) Γ
wk {A = A} = π₁ A idS 
{-# INJECTIVE_FOR_INFERENCE wk #-}

vz : Tm (Γ ▷ A) (A [ wk ])
vz {A = A} = π₂ A idS

vs : Tm Γ A → Tm (Γ ▷ B) (A [ wk ])
vs t = t [ wk ]t
{-# INJECTIVE_FOR_INFERENCE vs #-}

infixl 6 _↑t_
_↑t_ : (σ : Sub Γ Δ) → (A : Ty Δ ℓA) → Sub (Γ ▷ A [ σ ]) (Δ ▷ A)
σ ↑t A = (σ ∘ π₁ _ idS) ▸ π₂ _ idS
{-# INJECTIVE_FOR_INFERENCE _↑t_ #-}

infixl 40 _⁺
_⁺ : (σ : Sub Γ Δ) → Sub (Γ ▷ A [ σ ]) (Δ ▷ A)
_⁺ {A = A} σ = σ ↑t A
{-# INJECTIVE_FOR_INFERENCE _⁺ #-}

-- Shorthands: Singleton substitutions
⟨_⟩s : Tm Γ A → Sub Γ (Γ ▷ A)
⟨_⟩s = λ t → idS ▸ t
{-# INJECTIVE_FOR_INFERENCE ⟨_⟩s #-}

infixl 10 _⟨_⟩ _⟨_⟩t

_⟨_⟩ : Ty (Γ ▷ A) ℓA → Tm Γ A → Ty Γ ℓA
B ⟨ x ⟩ = B [ idS ▸ x ]

_⟨_⟩t : Tm (Γ ▷ A) B → (x : Tm Γ A) → Tm Γ (B ⟨ x ⟩)
t ⟨ x ⟩t = t [ idS ▸ x ]t

-- `,` does not work due to ambiguity
infixl 10 _⟨_&_⟩ _⟨_&_⟩t

_⟨_&_⟩ : Ty (Γ ▷ A ▷ B [ wk ]) ℓC → Tm Γ A → Tm Γ B → Ty Γ ℓC
C ⟨ x & y ⟩ = C [ idS ▸ x ▸ y ]

_⟨_&_⟩t
  : {Γ : Con ℓΓ}{A : Ty Γ ℓA}{B : Ty Γ ℓB}{C : Ty (Γ ▷ A ▷ B [ wk ]) ℓC}
  → (t : Tm (Γ ▷ A ▷ B [ wk ]) C) (x : Tm Γ A) (y : Tm Γ B) → Tm Γ (C ⟨ x & y ⟩)
t ⟨ x & y ⟩t = t [ idS ▸ x ▸ y ]t

-- dependent versions of _⟨_&_⟩ and _⟨_&_⟩t
infixl 10 _⟨_&&_⟩ _⟨_&&_⟩t

_⟨_&&_⟩
  : (C : Ty (Γ ▷ A ▷ B) ℓC)
  → (x : Tm Γ A) → Tm Γ (B ⟨ x ⟩)
  → Ty Γ ℓC
C ⟨ x && y ⟩ = C [ idS ▸ x ▸ y ]

_⟨_&&_⟩t
  : Tm (Γ ▷ A ▷ B) C
  → (x : Tm Γ A) → (y : Tm Γ (B ⟨ x ⟩)) → Tm Γ (C ⟨ x && y ⟩)
t ⟨ x && y ⟩t = t [ idS ▸ x ▸ y ]t

-- Utilities for improving performance

-- Repeated weakening
wk∘wk : {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB} → Sub (Γ ▷ A ▷ B) Γ
wk∘wk {A = A} {B} = π₁ _ (π₁ _ idS)
{-# INJECTIVE_FOR_INFERENCE wk∘wk #-}

wk∘wk∘wk : {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB} {C : Ty (Γ ▷ A ▷ B) ℓC} → Sub (Γ ▷ A ▷ B ▷ C) Γ
wk∘wk∘wk {A = A} {B} {C} = π₁ _ (π₁ _ (π₁ _ idS))
{-# INJECTIVE_FOR_INFERENCE wk∘wk∘wk #-}

wk∘wk∘wk∘wk
  : {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB} {C : Ty (Γ ▷ A ▷ B) ℓC} {D : Ty (Γ ▷ A ▷ B ▷ C) ℓ}
  → Sub (Γ ▷ A ▷ B ▷ C ▷ D) Γ
wk∘wk∘wk∘wk {A = A} {B} {C} {D} = π₁ _ (π₁ _ (π₁ _ (π₁ _ idS)))
{-# INJECTIVE_FOR_INFERENCE wk∘wk∘wk∘wk #-}

wk∘wk∘wk∘wk∘wk
  : {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB} {C : Ty (Γ ▷ A ▷ B) ℓC} {D : Ty (Γ ▷ A ▷ B ▷ C) ℓ} {E : Ty (Γ ▷ A ▷ B ▷ C ▷ D) ℓ'}
  → Sub (Γ ▷ A ▷ B ▷ C ▷ D ▷ E) Γ
wk∘wk∘wk∘wk∘wk {A = A} {B} {C} {D} {E} = π₁ _ (π₁ _ (π₁ _ (π₁ _ (π₁ _ idS))))
{-# INJECTIVE_FOR_INFERENCE wk∘wk∘wk∘wk∘wk #-}
