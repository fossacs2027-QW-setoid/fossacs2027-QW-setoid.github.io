data Term₀ {I : Set} (A : I → Set) (B : ∀ {i} → A i → Set)
  (ar : ∀ {i a} → B {i} a → I) (X : I → Set) : I → Set where
  var₀ : ∀ {i} → X i → Term₀ A B ar X i
  op₀ : ∀ {i} (a : A i) → ((b : B a) → Term₀ A B ar X (ar b)) → Term₀ A B ar X i

module _ {I₀ I₁ : Set} {A₀ : I₀ → Set} {A₁ : I₁ → Set}
  {B₀ : ∀ {i} → A₀ i → Set} {B₁ : ∀ {i} → A₁ i → Set}
  {ar₀ : ∀ {i a} → B₀ {i} a → I₀} {ar₁ : ∀ {i a} → B₁ {i} a → I₁}
  {X₀ : I₀ → Set} {X₁ : I₁ → Set} where

  map : (R : I₀ → I₁ → Prop)
    → (h : ∀ {i j} → R i j → A₀ i → A₁ j)
    → (g : ∀ {i j} (r : R i j) {a : A₀ i} → B₁ (h r a) → B₀ a)
    → (comm : ∀ {i j} (r : R i j) {a : A₀ i} (b : B₁ (h r a)) → R (ar₀ (g r b)) (ar₁ b))
    → (k : ∀ {i j} → R i j → X₀ i → X₁ j)
    → ∀ {i j} → R i j → Term₀ A₀ B₀ ar₀ X₀ i → Term₀ A₁ B₁ ar₁ X₁ j
  map R h g comm k r (var₀ x) = var₀ (k r x)
  map R h g comm k r (op₀ a f) = op₀ (h r a)
    (λ b → map R h g comm k (comm r b) (f (g r b)))

module Term-rel
  {I₀ I₁ : Set} {A₀ : I₀ → Set} {A₁ : I₁ → Set}
  (A~ : ∀ {i j} → A₀ i → A₁ j → Prop)
  {B₀ : ∀ {i} → A₀ i → Set} {B₁ : ∀ {i} → A₁ i → Set}
  (B~ : ∀ {i j a a'} → B₀ {i} a → B₁ {j} a' → Prop)
  {ar₀ : ∀ {i a} → B₀ {i} a → I₀} {ar₁ : ∀ {i a} → B₁ {i} a → I₁}
  {X₀ : I₀ → Set} {X₁ : I₁ → Set}
  (X~ : ∀ {i j} → X₀ i → X₁ j → Prop) where

  Term₀~ : ∀ {i j} → Term₀ A₀ B₀ ar₀ X₀ i → Term₀ A₁ B₁ ar₁ X₁ j → Prop
  Term₀~ (var₀ x) (var₀ y) = X~ x y
  Term₀~ (op₀ a f) (op₀ a' f') = A~ a a' ∧ (∀ {b b'} → B~ b b' → Term₀~ (f b) (f' b'))
  Term₀~ _ _ = ⊥ₚ

module _ {I : Set} {A : I → Set} {B : ∀ {i} → A i → Set}
  {ar : ∀ {i a} → B {i} a → I} {X : I → Set}
  (A~ : ∀ {i j} → A i → A j → Prop)
  (B~ : ∀ {i j a a'} → B {i} a → B {j} a' → Prop)
  (X~ : ∀ {i j} → X i → X j → Prop) where

  open Term-rel {I₀ = I} {I₁ = I} {A₀ = A} {A₁ = A} A~
    {B₀ = B} {B₁ = B} B~ {ar₀ = ar} {ar₁ = ar} {X₀ = X} {X₁ = X} X~

  extTerm : ∀ {i} → Term₀ A B ar X i → Prop
  extTerm (var₀ _) = ⊤ₚ
  extTerm (op₀ a f) = (∀ {b b'} → B~ b b' → Term₀~ (f b) (f b')) ∧ ((b : B a) → extTerm (f b))

  Term₀~-refl : (Arefl : ∀ {i} (a : A i) → A~ a a)
    → (Xrefl : ∀ {i} (x : X i) → X~ x x)
    → ∀ {i} (t : Term₀ A B ar X i) → extTerm t → Term₀~ t t
  Term₀~-refl Arefl Xrefl (var₀ x) _ = Xrefl x
  Term₀~-refl Arefl Xrefl (op₀ a f) (f~ , _) = Arefl a , f~

module Construction {Γ : Con ℓΓ} (S : Poly Γ) (X : Ty (Γ ▷ I S) lzero) where
  open At S

  X₀ : ∀ {γ} → I₀ γ → Set
  X₀ {γ} i = ∣ X ∣ (γ , i)

  X~ : ∀ {γ δ} → Γ ∋ γ ~ δ → ∀ {i j} → X₀ {γ} i → X₀ {δ} j → Prop
  X~ q {i} {j} x y = ∃[ ij ∈ (I S ∋ q ⊢ i ~ j) ] (X ∋ (q , ij) ⊢ x ~ y)

  Raw : (γ : ∣ Γ ∣) → I₀ γ → Set
  Raw γ = Term₀ (A₀ {γ}) B₀ ar₀ X₀

  Rel : ∀ {γ δ} → Γ ∋ γ ~ δ → ∀ {i j} → Raw γ i → Raw δ j → Prop
  Rel q = Term-rel.Term₀~ (A~ q) (B~ q) (X~ q)

  Ext : ∀ {γ i} → Raw γ i → Prop
  Ext {γ} = extTerm (A~ (Γ .ref γ)) (B~ (Γ .ref γ)) (X~ (Γ .ref γ))

  rel-index : ∀ {γ δ} {q : Γ ∋ γ ~ δ} {i j} {t : Raw γ i} {u : Raw δ j}
    → Rel q t u → I S ∋ q ⊢ i ~ j
  rel-index {t = var₀ x} {var₀ y} xy = xy .₁
  rel-index {t = op₀ a f} {op₀ a' f'} (aa , ff) = aa .₁

  rel-refl : ∀ {γ i} (t : Raw γ i) → Ext t → Rel (Γ .ref γ) t t
  rel-refl = Term₀~-refl _ _ _ A-refl (λ x → I S .refl* _ , X .refl* x)

  rel-sym : ∀ {γ δ} {q : Γ ∋ γ ~ δ} {i j} (t : Raw γ i) (u : Raw δ j)
    → Rel q t u → Rel (Γ .sym q) u t
  rel-sym (var₀ x) (var₀ y) (ij , xy) = I S .sym* _ _ ij , X .sym* _ _ xy
  rel-sym (op₀ a f) (op₀ a' f') (aa , ff) = A-sym aa , λ {b} {b'} bb → rel-sym (f b') (f' b) (ff (B-sym bb))

  rel-trans : ∀ {γ δ θ} {q : Γ ∋ γ ~ δ} {r : Γ ∋ δ ~ θ} {i j k}
    {t : Raw γ i} {u : Raw δ j} {v : Raw θ k}
    → Rel q t u → Rel r u v → Rel (Γ .trans q r) t v
  rel-trans {t = var₀ x} {var₀ y} {var₀ z} (ij , xy) (jk , yz) =
    I S .trans* _ _ _ ij jk , X .trans* _ _ _ xy yz
  rel-trans {t = op₀ a f} {op₀ a' f'} {op₀ a'' f''} (aa , ff) (aa' , ff') =
    A-trans aa aa' , λ {b} {b''} bb → rel-trans
      {t = f b} {u = f' (B-coe aa b)} {v = f'' b''}
      (ff (B-coh aa b)) (ff' (B-trans (B-sym (B-coh aa b)) bb))

  coerce : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) → I S ∋ q ⊢ i ~ j → Raw γ i → Raw δ j
  coerce q = map (I S .heq q) (λ ij → A S .coe (q , ij))
    (pull q) (λ ij b → ar~ (pull~ q ij b)) (λ ij → X .coe (q , ij))

  coerce-rel : ∀ {γ δ i i' j j'} (q : Γ ∋ γ ~ δ)
    (ij : I S ∋ q ⊢ i ~ j) (ij' : I S ∋ q ⊢ i' ~ j')
    {t : Raw γ i} {u : Raw γ i'} → Rel (Γ .ref γ) t u → Rel (Γ .ref δ) (coerce q ij t) (coerce q ij' u)
  coerce-rel q ij ij' {var₀ x} {var₀ y} (ii , xy) =
    I S .trans* _ _ _ (I S .sym* _ _ ij) (I S .trans* _ _ _ ii ij')
    , X .trans* _ _ _ (X .sym* _ _ (X .coh (q , ij) x)) (X .trans* _ _ _ xy (X .coh (q , ij') y))
  coerce-rel q ij ij' {op₀ a f} {op₀ a' f'} (aa , ff) =
    A-trans (A-sym (ij , A S .coh (q , ij) a)) (A-trans aa (ij' , A S .coh (q , ij') a'))
    , λ {b} {b'} bb → coerce-rel q (ar~ (pull~ q ij b)) (ar~ (pull~ q ij' b'))
        {t = f (pull q ij b)} {u = f' (pull q ij' b')}
        (ff (B-trans (pull~ q ij b) (B-trans bb (B-sym (pull~ q ij' b')))))

  coerce-ext : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) (ij : I S ∋ q ⊢ i ~ j)
    (t : Raw γ i) → Ext t → Ext (coerce q ij t)
  coerce-ext q ij (var₀ x) _ = tt
  coerce-ext q ij (op₀ a f) (f~ , extf) =
    (λ {b} {b'} bb → coerce-rel q (ar~ (pull~ q ij b)) (ar~ (pull~ q ij b'))
      {t = f (pull q ij b)} {u = f (pull q ij b')}
      (f~ (B-trans (pull~ q ij b) (B-trans bb (B-sym (pull~ q ij b'))))))
    , λ b → coerce-ext q (ar~ (pull~ q ij b)) (f (pull q ij b)) (extf (pull q ij b))

  coerce-coh : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) (ij : I S ∋ q ⊢ i ~ j)
    (t : Raw γ i) → Ext t → Rel q t (coerce q ij t)
  coerce-coh q ij (var₀ x) _ = ij , X .coh (q , ij) x
  coerce-coh q ij (op₀ a f) (f~ , extf) = (ij , A S .coh (q , ij) a)
    , λ {b} {b'} bb → rel-trans {t = f b} {u = f (pull q ij b')}
        (f~ (B-trans bb (B-sym (pull~ q ij b'))))
        (coerce-coh q (ar~ (pull~ q ij b')) (f (pull q ij b')) (extf (pull q ij b')))

  Term : Ty (Γ ▷ I S) lzero
  ∣ Term ∣ (γ , i) = Σₚ (Raw γ i) Ext
  Term .heq (q , ij) (t , _) (u , _) = Rel q t u
  Term .refl* (t , extt) = rel-refl t extt
  Term .sym* (t , _) (u , _) = rel-sym t u
  Term .trans* (t , _) (u , _) (v , _) = rel-trans {t = t} {u} {v}
  Term .coe (q , ij) (t , extt) = coerce q ij t , coerce-ext q ij t extt
  Term .coh (q , ij) (t , extt) = coerce-coh q ij t extt

  var : Tm (Γ ▷ I S ▷ X) (Term [ wk ])
  ∣ var ∣ ((γ , i) , x) = var₀ x , tt
  var .resp ((q , ij) , xy) = ij , xy

  op : Tm (Γ ▷ I S ▷ A S ▷ Branch S Term) (Term [ wk∘wk ])
  ∣ op ∣ (((γ , i) , a) , f) = op₀ a (λ b → f .₁ b .₁)
    , (λ bb → f .₂ (bb .₂)) , λ b → f .₁ b .₂
  op .resp (((q , ij) , aa) , ff) = (ij , aa) , λ bb → ff _ _ (bb .₂)

  varₑ : Tm (Γ ▷ I S) X → Tm (Γ ▷ I S) Term
  varₑ x = var ⟨ x ⟩t

  opₑ : (a : Tm (Γ ▷ I S) (A S))
    → Tm (Γ ▷ I S) (Branch S Term ⟨ a ⟩) → Tm (Γ ▷ I S) Term
  opₑ a f = op [ idS ▸ a ▸ f ]t

open Construction public using (Term; var; op; varₑ; opₑ)
