module FreeSteps {Γ : Con ℓΓ} (S : Poly Γ) (X : Ty (Γ ▷ I S) lzero)
  (P : Ty (Γ ▷ I S ▷ Term S X) ℓ) where

  Node : Con ℓΓ
  Node = Γ ▷ I S ▷ A S ▷ Branch S (Term S X)

  child : Sub (Node ▷ B S [ wk ]) (Γ ▷ I S ▷ Term S X)
  ∣ child ∣ ((((γ , i) , a) , f) , b) = (γ , ∣ ar S ∣ (((γ , i) , a) , b)) , f .₁ b
  child .resp ((((q , ij) , aa) , ff) , bb) = (q , ar S .resp (((q , ij) , aa) , bb)) , ff _ _ bb

  node : Sub Node (Γ ▷ I S ▷ Term S X)
  node = wk∘wk ▸ op S X

  IH : Ty Node ℓ
  IH = Π (B S [ wk ]) (P [ child ])

  Var-step : Set _
  Var-step = Tm (Γ ▷ I S ▷ X) (P [ wk ▸ var S X ])

  Op-step : Set _
  Op-step = Tm (Node ▷ IH) (P [ node ∘ wk ])

open FreeSteps public

module _ {Γ : Con ℓΓ} (S : Poly Γ) (X : Ty (Γ ▷ I S) lzero)
  (P : Ty (Γ ▷ I S ▷ Term S X) ℓ)
  (v : Var-step S X P) (p : Op-step S X P) where

  open At S
  open Construction S X using (Raw; Ext; Rel; X~; rel-index)

  elim₀ : ∀ {γ i} (t : Raw γ i) (extt : Ext t) → ∣ P ∣ ((γ , i) , (t , extt))
  elim₀ {γ} = Elim₀.val (A~ (Γ .ref γ)) (B~ (Γ .ref γ)) (X~ (Γ .ref γ))
    (λ {i} t → ∣ P ∣ ((γ , i) , t))
    (λ {t = t} {t' = t'} tt' → P .heq ((Γ .ref γ , rel-index {t = t .₁} {u = t' .₁} tt') , tt'))
    (λ x → ∣ v ∣ ((_ , _) , x)) (λ xy → v .resp ((Γ .ref γ , xy .₁) , xy .₂))
    (λ a f f~ ih ih~ → ∣ p ∣ ((((_ , _) , a)
      , f , λ bb → f~ (A-refl a , bb)) , ih , λ bb → ih~ (A-refl a , bb)))
    (λ ih~ ih'~ aa ff ii → p .resp
      ((((Γ .ref γ , aa .₁) , aa .₂) , λ b b' bb → ff (aa , bb)) , λ b b' bb → ii (aa , bb)))

  elim~ : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) (t : Raw γ i) (u : Raw δ j)
    (extt : Ext t) (extu : Ext u) (tu : Rel q t u)
    → P ∋ ((q , rel-index {t = t} {u} tu) , tu) ⊢ elim₀ t extt ~ elim₀ u extu
  elim~ q (var₀ x) (var₀ y) _ _ (ij , xy) = v .resp ((q , ij) , xy)
  elim~ q (op₀ a f) (op₀ a' f') (f~ , extf) (f'~ , extf') (aa , ff) =
    p .resp ((((q , aa .₁) , aa .₂) , λ b b' bb → ff (aa , bb))
      , λ b b' bb → elim~ q (f b) (f' b') (extf b) (extf' b') (ff (aa , bb)))

  elim-Term : Tm (Γ ▷ I S ▷ Term S X) P
  ∣ elim-Term ∣ ((γ , i) , (t , extt)) = elim₀ t extt
  elim-Term .resp {((_ , _) , (t , extt))} {((_ , _) , (u , extu))} ((q , ij) , tu) =
    elim~ q t u extt extu tu

  elim-Termₑ : (t : Tm (Γ ▷ I S) (Term S X)) → Tm (Γ ▷ I S) (P ⟨ t ⟩)
  elim-Termₑ t = elim-Term ⟨ t ⟩t
