module Prelude.Product where

-- Standalone replacement for the standard library's `Data.Product.Base`.
-- Provides exactly the names re-exported by `Prelude.Base`.

open import Agda.Primitive

private variable
  a b : Level

infixr 4 _,_ _,′_
infixr 2 _×_

record Σ (A : Set a) (B : A → Set b) : Set (a ⊔ b) where
  constructor _,_
  field
    proj₁ : A
    proj₂ : B proj₁
open Σ public

Σ-syntax : (A : Set a) → (A → Set b) → Set (a ⊔ b)
Σ-syntax = Σ

syntax Σ-syntax A (λ x → B) = Σ[ x ∈ A ] B

_×_ : (A : Set a) (B : Set b) → Set (a ⊔ b)
A × B = Σ[ _ ∈ A ] B

_,′_ : {A : Set a} {B : Set b} → A → B → A × B
_,′_ = _,_
