open import Prelude

module Model.Prop.QWI.Free.Rules where

open import Model.Prop.CwF hiding (A; B)
open import Model.Prop.Pi
open import Model.Prop.QWI.Polynomial
open import Model.Prop.QWI.Free.Core
open import Model.Prop.QWI.Free.Elim
open import Model.Prop.QWI.Free.Rec

module _ {Γ : Con ℓΓ} {Δ : Con ℓΔ} {S : Poly Γ} {X : Ty (Γ ▷ I S) lzero} {σ : Sub Δ Γ} where
  Term[] : Term S X [ σ ⁺ ] ≡ Term (S [ σ ]s) (X [ σ ⁺ ])
  Term[] = refl

  var[] : var S X [ σ ⁺ ⁺ ]t ≡ var (S [ σ ]s) (X [ σ ⁺ ])
  var[] = refl

  op[] : op S X [ σ ⁺ ⁺ ⁺ ]t ≡ op (S [ σ ]s) (X [ σ ⁺ ])
  op[] = refl

  elim-Term[] : (P : Ty (Γ ▷ I S ▷ Term S X) ℓ)
    (v : Var-step S X P) (p : Op-step S X P)
    → elim-Term S X P v p [ σ ⁺ ⁺ ]t
      ≡ elim-Term (S [ σ ]s) (X [ σ ⁺ ]) (P [ σ ⁺ ⁺ ])
          (v [ σ ⁺ ⁺ ]t) (p [ σ ⁺ ⁺ ⁺ ⁺ ]t)
  elim-Term[] P v p = refl

  elim-Termₑ[] : (P : Ty (Γ ▷ I S ▷ Term S X) ℓ)
    (v : Var-step S X P) (p : Op-step S X P) (t : Tm (Γ ▷ I S) (Term S X))
    → elim-Termₑ S X P v p t [ σ ⁺ ]t
      ≡ elim-Termₑ (S [ σ ]s) (X [ σ ⁺ ]) (P [ σ ⁺ ⁺ ])
          (v [ σ ⁺ ⁺ ]t) (p [ σ ⁺ ⁺ ⁺ ⁺ ]t) (t [ σ ⁺ ]t)
  elim-Termₑ[] P v p t = refl

module _ {Γ : Con ℓΓ} {S : Poly Γ} {X : Ty (Γ ▷ I S) lzero}
  (P : Ty (Γ ▷ I S ▷ Term S X) ℓ) (v : Var-step S X P) (p : Op-step S X P) where

  elim-var : elim-Term S X P v p [ wk ▸ var S X ]t ≡ v
  elim-var = refl

  elim-op : elim-Term S X P v p [ node S X P ]t
    ≡ p [ idS ▸ ƛ (elim-Term S X P v p [ child S X P ]t) ]t
  elim-op = refl

module _ {Γ : Con ℓΓ} {S : Poly Γ} {X : Ty (Γ ▷ I S) lzero}
  (C : Ty (Γ ▷ I S) ℓ) (v : Term-rec-var-step S X C) (p : Term-rec-op-step S X C) where

  rec-Term-elim : rec-Term S X C v p ≡ elim-Term S X (C [ wk ]) v p
  rec-Term-elim = refl

  rec-Term[] : {Δ : Con ℓΔ} {σ : Sub Δ Γ}
    → rec-Term S X C v p [ σ ⁺ ⁺ ]t
      ≡ rec-Term (S [ σ ]s) (X [ σ ⁺ ]) (C [ σ ⁺ ]) (v [ σ ⁺ ⁺ ]t) (p [ σ ⁺ ⁺ ⁺ ⁺ ]t)
  rec-Term[] = refl

  rec-Termₑ[] : {Δ : Con ℓΔ} {σ : Sub Δ Γ} (t : Tm (Γ ▷ I S) (Term S X))
    → rec-Termₑ S X C v p t [ σ ⁺ ]t
      ≡ rec-Termₑ (S [ σ ]s) (X [ σ ⁺ ]) (C [ σ ⁺ ])
          (v [ σ ⁺ ⁺ ]t) (p [ σ ⁺ ⁺ ⁺ ⁺ ]t) (t [ σ ⁺ ]t)
  rec-Termₑ[] t = refl

  rec-Term-var : rec-Term S X C v p [ wk ▸ var S X ]t ≡ v
  rec-Term-var = refl

  rec-Term-op : rec-Term S X C v p [ node S X (C [ wk ]) ]t
    ≡ p [ idS ▸ ƛ (rec-Term S X C v p [ child S X (C [ wk ]) ]t) ]t
  rec-Term-op = refl
