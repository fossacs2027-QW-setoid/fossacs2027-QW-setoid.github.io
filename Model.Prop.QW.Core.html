record EqSys₀ (A : Set) (B : A → Set) : Set₁ where
  field
    E       : Set
    V       : E → Set
    V~      : (e : E) → V e → V e → Prop
    lhs rhs : (e : E) → Term₀ A B (V e)
open EqSys₀ public

eval : ∀ {A : Set} {B : A → Set} {X : Set}
  → Term₀ A B X → (X → 𝒲 A B) → 𝒲 A B
eval (var₀ x)  ρ = ρ x
eval (op₀ a f) ρ = sup₀ a (λ y → eval (f y) ρ)

module QW-rel
  {A : Set} (B : A → Set)
  (A~ : A → A → Prop)
  (B~ : ∀ {a₀ a₁} → B a₀ → B a₁ → Prop)
  (ε : EqSys₀ A B) where

  open Term-rel A~ B B B~ using (Term₀~)

  mutual
    data QW₀~ : 𝒲 A B → 𝒲 A B → Prop where
      qw-sym
        : ∀ {w₀ w₁} → QW₀~ w₀ w₁ → QW₀~ w₁ w₀
      qw-trans
        : ∀ {w₀ w₁ w₂}
        → Ext w₁ → QW₀~ w₀ w₁ → QW₀~ w₁ w₂
        → QW₀~ w₀ w₂
      qw-sup
        : ∀ {a₀ a₁} {f₀ : B a₀ → 𝒲 A B} {f₁ : B a₁ → 𝒲 A B}
        → A~ a₀ a₁
        → (∀ {b₀ b₁} → B~ b₀ b₁ → QW₀~ (f₀ b₀) (f₁ b₁))
        → QW₀~ (sup₀ a₀ f₀) (sup₀ a₁ f₁)
      -- Only extensional assignments instantiate an internal equation.
      qw-equ : (e : ε .E)
        → (ρ : ε .V e → Σₚ (𝒲 A B) Ext)
        → (ρ~ : ∀ {x y} → ε .V~ e x y → QW₀~ (ρ x .₁) (ρ y .₁))
        → QW₀~
            (eval (ε .lhs e) (λ v → ρ v .₁))
            (eval (ε .rhs e) (λ v → ρ v .₁))

    Ext : 𝒲 A B → Prop
    Ext (sup₀ a f) =
      ({y₀ y₁ : B a} (y₀₁ : B~ y₀ y₁) → QW₀~ (f y₀) (f y₁))
      ∧ ((y : B a) → Ext (f y))

  |𝒲| : Set
  |𝒲| = Σₚ (𝒲 A B) Ext

  qw-refl : (Arefl : ∀ a → A~ a a) → ∀ {w} → Ext w → QW₀~ w w
  qw-refl Arefl {sup₀ a f} (f~ , _) = qw-sup (Arefl a) f~

  eval~ : ∀ {X₀ X₁} (X~ : X₀ → X₁ → Prop) {t₀ : Term₀ A B X₀} {t₁ : Term₀ A B X₁}
    → Term₀~ X~ t₀ t₁
    → (ρ₀ : X₀ → |𝒲|) (ρ₁ : X₁ → |𝒲|)
    → (∀ {x y} → X~ x y → QW₀~ (ρ₀ x .₁) (ρ₁ y .₁))
    → QW₀~ (eval t₀ (λ x → ρ₀ x .₁)) (eval t₁ (λ x → ρ₁ x .₁))
  eval~ X~ {var₀ x₀} {var₀ x₁} x₀₁ ρ₀ ρ₁ ρ~ = ρ~ x₀₁
  eval~ X~ {op₀ a₀ f₀} {op₀ a₁ f₁} (a₀₁ , f₀₁) ρ₀ ρ₁ ρ~ =
    qw-sup a₀₁ (λ {b₀} {b₁} b₀₁ →
      eval~ X~ {t₀ = f₀ b₀} {t₁ = f₁ b₁} (f₀₁ b₀₁) ρ₀ ρ₁ ρ~)

  eval-ext : ∀ {X} (X~ : X → X → Prop) (t : Term₀ A B X) → extTerm A~ B~ X~ t
    → (ρ : X → |𝒲|)
    → (∀ {x y} → X~ x y → QW₀~ (ρ x .₁) (ρ y .₁))
    → Ext (eval t (λ x → ρ x .₁))
  eval-ext X~ (var₀ x) _ ρ ρ~ = ρ x .₂
  eval-ext X~ (op₀ a f) (f~ , extf) ρ ρ~ =
    (λ {b₀} {b₁} b₀₁ → eval~ X~ {t₀ = f b₀} {t₁ = f b₁} (f~ b₀₁) ρ ρ ρ~)
    , λ b → eval-ext X~ (f b) (extf b) ρ ρ~

  eval-QExt : ∀ {X} (X~ : X → X → Prop) → Σₚ (Term₀ A B X) (extTerm A~ B~ X~)
    → (ρ : X → |𝒲|)
    → (∀ {x y} → X~ x y → QW₀~ (ρ x .₁) (ρ y .₁))
    → |𝒲|
  eval-QExt X~ (t , extt) ρ ρ~ = eval t (λ x → ρ x .₁) , eval-ext X~ t extt ρ ρ~

  module Evaluation (Arefl : ∀ a → A~ a a)
    (Q~ : |𝒲| → |𝒲| → Prop)
    (to-QEq : ∀ {u v} → Q~ u v → QW₀~ (u .₁) (v .₁))
    (from-QEq : ∀ {u v} → QW₀~ (u .₁) (v .₁) → Q~ u v)
    {X : Set} (X~ : X → X → Prop)
    (ρ : X → |𝒲|)
    (ρ~ : ∀ {x y} → X~ x y → QW₀~ (ρ x .₁) (ρ y .₁)) where

    open FreeRaw.Elim₀ A~ B~ X~ (λ _ → |𝒲|) (λ _ → Q~)
      ρ (λ xy → from-QEq (ρ~ xy))
      (λ a f f~ ih ih~ → sup₀ a (λ b → ih b .₁)
        , (λ bb → to-QEq (ih~ bb)) , λ b → ih b .₂)
      (λ ih~ ih'~ aa ff ii → from-QEq (qw-sup aa (λ bb → to-QEq (ii bb))))
      using (val; val~)

    eval-W : (t : Term₀ A B X) → extTerm A~ B~ X~ t → |𝒲|
    eval-W = val

    eval-W~ : {t₀ t₁ : Term₀ A B X}
      {ext₀ : extTerm A~ B~ X~ t₀} {ext₁ : extTerm A~ B~ X~ t₁}
      → Term₀~ X~ t₀ t₁ → QW₀~ (eval-W t₀ ext₀ .₁) (eval-W t₁ ext₁ .₁)
    eval-W~ {t₀} {t₁} {ext₀} {ext₁} tt' =
      to-QEq (val~ t₀ t₁ ext₀ ext₁ tt')

    rec-raw : (t : Term₀ A B X) (extt : extTerm A~ B~ X~ t)
      → QW₀~ (eval-W t extt .₁) (eval t (λ x → ρ x .₁))
    rec-raw (var₀ x) _ = qw-refl Arefl (ρ x .₂)
    rec-raw (op₀ a f) (f~ , extf) = qw-sup (Arefl a) λ {b} {b'} bb →
      qw-trans (eval-QExt X~ (f b , extf b) ρ ρ~ .₂)
        (rec-raw (f b) (extf b))
        (eval~ X~ {t₀ = f b} {t₁ = f b'} (f~ bb) ρ ρ ρ~)

    rec-eq : (t u : Term₀ A B X)
      (extt : extTerm A~ B~ X~ t) (extu : extTerm A~ B~ X~ u)
      → QW₀~ (eval t (λ x → ρ x .₁)) (eval u (λ x → ρ x .₁))
      → QW₀~ (eval-W t extt .₁) (eval-W u extu .₁)
    rec-eq t u extt extu eq =
      qw-trans (eval-QExt X~ (t , extt) ρ ρ~ .₂) (rec-raw t extt)
        (qw-trans (eval-QExt X~ (u , extu) ρ ρ~ .₂) eq
          (qw-sym (rec-raw u extu)))

QW₀ : (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero) → ∣ Γ ∣ → Set
QW₀ A B γ = 𝒲 (∣ A ∣ γ) (λ a → ∣ B ∣ (γ , a))

module _ (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero)
  (ε : EqSys A B) where

  open Term-rel using (Term₀~)

  SigAt : (γ : ∣ Γ ∣) → EqSys₀ (∣ A ∣ γ) (λ a → ∣ B ∣ (γ , a))
  SigAt γ .E = ∣ E ε ∣ γ
  SigAt γ .V e = ∣ V ε ∣ (γ , e)
  SigAt γ .V~ e = V ε ∋ (Γ .ref γ , E ε .refl* e) ⊢_~_
  SigAt γ .lhs e = ∣ lhs ε ∣ (γ , e) .₁
  SigAt γ .rhs e = ∣ rhs ε ∣ (γ , e) .₁

  module RelAt (γ : ∣ Γ ∣) = QW-rel (λ a → ∣ B ∣ (γ , a))
    (A ∋ Γ .ref γ ⊢_~_) (B~ A B (Γ .ref γ))
    (SigAt γ)

  open RelAt

  private
    g : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) {a₀ : ∣ A ∣ γ₀}
      → ∣ B ∣ (γ₁ , A .coe γ₀₁ a₀) → ∣ B ∣ (γ₀ , a₀)
    g γ₀₁ {a₀} b₁ = B .coe (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a₀)) b₁

    g~ : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) {a₀ a₀' : ∣ A ∣ γ₀}
        {b₁ : ∣ B ∣ (γ₁ , A .coe γ₀₁ a₀)} {b₁' : ∣ B ∣ (γ₁ , A .coe γ₀₁ a₀')}
      → B~ A B (Γ .ref γ₁) b₁ b₁' → B~ A B (Γ .ref γ₀) (g γ₀₁ b₁) (g γ₀₁ b₁')
    g~ γ₀₁ {a₀} {a₀'} {b₁} {b₁'} (a₁₁' , b₁₁') =
      A .trans* _ _ _ (A .coh γ₀₁ a₀) (A .trans* _ _ _ a₁₁' (A .sym* _ _ (A .coh γ₀₁ a₀')))
      , B .trans* _ _ _ (B .sym* _ _ (B .coh (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a₀)) b₁))
          (B .trans* _ _ _ b₁₁' (B .coh (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a₀')) b₁'))

  QW-coe : ∀ {γ₀ γ₁} → Γ ∋ γ₀ ~ γ₁ → QW₀ A B γ₀ → QW₀ A B γ₁
  QW-coe γ₀₁ = W.map (A .coe γ₀₁) (g γ₀₁)

  |QW| : ∣ Γ ∣ → Set
  |QW| γ = Σₚ (QW₀ A B γ) (Ext γ)

  eval-map : ∀ {γ₀ γ₁ X₀ X₁} (q : Γ ∋ γ₀ ~ γ₁) (X~ : X₀ → X₁ → Prop)
      {t₀ : Term₀ (∣ A ∣ γ₀) (λ a → ∣ B ∣ (γ₀ , a)) X₀}
      {t₁ : Term₀ (∣ A ∣ γ₁) (λ a → ∣ B ∣ (γ₁ , a)) X₁}
    → Term₀~ (A ∋ q ⊢_~_) _ _ (B~ A B q) X~ t₀ t₁
    → (ρ₀ : X₀ → QW₀ A B γ₀) (ρ₁ : X₁ → QW₀ A B γ₁)
    → (∀ {x y} → X~ x y → QW₀~ γ₁ (QW-coe q (ρ₀ x)) (ρ₁ y))
    → QW₀~ γ₁ (QW-coe q (eval t₀ ρ₀)) (eval t₁ ρ₁)
  eval-map q X~ {var₀ x₀} {var₀ x₁} x₀₁ ρ₀ ρ₁ ρ~ = ρ~ x₀₁
  eval-map q X~ {op₀ a₀ f₀} {op₀ a₁ f₁} (a₀₁ , f₀₁) ρ₀ ρ₁ ρ~ =
    qw-sup (A .trans* _ _ _ (A .sym* _ _ (A .coh q a₀)) a₀₁)
      (λ {b₀} {b₁} (_ , b₀₁) →
        eval-map q X~ {t₀ = f₀ (g q b₀)} {t₁ = f₁ b₁}
          (f₀₁ (a₀₁ , B .trans* _ _ _
            (B .sym* _ _ (B .coh (Γ .sym q , A .sym* _ _ (A .coh q a₀)) b₀)) b₀₁))
          ρ₀ ρ₁ ρ~)

  mutual
    QW-coe~ : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) {w₀ w₁}
      → QW₀~ γ₀ w₀ w₁ → QW₀~ γ₁ (QW-coe γ₀₁ w₀) (QW-coe γ₀₁ w₁)
    QW-coe~ γ₀₁ (qw-sym w₀₁) = qw-sym (QW-coe~ γ₀₁ w₀₁)
    QW-coe~ γ₀₁ (qw-trans extw₁ w₀₁ w₁₂) =
      qw-trans (ext-coe γ₀₁ _ extw₁)
        (QW-coe~ γ₀₁ w₀₁) (QW-coe~ γ₀₁ w₁₂)
    QW-coe~ γ₀₁ (qw-sup {a₀} {a₁} a₀₁ f₀₁) =
      qw-sup (A .trans* _ _ _ (A .sym* _ _ (A .coh γ₀₁ a₀)) (A .trans* _ _ _ a₀₁ (A .coh γ₀₁ a₁)))
        (λ b₀₁ → QW-coe~ γ₀₁ (f₀₁ (g~ γ₀₁ b₀₁)))
    QW-coe~ {γ₀} {γ₁} q (qw-equ e ρ ρ~) =
      qw-trans (eval-QExt _ (V~ (SigAt γ₁) e₁) (∣ lhs ε ∣ (γ₁ , e₁)) ρ₁ ρ₁~ .₂)
        (eval-map q (V ε ∋ (q , e₀₁) ⊢_~_)
          {t₀ = ∣ lhs ε ∣ (γ₀ , e) .₁} {t₁ = ∣ lhs ε ∣ (γ₁ , e₁) .₁}
          (lhs ε .resp (q , e₀₁)) (λ x → ρ x .₁) (λ x → ρ₁ x .₁) ρ₀₁)
        (qw-trans (eval-QExt _ (V~ (SigAt γ₁) e₁) (∣ rhs ε ∣ (γ₁ , e₁)) ρ₁ ρ₁~ .₂)
          (qw-equ e₁ ρ₁ ρ₁~)
          (qw-sym (eval-map q (V ε ∋ (q , e₀₁) ⊢_~_)
            {t₀ = ∣ rhs ε ∣ (γ₀ , e) .₁} {t₁ = ∣ rhs ε ∣ (γ₁ , e₁) .₁}
            (rhs ε .resp (q , e₀₁)) (λ x → ρ x .₁) (λ x → ρ₁ x .₁) ρ₀₁)))
      where
        e₁ = E ε .coe q e
        e₀₁ = E ε .coh q e
        back = Γ .sym q , E ε .sym* _ _ e₀₁
        pull : ∣ V ε ∣ (γ₁ , e₁) → ∣ V ε ∣ (γ₀ , e)
        pull = V ε .coe back
        ρ₁ : ∣ V ε ∣ (γ₁ , e₁) → |QW| γ₁
        ρ₁ v = QW-coe q (ρ (pull v) .₁) , ext-coe q (ρ (pull v) .₁) (ρ (pull v) .₂)
        ρ₁~ : ∀ {x y} → V~ (SigAt γ₁) e₁ x y → QW₀~ γ₁ (ρ₁ x .₁) (ρ₁ y .₁)
        ρ₁~ {x} {y} xy = QW-coe~ q (ρ~
          (V ε .trans* _ _ _ (V ε .sym* _ _ (V ε .coh back x))
            (V ε .trans* _ _ _ xy (V ε .coh back y))))
        ρ₀₁ : ∀ {x y} → V ε ∋ (q , e₀₁) ⊢ x ~ y
          → QW₀~ γ₁ (QW-coe q (ρ x .₁)) (ρ₁ y .₁)
        ρ₀₁ {y = y} xy = QW-coe~ q (ρ~ (V ε .trans* _ _ _ xy (V ε .coh back y)))

    ext-coe : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (w : QW₀ A B γ₀)
      → Ext γ₀ w → Ext γ₁ (QW-coe γ₀₁ w)
    ext-coe γ₀₁ (sup₀ a f) (f~ , extf) =
      (λ b₀₁ → QW-coe~ γ₀₁ (f~ (g~ γ₀₁ b₀₁)))
      , λ b → ext-coe γ₀₁ (f (g γ₀₁ b)) (extf (g γ₀₁ b))

  QW-coe-id : ∀ {γ} (w : QW₀ A B γ) → Ext γ w → QW₀~ γ w (QW-coe (Γ .ref γ) w)
  QW-coe-id (sup₀ a f) (f~ , extf) =
    qw-sup (A .coh (Γ .ref _) a) λ {b₀} {b₂} (a₀₂ , b₀₂) →
      let b₁ = g (Γ .ref _) b₂
      in qw-trans (extf b₁)
           (f~ (A .sym* _ _ (A .trans* _ _ _ (A .coh (Γ .ref _) a) (A .sym* _ _ a₀₂))
             , B .trans* _ _ _ b₀₂
                 (B .coh (Γ .sym (Γ .ref _) , A .sym* _ _ (A .coh (Γ .ref _) a)) b₂)))
           (QW-coe-id (f b₁) (extf b₁))

  QW-coe-∘ : ∀ {γ₀ γ₁ γ₂} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (γ₁₂ : Γ ∋ γ₁ ~ γ₂)
      (w : QW₀ A B γ₀) → Ext γ₀ w
    → QW₀~ γ₂ (QW-coe γ₁₂ (QW-coe γ₀₁ w)) (QW-coe (Γ .trans γ₀₁ γ₁₂) w)
  QW-coe-∘ γ₀₁ γ₁₂ (sup₀ a f) (f~ , extf) =
    qw-sup
      (A .trans* _ _ _ (A .sym* _ _ (A .coh γ₁₂ (A .coe γ₀₁ a)))
        (A .trans* _ _ _ (A .sym* _ _ (A .coh γ₀₁ a)) (A .coh (Γ .trans γ₀₁ γ₁₂) a)))
      (λ {b₀} {b₁} (_ , b₀₁) →
        let x₀ = g γ₀₁ (g γ₁₂ b₀)
        in qw-trans
             (ext-coe (Γ .trans γ₀₁ γ₁₂) (f x₀) (extf x₀))
             (QW-coe-∘ γ₀₁ γ₁₂ (f x₀) (extf x₀))
             (QW-coe~ (Γ .trans γ₀₁ γ₁₂) (f~ (A .refl* a
               , B .trans* _ _ _ (B .sym* _ _ (B .coh (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ a)) (g γ₁₂ b₀)))
                   (B .trans* _ _ _ (B .sym* _ _ (B .coh (Γ .sym γ₁₂ , A .sym* _ _ (A .coh γ₁₂ (A .coe γ₀₁ a))) b₀))
                     (B .trans* _ _ _ b₀₁ (B .coh (Γ .sym (Γ .trans γ₀₁ γ₁₂) , A .sym* _ _ (A .coh (Γ .trans γ₀₁ γ₁₂) a)) b₁)))))))

  QW~ : ∀ {γ₀ γ₁} → Γ ∋ γ₀ ~ γ₁ → |QW| γ₀ → |QW| γ₁ → Prop
  QW~ {γ₀} {γ₁} γ₀₁ (w₀ , _) (w₁ , _) = QW₀~ γ₁ (QW-coe γ₀₁ w₀) w₁

  QW-refl : ∀ {γ} (w : |QW| γ) → QW~ (Γ .ref γ) w w
  QW-refl (w , extw) = qw-sym (QW-coe-id w extw)

  QW-sym : ∀ {γ₀ γ₁} {γ₀₁ : Γ ∋ γ₀ ~ γ₁} {w₀ : |QW| γ₀} {w₁ : |QW| γ₁}
    → QW~ γ₀₁ w₀ w₁ → QW~ (Γ .sym γ₀₁) w₁ w₀
  QW-sym {γ₀₁ = γ₀₁} {w₀ = w₀ , extw₀} {w₁ = w₁ , extw₁} w₀₁ =
    qw-trans
      (ext-coe (Γ .sym γ₀₁) (QW-coe γ₀₁ w₀)
        (ext-coe γ₀₁ w₀ extw₀))
      (qw-sym (QW-coe~ (Γ .sym γ₀₁) w₀₁))
      (qw-trans
        (ext-coe (Γ .trans γ₀₁ (Γ .sym γ₀₁)) w₀ extw₀)
        (QW-coe-∘ γ₀₁ (Γ .sym γ₀₁) w₀ extw₀)
        (QW-refl (w₀ , extw₀)))

  QW-trans : ∀ {γ₀ γ₁ γ₂} {γ₀₁ : Γ ∋ γ₀ ~ γ₁} {γ₁₂ : Γ ∋ γ₁ ~ γ₂}
      {w₀ : |QW| γ₀} {w₁ : |QW| γ₁} {w₂ : |QW| γ₂}
    → QW~ γ₀₁ w₀ w₁ → QW~ γ₁₂ w₁ w₂ → QW~ (Γ .trans γ₀₁ γ₁₂) w₀ w₂
  QW-trans {γ₀₁ = γ₀₁} {γ₁₂}
    {w₀ = w₀ , extw₀} {w₁ = w₁ , extw₁} w₀₁ w₁₂ =
    qw-trans
      (ext-coe γ₁₂ (QW-coe γ₀₁ w₀) (ext-coe γ₀₁ w₀ extw₀))
      (qw-sym (QW-coe-∘ γ₀₁ γ₁₂ w₀ extw₀))
      (qw-trans (ext-coe γ₁₂ w₁ extw₁)
        (QW-coe~ γ₁₂ w₀₁) w₁₂)

  QW : Ty Γ lzero
  ∣ QW ∣ = |QW|
  QW .heq = QW~
  QW .refl* = QW-refl
  QW .sym* w₀ w₁ = QW-sym {w₀ = w₀} {w₁ = w₁}
  QW .trans* w₀ w₁ w₂ = QW-trans {w₀ = w₀} {w₁ = w₁} {w₂ = w₂}
  QW .coe γ₀₁ (w , extw) = QW-coe γ₀₁ w , ext-coe γ₀₁ w extw
  QW .coh q (w , extw) = qw-refl _ (A .refl*) (ext-coe q w extw)
  {-# INJECTIVE_FOR_INFERENCE QW #-}

  sup-QW : (a : Tm Γ A) (f : Tm Γ (Π (B ⟨ a ⟩) (QW [ wk ]))) → Tm Γ QW
  ∣ sup-QW a f ∣ γ =
    sup₀ (∣ a ∣ γ) (λ b → ∣ f ∣ γ .₁ b .₁)
    , (λ {b₀} {b₁} (_ , b₀₁) →
        qw-trans
          (ext-coe (Γ .ref γ)
            (∣ f ∣ γ .₁ b₀ .₁) (∣ f ∣ γ .₁ b₀ .₂))
          (QW-coe-id (∣ f ∣ γ .₁ b₀ .₁) (∣ f ∣ γ .₁ b₀ .₂))
          (∣ f ∣ γ .₂ b₀₁))
    , λ b → ∣ f ∣ γ .₁ b .₂
  sup-QW a f .resp γ₀₁ =
    qw-sup (A .trans* _ _ _ (A .sym* _ _ (A .coh γ₀₁ (∣ a ∣ _))) (a .resp γ₀₁))
      (λ {b₀} {b₁} (_ , b₀₁) →
        f .resp γ₀₁ (g γ₀₁ b₀) b₁
          (B .trans* _ _ _ (B .sym* _ _ (B .coh (Γ .sym γ₀₁ , A .sym* _ _ (A .coh γ₀₁ (∣ a ∣ _))) b₀)) b₀₁))

module _ {Γ : Con ℓΓ} (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero) (ε : EqSys A B) where
  open RelAt A B ε

  uncoe-QEq : ∀ {γ} {u v : ∣ QW A B ε ∣ γ}
    → QW A B ε ∋ Γ .ref γ ⊢ u ~ v
    → QW₀~ γ (u .₁) (v .₁)
  uncoe-QEq {γ} {u} uv = qw-trans
    (ext-coe A B ε (Γ .ref γ) (u .₁) (u .₂))
    (QW-coe-id A B ε (u .₁) (u .₂)) uv

  coe-QEq : ∀ {γ} {u v : ∣ QW A B ε ∣ γ}
    → QW₀~ γ (u .₁) (v .₁)
    → QW A B ε ∋ Γ .ref γ ⊢ u ~ v
  coe-QEq {u = u} uv = qw-trans (u .₂) (QW-refl A B ε u) uv

module _ {Γ : Con ℓΓ} (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero) (ε : EqSys A B) where
  private
    sup-step : (X : Ty Γ lzero) → Term-rec-op-step A B X (QW A B ε)
    sup-step X =
      sup-QW {Γ = Γ ▷ A ▷ (B ⇒ QW A B ε [ wk ])}
        (A [ wk∘wk ]) (B [ wk∘wk ⁺ ]) (ε [ wk∘wk ]ε) (vs vz) vz
        [ wk ↑t (B ⇒ QW A B ε [ wk ]) ]t

  eval-QW : {X : Ty Γ lzero} → Tm Γ (Term A B X)
    → Tm (Γ ▷ X) (QW A B ε [ wk ]) → Tm Γ (QW A B ε)
  eval-QW {X} t ρ = rec-Termₑ A B X (QW A B ε) ρ (sup-step X) t
  {-# INJECTIVE_FOR_INFERENCE eval-QW #-}
  {-# NOINLINE eval-QW #-}

module _ {Γ : Con ℓΓ} (A : Ty Γ lzero) (B : Ty (Γ ▷ A) lzero) (ε : EqSys A B) where
  open RelAt A B ε

  QW-eq-env : Ty (Γ ▷ E ε) lzero
  QW-eq-env = V ε ⇒ QW A B ε [ wk ]

  private
    Γeq : Con _
    Γeq = Γ ▷ E ε ▷ QW-eq-env

    eval-eq : Tm (Γ ▷ E ε) (Term (A [ wk ]) (B [ wk ⁺ ]) (V ε))
      → Tm Γeq (QW A B ε [ wk∘wk ])
    eval-eq t = eval-QW (A [ wk∘wk ]) (B [ wk∘wk ⁺ ]) (ε [ wk∘wk ]ε)
      {X = V ε [ wk ]} (t [ wk ]t) (app vz)

    module EvalAt (γ : ∣ Γ ∣) (e : ∣ E ε ∣ γ) (ρ : ∣ QW-eq-env ∣ (γ , e)) =
      Evaluation γ (A .refl*)
        (QW A B ε ∋ Γ .ref γ ⊢_~_)
        (λ {u} {v} → uncoe-QEq A B ε {u = u} {v = v})
        (λ {u} {v} → coe-QEq A B ε {u = u} {v = v})
        (V ε ∋ (Γ .ref γ , E ε .refl* e) ⊢_~_) (ρ .₁)
        (λ {x} {y} xy → uncoe-QEq A B ε {u = ρ .₁ x} {v = ρ .₁ y} (ρ .₂ xy))

  qw-eq : Tm Γeq (⇑ Eq (QW A B ε) wk∘wk (eval-eq (lhs ε)) (eval-eq (rhs ε)))
  ∣ qw-eq ∣ ((γ , e) , ρ) = ↑
    (coe-QEq A B ε
      {u = ∣ eval-eq (lhs ε) ∣ ((γ , e) , ρ)}
      {v = ∣ eval-eq (rhs ε) ∣ ((γ , e) , ρ)}
      (EvalAt.rec-eq γ e ρ (l .₁) (r .₁) (l .₂) (r .₂)
        (qw-equ e (ρ .₁)
          (λ {x} {y} xy → uncoe-QEq A B ε {u = ρ .₁ x} {v = ρ .₁ y} (ρ .₂ xy)))))
    where
      l = ∣ lhs ε ∣ (γ , e)
      r = ∣ rhs ε ∣ (γ , e)
  qw-eq .resp _ = tt

  qw-eq-at : (e : Tm Γ (E ε))
    (ρ : Tm (Γ ▷ V ε ⟨ e ⟩) (QW A B ε [ wk ]))
    → Tm Γ (⇑ Eq (QW A B ε) idS
      (eval-QW A B ε (lhs ε ⟨ e ⟩t) ρ)
      (eval-QW A B ε (rhs ε ⟨ e ⟩t) ρ))
  qw-eq-at e ρ = qw-eq [ idS ▸ e ▸ ƛ ρ ]t
