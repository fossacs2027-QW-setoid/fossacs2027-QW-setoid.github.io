fst-Eq
  : {A : Ty Γ ℓA} (B : Ty (Γ ▷ A) ℓB)
  → {x y : Tm Γ A} {p : Tm Γ (B ⟨ x ⟩)} {q : Tm Γ (B ⟨ y ⟩)}
  → (e : Tm Γ (⇑ Eq (Sigma A B) idS (dpair B x p) (dpair B y q)))
  → Tm Γ (⇑ Eq A idS x y)
fst-Eq B {x} {y} {p} {q} e =
  congT (fst B)
        {σ₀ = ⟨ dpair B x p ⟩s} {σ₁ = ⟨ dpair B y q ⟩s}
        (sgC e)

-- Heterogeneous equality allows for expressing displayed equality directly as in PathP
snd-Eq
  : {A : Ty Γ ℓA} (B : Ty (Γ ▷ A) ℓB)
  → {x y : Tm Γ A} {p : Tm Γ (B ⟨ x ⟩)} {q : Tm Γ (B ⟨ y ⟩)}
  → (e : Tm Γ (⇑ Eq (Sigma A B) idS (dpair B x p) (dpair B y q)))
  → Tm Γ (⇑ EqT B ⟨ x ⟩s ⟨ y ⟩s
              (sgC (fst-Eq B e))
              p q)
  -- ^~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ This is displayed equality over `fst`.
snd-Eq B {x} {y} {p} {q} e =
  congT (snd B)
        {σ₀ = ⟨ dpair B x p ⟩s} {σ₁ = ⟨ dpair B y q ⟩s}
        (sgC e)
