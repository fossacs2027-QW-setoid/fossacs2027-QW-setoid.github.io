-- The decoded identity code.  Its carrier is the ordinary indexed family
-- FId, rather than an inductive-recursive or inductive-inductive object.
Id
  : (A : Tm Γ U)
  → Tm Γ (El A) → Tm Γ (El A)
  → Ty Γ lzero
Id A a b = El (idᵁ A a b)

Id[]
  : {A : Tm Γ U} {a b : Tm Γ (El A)}
  → Id A a b [ σ ] ≡ Id (A [ σ ]t) (a [ σ ]t) (b [ σ ]t)
Id[] = refl

reflId : (A : Tm Γ U) (a : Tm Γ (El A)) → Tm Γ (Id A a a)
∣ reflId A a ∣ _ = frefl
reflId A a .resp _ = tt

reflId[]
  : {A : Tm Γ U} {a : Tm Γ (El A)}
  → reflId A a [ σ ]t ≡ reflId (A [ σ ]t) (a [ σ ]t)
reflId[] = refl

-- The context in which a large identity motive is formed: an endpoint and
-- an identity from the fixed left endpoint to that endpoint.
IdCtx
  : (A : Tm Γ U) (a : Tm Γ (El A))
  → Ty (Γ ▷ El A) lzero
IdCtx A a = Id (A [ wk ]t) (a [ wk ]t) vz

IdSub
  : (A : Tm Γ U) (a b : Tm Γ (El A))
  → Tm Γ (Id A a b)
  → Sub Γ (Γ ▷ El A ▷ IdCtx A a)
IdSub A a b p = idS ▸ b ▸ p

private
  FId-elim
    : ∀ {A : Set} {R : A → A → Prop} {a : A}
      (P : (b : A) → FId A R a b → Set ℓP)
    → P a frefl
    → (∀ {b} (q : R a b) → P b (ford q))
    → ∀ {b} (p : FId A R a b) → P b p
  FId-elim P d f frefl    = d
  FId-elim P d f (ford q) = f q

  J-aux
    : (A : Tm Γ U) (a : Tm Γ (El A))
      (P : Ty (Γ ▷ El A ▷ IdCtx A a) ℓP)
      (d : Tm Γ (P [ IdSub A a a (reflId A a) ]))
      (γ : ∣ Γ ∣) (b : El₀ (∣ A ∣ γ))
      (p : FId (El₀ (∣ A ∣ γ))
        (eq₀ (∣ A ∣ γ) (∣ A ∣ γ)) (∣ a ∣ γ) b)
    → ∣ P ∣ ((γ , b) , p)
  J-aux {Γ = Γ} A a P d γ b p = FId-elim
    (λ b p → ∣ P ∣ ((γ , b) , p))
    (∣ d ∣ γ)
    (λ q → P .coe ((Γ .ref γ , q) , tt) (∣ d ∣ γ)) p

  J-aux-resp
    : (A : Tm Γ U) (a : Tm Γ (El A))
      (P : Ty (Γ ▷ El A ▷ IdCtx A a) ℓP)
      (d : Tm Γ (P [ IdSub A a a (reflId A a) ]))
      {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      {b₀ : El₀ (∣ A ∣ γ₀)} {b₁ : El₀ (∣ A ∣ γ₁)}
      (b₀₁ : eq₀ (∣ A ∣ γ₀) (∣ A ∣ γ₁) b₀ b₁)
      (p₀ : FId (El₀ (∣ A ∣ γ₀))
        (eq₀ (∣ A ∣ γ₀) (∣ A ∣ γ₀)) (∣ a ∣ γ₀) b₀)
      (p₁ : FId (El₀ (∣ A ∣ γ₁))
        (eq₀ (∣ A ∣ γ₁) (∣ A ∣ γ₁)) (∣ a ∣ γ₁) b₁)
    → P ∋ ((γ₀₁ , b₀₁) , tt)
        ⊢ J-aux A a P d γ₀ b₀ p₀ ~ J-aux A a P d γ₁ b₁ p₁
  J-aux-resp A a P d γ₀₁ b₀₁ frefl frefl = d .resp γ₀₁
  J-aux-resp {Γ = Γ} A a P d {γ₀} {γ₁} γ₀₁ b₀₁ frefl (ford q₁) =
    P .trans* _ _ _ (d .resp γ₀₁)
      (P .coh ((Γ .ref γ₁ , q₁) , tt) (∣ d ∣ γ₁))
  J-aux-resp {Γ = Γ} A a P d {γ₀} {γ₁} γ₀₁ b₀₁ (ford q₀) frefl =
    P .trans* _ _ _
      (P .sym* _ _ (P .coh ((Γ .ref γ₀ , q₀) , tt) (∣ d ∣ γ₀)))
      (d .resp γ₀₁)
  J-aux-resp {Γ = Γ} A a P d {γ₀} {γ₁} γ₀₁ b₀₁
    (ford q₀) (ford q₁) =
    P .trans* _ _ _
      (P .sym* _ _ (P .coh ((Γ .ref γ₀ , q₀) , tt) (∣ d ∣ γ₀)))
      (P .trans* _ _ _ (d .resp γ₀₁)
        (P .coh ((Γ .ref γ₁ , q₁) , tt) (∣ d ∣ γ₁)))

-- Large J: the motive may live in any Set level.  The frefl branch is
-- judgmentally the supplied reflexivity case.
J
  : (A : Tm Γ U) (a : Tm Γ (El A))
    (P : Ty (Γ ▷ El A ▷ IdCtx A a) ℓP)
    (d : Tm Γ (P [ IdSub A a a (reflId A a) ]))
    (b : Tm Γ (El A)) (p : Tm Γ (Id A a b))
  → Tm Γ (P [ IdSub A a b p ])
∣ J A a P d b p ∣ γ = J-aux A a P d γ (∣ b ∣ γ) (∣ p ∣ γ)
J A a P d b p .resp γ₀₁ =
  J-aux-resp A a P d γ₀₁ (b .resp γ₀₁) (∣ p ∣ _) (∣ p ∣ _)

Jβ
  : (A : Tm Γ U) (a : Tm Γ (El A))
    (P : Ty (Γ ▷ El A ▷ IdCtx A a) ℓP)
    (d : Tm Γ (P [ IdSub A a a (reflId A a) ]))
  → J A a P d a (reflId A a) ≡ d
Jβ A a P d = refl

J[]
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ)
    (A : Tm Γ U) (a : Tm Γ (El A))
    (P : Ty (Γ ▷ El A ▷ IdCtx A a) ℓP)
    (d : Tm Γ (P [ IdSub A a a (reflId A a) ]))
    (b : Tm Γ (El A)) (p : Tm Γ (Id A a b))
  → J A a P d b p [ σ ]t
  ≡ J (A [ σ ]t) (a [ σ ]t) (P [ (σ ⁺) ⁺ ])
      (d [ σ ]t) (b [ σ ]t) (p [ σ ]t)
J[] σ A a P d b p = refl

-- Conversions to and from the model's pre-existing observational equality.
Id→Eq
  : (A : Tm Γ U) (a b : Tm Γ (El A))
  → Tm Γ (Id A a b) → Tm Γ (⇑ Eq (El A) idS a b)
∣ Id→Eq A a b p ∣ γ =
  ↑ (FId→R (refl₀ (∣ A ∣ γ)) (∣ p ∣ γ))
Id→Eq A a b p .resp _ = tt

Eq→Id
  : (A : Tm Γ U) (a b : Tm Γ (El A))
  → Tm Γ (⇑ Eq (El A) idS a b) → Tm Γ (Id A a b)
∣ Eq→Id A a b p ∣ γ = ford (∣ p ∣ γ .↓)
Eq→Id A a b p .resp _ = tt

Id→Eq[]
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} {A : Tm Γ U}
    {a b : Tm Γ (El A)} {p : Tm Γ (Id A a b)}
    (σ : Sub Δ Γ) (δ : ∣ Δ ∣)
  → eq₀ (∣ A ∣ (∣ σ ∣ δ)) (∣ A ∣ (∣ σ ∣ δ))
      (∣ a ∣ (∣ σ ∣ δ)) (∣ b ∣ (∣ σ ∣ δ))
Id→Eq[] {A = A} {a} {b} {p} σ δ =
  ∣ Id→Eq A a b p [ σ ]t ∣ δ .↓

Eq→Id[]
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} {A : Tm Γ U}
    {a b : Tm Γ (El A)} {p : Tm Γ (⇑ Eq (El A) idS a b)}
    (σ : Sub Δ Γ) (δ : ∣ Δ ∣)
  → ∣ Eq→Id A a b p [ σ ]t ∣ δ
  ≡ ford (∣ p ∣ (∣ σ ∣ δ) .↓)
Eq→Id[] σ δ = refl
