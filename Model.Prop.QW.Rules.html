QW[]
  : ∀ {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
    {ε : EqSys A B} {σ : Sub Δ Γ}
  → QW A B ε [ σ ] ≡ QW (A [ σ ]) (B [ σ ⁺ ]) (ε [ σ ]ε)
QW[] = refl

sup-QW[]
  : ∀ {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
    {ε : EqSys A B} {σ : Sub Δ Γ}
  → (a : Tm Γ A) (f : Tm Γ (Π (B ⟨ a ⟩) (QW A B ε [ wk ])))
  → sup-QW A B ε a f [ σ ]t
  ≡ sup-QW (A [ σ ]) (B [ σ ⁺ ]) (ε [ σ ]ε) (a [ σ ]t) (f [ σ ]t)
sup-QW[] a f = refl

module _ {Γ : Con ℓΓ} {Δ : Con ℓΔ} {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
  {ε : EqSys A B} {σ : Sub Δ Γ} where

  qw-eq[] : qw-eq A B ε [ σ ⁺ ⁺ ]t
    ≡ qw-eq (A [ σ ]) (B [ σ ⁺ ]) (ε [ σ ]ε)
  qw-eq[] = refl

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
  {ε : EqSys A B} {X : Ty Γ lzero} (ρ : Tm (Γ ▷ X) (QW A B ε [ wk ])) where

  eval-QW-var : (x : Tm Γ X)
    → eval-QW A B ε (var A B X x) ρ ≡ ρ ⟨ x ⟩t
  eval-QW-var x = refl

  eval-QW-op : (a : Tm Γ A) (f : Tm Γ ((B ⟨ a ⟩) ⇒ Term A B X))
    → eval-QW A B ε (op A B X a f) ρ
      ≡ sup-QW A B ε a (ƛ: (B ⟨ a ⟩)
        (eval-QW (A [ wk ]) (B [ wk ⁺ ]) (ε [ wk ]ε) (app f) (ρ [ wk ⁺ ]t)))
  eval-QW-op a f = refl

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
  {ε : EqSys A B} {σ : Sub Δ Γ} where

  elim-QW[]
    : (P : Ty (Γ ▷ QW A B ε) ℓ)
    → (p : QW-step {Γ = Γ} {A = A} B ε P)
    → (equ-p : QW-elim-coh {Γ = Γ} {A = A} B ε P p)
    → elim-QW {Γ = Γ} {A = A} B ε P p equ-p [ σ ⁺ ]t
    ≡ elim-QW {Γ = Δ} {A = A [ σ ]}
        (B [ σ ⁺ ]) (ε [ σ ]ε)
        (P [ σ ⁺ ]) (p [ σ ⁺ ⁺ ⁺ ]t)
        (equ-p [ σ ]t)
  elim-QW[] P p equ-p = refl

  elim-sup-QW
    : {P : Ty (Γ ▷ QW A B ε) ℓ}
    → {p : QW-step {Γ = Γ} {A = A} B ε P}
    → {equ-p : QW-elim-coh {Γ = Γ} {A = A} B ε P p}
    → {a : Tm Γ A}
    → {f : Tm Γ ((B ⟨ a ⟩) ⇒ QW A B ε)}
    → elim-QW {Γ = Γ} {A = A} B ε P p equ-p ⟨ sup-QW A B ε a f ⟩t
    ≡ p [ idS ▸ a ▸ f ▸
      ƛ: (B ⟨ a ⟩) (appₑ (P [ wk ⁺ ])
        (elim-QWᵢ {Γ = Γ} {A = A} B ε P p equ-p [ wk ]t) (app f)) ]t
  elim-sup-QW = refl

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
  {ε : EqSys A B} (C : Ty Γ ℓ) (p : QW-rec-step B ε C)
  (coh : QW-rec-coh B ε C p) where

  rec-QW-elim : rec-QW B ε C p coh ≡ elim-QW B ε (C [ wk ]) p coh
  rec-QW-elim = refl

  rec-QWᵢ-elim : rec-QWᵢ B ε C p coh ≡ elim-QWᵢ B ε (C [ wk ]) p coh
  rec-QWᵢ-elim = refl

  rec-QW[] : {σ : Sub Δ Γ}
    → rec-QW B ε C p coh [ σ ⁺ ]t
    ≡ rec-QW (B [ σ ⁺ ]) (ε [ σ ]ε) (C [ σ ])
        (p [ σ ⁺ ⁺ ⁺ ]t) (coh [ σ ]t)
  rec-QW[] = refl

  rec-sup-QW : (a : Tm Γ A) (f : Tm Γ ((B ⟨ a ⟩) ⇒ QW A B ε))
    → rec-QW B ε C p coh ⟨ sup-QW A B ε a f ⟩t
    ≡ p [ idS ▸ a ▸ f ▸
      ƛ: (B ⟨ a ⟩) {B = C [ wk ]}
        (appₑ' {A = QW A B ε [ wk ]} {B = C [ wk ]}
          (rec-QWᵢ B ε C p coh [ wk ]t) (app f)) ]t
  rec-sup-QW a f = refl
