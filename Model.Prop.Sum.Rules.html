open import Prelude

module Model.Prop.Sum.Rules where

open import Model.Prop.CwF
open import Model.Prop.Sum.Core
open import Model.Prop.Sum.Derived
open import Model.Prop.Sum.Elim

Sum[] : {A : Ty Γ ℓA} {B : Ty Γ ℓB} {σ : Sub Δ Γ}
    → Sum A B [ σ ] ≡ Sum (A [ σ ]) (B [ σ ])
Sum[] = refl

Inl[] : {Γ : Con ℓΓ} {A : Ty Γ ℓA} {B : Ty Γ ℓB} {σ : Sub Δ Γ}
       → (x : Tm Γ A)
       → Inl A B x [ σ ]t ≡ Inl (A [ σ ]) (B [ σ ]) (x [ σ ]t)
Inl[] x = refl

Inr[] : {Γ : Con ℓΓ} {A : Ty Γ ℓA} {B : Ty Γ ℓB} {σ : Sub Δ Γ}
       → (y : Tm Γ B)
       → Inr A B y [ σ ]t ≡ Inr (A [ σ ]) (B [ σ ]) (y [ σ ]t)
Inr[] y = refl

module _ {Γ : Con ℓΓ} {A : Ty Γ ℓA} {B : Ty Γ ℓB}
    (P : Ty (Γ ▷ Sum A B) ℓ)
    (pinl : Tm (Γ ▷ A) (P [ wk ▸ Inlⁱ A B ]))
    (pinr : Tm (Γ ▷ B) (P [ wk ▸ Inrⁱ A B ])) where

  elim-Inl
    : (x : Tm Γ A) (y : Tm Γ B)
    → case P pinl pinr ⟨ Inl A B x ⟩t ≡ pinl ⟨ x ⟩t
  elim-Inl x y = refl

  elim-Inr
    : (x : Tm Γ A) (y : Tm Γ B)
    → case P pinl pinr ⟨ Inr A B y ⟩t ≡ pinr ⟨ y ⟩t
  elim-Inr x y = refl

  elim-Sum[]
    : {σ : Sub Δ Γ} {x : Tm Γ (Sum A B)}
    → case P pinl pinr ⟨ x ⟩t [ σ ]t
      ≡ (case (P [ σ ⁺ ]) (pinl [ σ ⁺ ]t) (pinr [ σ ⁺ ]t) ⟨ x [ σ ]t ⟩t)
  elim-Sum[] = refl
