zeroᵁ : Tm Γ (El natᵁ)
∣ zeroᵁ ∣ _ = zero
zeroᵁ .resp _ = tt

El-nat-carrier
  : {Γ : Con ℓΓ} (γ : ∣ Γ ∣)
  → ∣ El (natᵁ {Γ = Γ}) ∣ γ ≡ ℕ
El-nat-carrier γ = refl

El-nat-coe
  : {Γ : Con ℓΓ} {γ₀ γ₁ : ∣ Γ ∣}
    (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (n : ℕ)
  → El (natᵁ {Γ = Γ}) .coe γ₀₁ n ≡ n
El-nat-coe γ₀₁ n = refl

El-nat-coherence
  : {Γ : Con ℓΓ} {γ₀ γ₁ : ∣ Γ ∣}
    (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (n : ℕ)
  → N._~_ n (El (natᵁ {Γ = Γ}) .coe γ₀₁ n)
El-nat-coherence {Γ = Γ} γ₀₁ n =
  El (natᵁ {Γ = Γ}) .coh γ₀₁ n

-- Constant Π/Σ codes.
nat→natᵁ : Tm Γ U
nat→natᵁ = piᵁ natᵁ natᵁ

nat×natᵁ : Tm Γ U
nat×natᵁ = sigmaᵁ natᵁ natᵁ

-- A genuinely varying family, x ↦ Id Nat x x, and its Π/Σ closures.
diagIdᵁ : Tm (Γ ▷ El natᵁ) U
diagIdᵁ = idᵁ natᵁ vz vz

dependentΠᵁ : Tm Γ U
dependentΠᵁ = piᵁ natᵁ diagIdᵁ

dependentΣᵁ : Tm Γ U
dependentΣᵁ = sigmaᵁ natᵁ diagIdᵁ

-- A nested code exercises recursive closure below a Π codomain.
nestedᵁ : Tm Γ U
nestedᵁ = piᵁ natᵁ (sigmaᵁ natᵁ diagIdᵁ)

-- Identity introduction and both directions of the correspondence with the
-- model's observational equality.
zeroId : Tm Γ (Id natᵁ zeroᵁ zeroᵁ)
zeroId = reflId natᵁ zeroᵁ

zeroId-carrier
  : {Γ : Con ℓΓ} (γ : ∣ Γ ∣)
  → ∣ Id (natᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}) ∣ γ
  ≡ FId ℕ N._~_ zero zero
zeroId-carrier γ = refl

zeroEq : Tm Γ (⇑ Eq (El natᵁ) idS zeroᵁ zeroᵁ)
zeroEq = Id→Eq natᵁ zeroᵁ zeroᵁ zeroId

zeroId′ : Tm Γ (Id natᵁ zeroᵁ zeroᵁ)
zeroId′ = Eq→Id natᵁ zeroᵁ zeroᵁ zeroEq

nat-decoding-strict
  : {Γ : Con ℓΓ} → El (natᵁ {Γ = Γ}) ≡ Nat
nat-decoding-strict = El-nat≡Nat

sigma-decoding-strict
  : {Γ : Con ℓΓ}
  → El (nat×natᵁ {Γ = Γ})
  ≡ Sigma (El (natᵁ {Γ = Γ}))
      (El (natᵁ {Γ = Γ ▷ El (natᵁ {Γ = Γ})}))
sigma-decoding-strict {Γ = Γ} =
  El-sigma≡Sigma (natᵁ {Γ = Γ})
    (natᵁ {Γ = Γ ▷ El (natᵁ {Γ = Γ})})

pi-decoding-strict
  : {Γ : Con ℓΓ}
  → El (nat→natᵁ {Γ = Γ})
  ≡ Π (El (natᵁ {Γ = Γ}))
      (El (natᵁ {Γ = Γ ▷ El (natᵁ {Γ = Γ})}))
pi-decoding-strict {Γ = Γ} =
  El-pi≡Π (natᵁ {Γ = Γ})
    (natᵁ {Γ = Γ ▷ El (natᵁ {Γ = Γ})})

identity-decoding-strict
  : {Γ : Con ℓΓ}
  → El (idᵁ (natᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}))
  ≡ Id (natᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ})
identity-decoding-strict {Γ = Γ} =
  El-id≡Id (natᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ})

-- Concrete checks for structural equivalence and both coercion directions.
nat-code-refl : eqU₀ nat₀ nat₀
nat-code-refl = reflU₀ nat₀

nat-code-sym : eqU₀ nat₀ nat₀
nat-code-sym = symU₀ nat₀ nat₀ nat-code-refl

nat-code-trans : eqU₀ nat₀ nat₀
nat-code-trans =
  transU₀ {A = nat₀} {B = nat₀} {C = nat₀} nat-code-refl nat-code-sym

nat-cast : cast₀ nat₀ nat₀ nat-code-refl 3 ≡ 3
nat-cast = refl

nat-cast-back : cast₀-back nat₀ nat₀ nat-code-refl 3 ≡ 3
nat-cast-back = refl

nat-cast-coherence
  : eq₀ nat₀ nat₀ 3 (cast₀ nat₀ nat₀ nat-code-refl 3)
nat-cast-coherence = cast₀-eq nat₀ nat₀ nat-code-refl 3

-- Substitution equations for every constructor are judgmental.
nat-substitution
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ)
  → natᵁ {Γ = Γ} [ σ ]t ≡ natᵁ {Γ = Δ}
nat-substitution σ = refl

pi-substitution
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ)
  → nat→natᵁ {Γ = Γ} [ σ ]t ≡ nat→natᵁ {Γ = Δ}
pi-substitution σ = refl

sigma-substitution
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ)
  → nat×natᵁ {Γ = Γ} [ σ ]t ≡ nat×natᵁ {Γ = Δ}
sigma-substitution σ = refl

id-substitution
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ)
  → idᵁ (natᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}) [ σ ]t
  ≡ idᵁ (natᵁ {Γ = Δ}) (zeroᵁ {Γ = Δ}) (zeroᵁ {Γ = Δ})
id-substitution σ = refl

-- A small concrete use of large J.  J itself is universe-polymorphic in the
-- motive; here a constant Nat motive makes the judgmental beta rule visible.
J-zero : {Γ : Con ℓΓ} → Tm Γ Nat
J-zero {Γ = Γ} =
  J (natᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}) Nat (N.z {Γ = Γ})
    (zeroᵁ {Γ = Γ}) (zeroId {Γ = Γ})

J-zero-β : {Γ : Con ℓΓ} → J-zero {Γ = Γ} ≡ N.z {Γ = Γ}
J-zero-β {Γ = Γ} =
  Jβ (natᵁ {Γ = Γ}) (zeroᵁ {Γ = Γ}) Nat (N.z {Γ = Γ})

-- Smoke tests for the two induction principles derived from Ext.
all-codes : (A : U₀) → ⊤
all-codes = U₀-rec (λ _ → ⊤) tt
  (λ _ _ _ _ _ → tt)
  (λ _ _ _ _ _ → tt)
  (λ _ _ _ _ → tt)

equal-codes : ∀ (A B : U₀) (e : eqU₀ A B) → ⊤
equal-codes = U₀-rec₂ (λ _ _ _ → ⊤) tt
  (λ _ _ _ _ _ _ _ _ _ _ → tt)
  (λ _ _ _ _ _ _ _ _ _ _ → tt)
  (λ _ _ _ _ _ _ _ _ _ _ → tt)
