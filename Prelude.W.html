module Prelude.W where

open import Prelude.Base

module _ (A : Set) (B : A → Set) where
  data 𝒲 : Set where
    sup₀ : (x : A) (f : B x → 𝒲) → 𝒲

  module _
    (P : 𝒲 → Set ℓ)
    (supᵖ : {x : A} {f : B x → 𝒲} → ((y : B x) → P (f y)) → P (sup₀ x f))
    where

    rec : (t : 𝒲) → P t
    rec (sup₀ x f) = supᵖ (λ (y : B x) → rec (f y))
  
map
  : {A₀ A₁ : Set} (h : A₀ → A₁)
  → {B₀ : A₀ → Set} {B₁ : A₁ → Set}
  → (g : ∀ {x} → B₁ (h x) → B₀ x)
  → 𝒲 A₀ B₀ → 𝒲 A₁ B₁
map h g (sup₀ x f) = sup₀ (h x) λ y' →
  map h g (f (g y'))
