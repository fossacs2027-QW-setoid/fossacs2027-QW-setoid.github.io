Term[]
  : ∀ {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero} {X : Ty Γ lzero} {σ : Sub Δ Γ}
  → Term A B X [ σ ] ≡ Term (A [ σ ]) (B [ σ ⁺ ]) (X [ σ ])
Term[] = refl

var[]
  : ∀ {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero} {X : Ty Γ lzero} {σ : Sub Δ Γ} (x : Tm Γ X)
  → var A B X x [ σ ]t ≡ var (A [ σ ]) (B [ σ ⁺ ]) (X [ σ ]) (x [ σ ]t)
var[] x = refl

op[]
  : ∀ {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero} {X : Ty Γ lzero} {σ : Sub Δ Γ}
  → (a : Tm Γ A) (f : Tm Γ (Π (B ⟨ a ⟩) (Term A B X [ wk ])))
  → op A B X a f [ σ ]t
  ≡ op (A [ σ ]) (B [ σ ⁺ ]) (X [ σ ]) (a [ σ ]t) (f [ σ ]t)
op[] a f = refl

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
  {X : Ty Γ lzero} (P : Ty (Γ ▷ Term A B X) ℓ)
  (v : Var-step A B X P) (p : Op-step A B X P) where

  elim-Termₑ[] : {σ : Sub Δ Γ} (t : Tm Γ (Term A B X))
    → elim-Termₑ A B X P v p t [ σ ]t
    ≡ elim-Termₑ (A [ σ ]) (B [ σ ⁺ ]) (X [ σ ]) (P [ σ ⁺ ])
        (v [ σ ⁺ ]t) (p [ σ ⁺ ⁺ ⁺ ]t) (t [ σ ]t)
  elim-Termₑ[] t = refl

  elim-var : (x : Tm Γ X)
    → elim-Termₑ A B X P v p (var A B X x) ≡ v ⟨ x ⟩t
  elim-var x = refl

  elim-op : (a : Tm Γ A) (f : Tm Γ ((B ⟨ a ⟩) ⇒ Term A B X))
    → elim-Termₑ A B X P v p (op A B X a f)
      ≡ p [ idS ▸ a ▸ f ▸
          ƛ: (B ⟨ a ⟩) (elim-Termₑ (A [ wk ]) (B [ wk ⁺ ]) (X [ wk ])
            (P [ wk ⁺ ]) (v [ wk ⁺ ]t) (p [ wk ⁺ ⁺ ⁺ ]t) (app f)) ]t
  elim-op a f = refl

module _ {Γ : Con ℓΓ} {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero}
  {X : Ty Γ lzero} (C : Ty Γ ℓ)
  (v : Term-rec-var-step A B X C) (p : Term-rec-op-step A B X C) where

  rec-Term-elim : rec-Term A B X C v p ≡ elim-Term A B X (C [ wk ]) v p
  rec-Term-elim = refl

  rec-Termₑ-elim : (t : Tm Γ (Term A B X))
    → rec-Termₑ A B X C v p t ≡ elim-Termₑ A B X (C [ wk ]) v p t
  rec-Termₑ-elim t = refl

  rec-Term[] : {σ : Sub Δ Γ}
    → rec-Term A B X C v p [ σ ⁺ ]t
    ≡ rec-Term (A [ σ ]) (B [ σ ⁺ ]) (X [ σ ]) (C [ σ ])
        (v [ σ ⁺ ]t) (p [ σ ⁺ ⁺ ⁺ ]t)
  rec-Term[] = refl

  rec-Termₑ[] : {σ : Sub Δ Γ} (t : Tm Γ (Term A B X))
    → rec-Termₑ A B X C v p t [ σ ]t
    ≡ rec-Termₑ (A [ σ ]) (B [ σ ⁺ ]) (X [ σ ]) (C [ σ ])
        (v [ σ ⁺ ]t) (p [ σ ⁺ ⁺ ⁺ ]t) (t [ σ ]t)
  rec-Termₑ[] t = refl

  rec-Term-var : (x : Tm Γ X)
    → rec-Termₑ A B X C v p (var A B X x) ≡ v ⟨ x ⟩t
  rec-Term-var x = refl

  rec-Term-op : (a : Tm Γ A) (f : Tm Γ ((B ⟨ a ⟩) ⇒ Term A B X))
    → rec-Termₑ A B X C v p (op A B X a f)
    ≡ p [ idS ▸ a ▸ f ▸
      ƛ: (B ⟨ a ⟩) (rec-Termₑ (A [ wk ]) (B [ wk ⁺ ]) (X [ wk ]) (C [ wk ])
        (v [ wk ⁺ ]t) (p [ wk ⁺ ⁺ ⁺ ]t) (app f)) ]t
  rec-Term-op a f = refl
