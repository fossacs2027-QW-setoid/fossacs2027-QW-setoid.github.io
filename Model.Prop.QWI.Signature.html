open import Prelude

module Model.Prop.QWI.Signature where

open import Model.Prop.CwF hiding (A; B)
open import Model.Prop.QWI.Polynomial public
open import Model.Prop.QWI.Free.Core

record EqSys {Γ : Con ℓΓ} (S : Poly Γ) : Set (lsuc ℓΓ) where
  field
    E : Ty (Γ ▷ I S) lzero
    V : Ty (Γ ▷ I S ▷ E ▷ I S [ wk∘wk ]) lzero
    lhs rhs : Tm (Γ ▷ I S ▷ E)
      (Term (S [ wk∘wk ]s) V [ idS ▸ vs vz ])

open EqSys public

_[_]ε : {S : Poly Γ} → EqSys S → (σ : Sub Δ Γ) → EqSys (S [ σ ]s)
E (ε [ σ ]ε) = E ε [ σ ⁺ ]
V (ε [ σ ]ε) = V ε [ σ ⁺ ⁺ ⁺ ]
lhs (ε [ σ ]ε) = lhs ε [ σ ⁺ ⁺ ]t
rhs (ε [ σ ]ε) = rhs ε [ σ ⁺ ⁺ ]t
