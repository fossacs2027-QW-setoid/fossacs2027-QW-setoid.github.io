open import Prelude

module Model.Prop.QWI.Polynomial where

open import Model.Prop.CwF hiding (A; B)
open import Model.Prop.Pi

record Poly (Γ : Con ℓΓ) : Set (lsuc ℓΓ) where
  field
    I  : Ty Γ lzero
    A  : Ty (Γ ▷ I) lzero
    B  : Ty (Γ ▷ I ▷ A) lzero
    ar : Tm (Γ ▷ I ▷ A ▷ B) (I [ wk∘wk∘wk ])

  next : Sub (Γ ▷ I ▷ A ▷ B) (Γ ▷ I)
  next = wk∘wk∘wk ▸ ar

  Branch : Ty (Γ ▷ I) ℓ → Ty (Γ ▷ I ▷ A) ℓ
  Branch Y = Π B (Y [ next ])

open Poly public

_[_]s : Poly Γ → (σ : Sub Δ Γ) → Poly Δ
I  (S [ σ ]s) = I S [ σ ]
A  (S [ σ ]s) = A S [ σ ⁺ ]
B  (S [ σ ]s) = B S [ σ ⁺ ⁺ ]
ar (S [ σ ]s) = ar S [ σ ⁺ ⁺ ⁺ ]t

module At {Γ : Con ℓΓ} (S : Poly Γ) where
  I₀ : ∣ Γ ∣ → Set
  I₀ γ = ∣ I S ∣ γ

  A₀ : ∀ {γ} → I₀ γ → Set
  A₀ {γ} i = ∣ A S ∣ (γ , i)

  B₀ : ∀ {γ i} → A₀ {γ} i → Set
  B₀ {γ} {i} a = ∣ B S ∣ ((γ , i) , a)

  ar₀ : ∀ {γ i a} → B₀ {γ} {i} a → I₀ γ
  ar₀ {γ} {i} {a} b = ∣ ar S ∣ (((γ , i) , a) , b)

  A~ : ∀ {γ δ} → Γ ∋ γ ~ δ → ∀ {i j} → A₀ {γ} i → A₀ {δ} j → Prop
  A~ q {i} {j} a a' = ∃[ ij ∈ (I S ∋ q ⊢ i ~ j) ] (A S ∋ (q , ij) ⊢ a ~ a')

  B~ : ∀ {γ δ} → Γ ∋ γ ~ δ → ∀ {i j a a'} → B₀ {γ} {i} a → B₀ {δ} {j} a' → Prop
  B~ q {a = a} {a'} b b' = ∃[ aa ∈ A~ q a a' ] (B S ∋ ((q , aa .₁) , aa .₂) ⊢ b ~ b')

  pull : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) (ij : I S ∋ q ⊢ i ~ j) {a : A₀ i}
    → B₀ (A S .coe (q , ij) a) → B₀ a
  pull q ij {a} = B S .coe ((Γ .sym q , I S .sym* _ _ ij) , A S .sym* _ _ (A S .coh (q , ij) a))

  pull~ : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) (ij : I S ∋ q ⊢ i ~ j) {a : A₀ i}
    (b : B₀ (A S .coe (q , ij) a)) → B~ q (pull q ij b) b
  pull~ q ij {a} b = (ij , A S .coh (q , ij) a) , B S .sym* _ _
    (B S .coh ((Γ .sym q , I S .sym* _ _ ij) , A S .sym* _ _ (A S .coh (q , ij) a)) b)

  ar~ : ∀ {γ δ} {q : Γ ∋ γ ~ δ} {i j a a'} {b : B₀ {γ} {i} a} {b' : B₀ {δ} {j} a'}
    → B~ q b b' → I S ∋ q ⊢ ar₀ b ~ ar₀ b'
  ar~ ((ij , aa) , bb) = ar S .resp (((_ , ij) , aa) , bb)

  A-refl : ∀ {γ i} (a : A₀ {γ} i) → A~ (Γ .ref γ) a a
  A-refl a = I S .refl* _ , A S .refl* a

  A-sym : ∀ {γ δ} {q : Γ ∋ γ ~ δ} {i j} {a : A₀ {γ} i} {a' : A₀ {δ} j}
    → A~ q a a' → A~ (Γ .sym q) a' a
  A-sym (ij , aa) = I S .sym* _ _ ij , A S .sym* _ _ aa

  A-trans : ∀ {γ δ θ} {q : Γ ∋ γ ~ δ} {r : Γ ∋ δ ~ θ} {i j k}
    {a : A₀ {γ} i} {a' : A₀ {δ} j} {a'' : A₀ {θ} k}
    → A~ q a a' → A~ r a' a'' → A~ (Γ .trans q r) a a''
  A-trans (ij , aa) (jk , aa') = I S .trans* _ _ _ ij jk , A S .trans* _ _ _ aa aa'

  B-sym : ∀ {γ δ} {q : Γ ∋ γ ~ δ} {i j a a'} {b : B₀ {γ} {i} a} {b' : B₀ {δ} {j} a'}
    → B~ q b b' → B~ (Γ .sym q) b' b
  B-sym (aa , bb) = A-sym aa , B S .sym* _ _ bb

  B-trans : ∀ {γ δ θ} {q : Γ ∋ γ ~ δ} {r : Γ ∋ δ ~ θ} {i j k a a' a''}
    {b : B₀ {γ} {i} a} {b' : B₀ {δ} {j} a'} {b'' : B₀ {θ} {k} a''}
    → B~ q b b' → B~ r b' b'' → B~ (Γ .trans q r) b b''
  B-trans (aa , bb) (aa' , bb') = A-trans aa aa' , B S .trans* _ _ _ bb bb'

  B-coe : ∀ {γ δ} {q : Γ ∋ γ ~ δ} {i j} {a : A₀ {γ} i} {a' : A₀ {δ} j}
    → A~ q a a' → B₀ a → B₀ a'
  B-coe {q = q} (ij , aa) = B S .coe ((q , ij) , aa)

  B-coh : ∀ {γ δ} {q : Γ ∋ γ ~ δ} {i j} {a : A₀ {γ} i} {a' : A₀ {δ} j}
    (aa : A~ q a a') (b : B₀ a) → B~ q b (B-coe aa b)
  B-coh {q = q} (ij , aa) b = (ij , aa) , B S .coh ((q , ij) , aa) b
