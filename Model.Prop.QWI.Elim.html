module QWISteps {Γ : Con ℓΓ} (S : Poly Γ) (ε : EqSys S)
  (P : Ty (Γ ▷ I S ▷ QWI S ε) ℓ) where

  QWI-node : Con ℓΓ
  QWI-node = Γ ▷ I S ▷ A S ▷ Branch S (QWI S ε)

  QWI-child : Sub (QWI-node ▷ B S [ wk ]) (Γ ▷ I S ▷ QWI S ε)
  ∣ QWI-child ∣ ((((γ , i) , a) , f) , b) = (γ , ∣ ar S ∣ (((γ , i) , a) , b)) , f .₁ b
  QWI-child .resp ((((q , ij) , aa) , ff) , bb) = (q , ar S .resp (((q , ij) , aa) , bb)) , ff _ _ bb

  QWI-head : Sub QWI-node (Γ ▷ I S ▷ QWI S ε)
  QWI-head = wk∘wk ▸ sup-QWI S ε

  QWI-IH : Ty QWI-node ℓ
  QWI-IH = Π (B S [ wk ]) (P [ QWI-child ])

  QWI-step : Set _
  QWI-step = Tm (QWI-node ▷ QWI-IH) (P [ QWI-head ∘ wk ])

open QWISteps public

module FiberConstruction {Γ : Con ℓΓ} (S : Poly Γ) (ε : EqSys S)
  (P : Ty (Γ ▷ I S ▷ QWI S ε) ℓ) (p : QWI-step S ε P) where

  open At S
  open QuotientConstruction S ε hiding (QWI; sup-QWI; eval-QWI; eval-QWIₑ)
  open RelAt

  P-at : (γ : ∣ Γ ∣) → ∀ {i} (w : QWI₀ γ i) → Ext γ w → Set ℓ
  P-at γ {i} w extw = ∣ P ∣ ((γ , i) , (w , extw))

  P~-at : (γ : ∣ Γ ∣) → ∀ {i j} {w : QWI₀ γ i} {v : QWI₀ γ j}
    → QWI₀~ γ w v → {extw : Ext γ w} {extv : Ext γ v}
    → P-at γ w extw → P-at γ v extv → Prop ℓ
  P~-at γ eq {extw} {extv} = P .heq
    ((Γ .ref γ , qwi-index eq) , coe-QEq {u = _ , extw} {v = _ , extv} eq)

  private
    module StepAt (γ : ∣ Γ ∣) = StepTypes (A~ (Γ .ref γ)) (B~ (Γ .ref γ)) (SigAt γ) (P-at γ) (P~-at γ)

  alg-at : (γ : ∣ Γ ∣) → StepAt.Step γ
  alg-at γ {a = a} {f} extf f~ ih ih~ = ∣ p ∣
    ((((γ , _) , a) , (λ b → f b , extf b)
      , λ {b} {b'} bb → coe-QEq {u = _ , extf b} {v = _ , extf b'} (f~ (A-refl a , bb)))
      , ih , λ bb → ih~ (A-refl a , bb))

  alg-at~ : (γ : ∣ Γ ∣) → StepAt.Step~ γ (alg-at γ)
  alg-at~ γ f~ extf ih ih~ f'~ extf' ih' ih'~ aa ff ii = p .resp
    ((((Γ .ref γ , aa .₁) , aa .₂)
      , λ b b' bb → coe-QEq {u = _ , extf b} {v = _ , extf' b'} (ff (aa , bb)))
      , λ b b' bb → ii (aa , bb))

  module Fiber (γ : ∣ Γ ∣) = QWI-elim₀
    (A~ (Γ .ref γ)) A-refl A-sym (B~ (Γ .ref γ)) B-sym B-trans B-coe B-coh
    (SigAt γ) (λ e x → I S .refl* _ , V ε .refl* x)
    (λ {i} e → ∣ lhs ε ∣ ((γ , i) , e) .₂)
    (λ {i} e → ∣ rhs ε ∣ ((γ , i) , e) .₂)
    (P-at γ) (P~-at γ)
    (λ w extw v → P .refl* v)
    (λ {v₀ = x} {v₁ = y} → P .sym* x y)
    (λ {v₀ = x} {v₁ = y} {v₂ = z} → P .trans* x y z)
    (alg-at γ) (alg-at~ γ)

module DependentEvaluation {Γ : Con ℓΓ} (S : Poly Γ) (ε : EqSys S)
  (P : Ty (Γ ▷ I S ▷ QWI S ε) ℓ) (p : QWI-step S ε P)
  (X : Ty (Γ ▷ I S) lzero)
  (ρ : Tm (Γ ▷ I S ▷ X) (Sigma (QWI S ε) P [ wk ])) where

  open At S
  open QuotientConstruction S ε hiding (QWI; sup-QWI; eval-QWI; eval-QWIₑ)
  open RelAt
  module F = Free.Construction S X
  open FiberConstruction S ε P p
  import Model.Prop.QWI.Core as Core

  assignment-base : Tm (Γ ▷ I S ▷ X) (QWI S ε [ wk ])
  assignment-base = fstₑ (QWI S ε [ wk ]) (P [ wk ⁺ ]) ρ

  evaluated : Tm (Γ ▷ I S ▷ Term S X) (QWI S ε [ wk ])
  evaluated = Core.eval-QWI S ε assignment-base

  deval-type : Ty (Γ ▷ I S ▷ Term S X) ℓ
  deval-type = P [ wk ▸ evaluated ]

  deval₀ : ∀ {γ i} (t : F.Raw γ i) (extt : F.Ext t)
    → ∣ deval-type ∣ ((γ , i) , (t , extt))
  deval₀ {γ} = Fiber.deval γ (F.X~ (Γ .ref γ))
    (λ {i} x → ∣ ρ ∣ ((γ , i) , x) .₁)
    (λ {i} {j} {x} {y} xy → uncoe-QEq
      {u = ∣ ρ ∣ ((γ , i) , x) .₁} {v = ∣ ρ ∣ ((γ , j) , y) .₁}
      (ρ .resp ((Γ .ref γ , xy .₁) , xy .₂) .₁))
    (λ {i} x → ∣ ρ ∣ ((γ , i) , x) .₂)
    (λ xy → ρ .resp ((Γ .ref γ , xy .₁) , xy .₂) .₂)

  deval-resp : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) (t : F.Raw γ i) (u : F.Raw δ j)
    (extt : F.Ext t) (extu : F.Ext u) (tu : F.Rel q t u)
    → deval-type ∋ ((q , F.rel-index {t = t} {u} tu) , tu) ⊢ deval₀ t extt ~ deval₀ u extu
  deval-resp q (var₀ x) (var₀ y) _ _ (ij , xy) = ρ .resp ((q , ij) , xy) .₂
  deval-resp {γ} {δ} q (op₀ a f) (op₀ a' f') (f~ , extf) (f'~ , extf') (aa , ff) =
    p .resp ((((q , aa .₁) , aa .₂)
      , λ b b' bb → evaluated .resp
          {((γ , ar₀ b) , (f b , extf b))} {((δ , ar₀ b') , (f' b' , extf' b'))}
          ((q , ar~ (aa , bb)) , ff (aa , bb)))
      , λ b b' bb → deval-resp q (f b) (f' b') (extf b) (extf' b') (ff (aa , bb)))

  deval : Tm (Γ ▷ I S ▷ Term S X) deval-type
  ∣ deval ∣ ((γ , i) , (t , extt)) = deval₀ t extt
  deval .resp {((_ , _) , (t , extt))} {((_ , _) , (u , extu))} ((q , ij) , tu) =
    deval-resp q t u extt extu tu

open DependentEvaluation public using (deval; assignment-base)

module InternalCoherence {Γ : Con ℓΓ} (S : Poly Γ) (ε : EqSys S)
  (P : Ty (Γ ▷ I S ▷ QWI S ε) ℓ) (p : QWI-step S ε P) where

  open QuotientConstruction S ε hiding (QWI; sup-QWI; eval-QWI; eval-QWIₑ)
  open FiberConstruction S ε P p

  QWI-elim-env : Ty (Γ ▷ I S ▷ E ε) ℓ
  QWI-elim-env = Π (I S [ wk∘wk ]) (Π (V ε) (Sigma (QWI S ε) P [ variable-sort ∘ wk ]))

  Γeq : Con (ℓΓ ⊔ ℓ)
  Γeq = Γ ▷ I S ▷ E ε ▷ QWI-elim-env

  base : Sub Γeq Γ
  base = wk∘wk∘wk

  env-base : Tm Γeq (QWI-eq-env [ wk ])
  ∣ env-base ∣ (γie , env) = (λ j → (λ x → env .₁ j .₁ x .₁) , λ xy → env .₁ j .₂ xy .₁)
    , λ jk x y xy → env .₂ jk x y xy .₁
  env-base .resp (q , ee) j k jk x y xy = ee j k jk x y xy .₁

  eq-base : Sub Γeq EqContext
  eq-base = wk ▸ env-base

  private
    X : Ty (Γeq ▷ I S [ base ]) lzero
    X = V ε [ wk ⁺ ]

    ρ : Tm (Γeq ▷ I S [ base ] ▷ X) (Sigma (QWI S ε) P [ base ⁺ ∘ wk ])
    ∣ ρ ∣ (((γie , env) , j) , x) = env .₁ j .₁ x
    ρ .resp (((q , ee) , jk) , xy) = ee _ _ jk _ _ xy

    module D = DependentEvaluation (S [ base ]s) (ε [ base ]ε)
      (P [ base ⁺ ⁺ ]) (p [ base ⁺ ⁺ ⁺ ⁺ ]t) X ρ

  deval-eq : (t : Tm (Γ ▷ I S ▷ E ε) (Term (S [ wk∘wk ]s) (V ε) [ idS ▸ vs vz ]))
    → Tm Γeq (P [ (equation-sort ∘ eq-base) ▸ (eval-eq t [ eq-base ]t) ])
  deval-eq t = D.deval [ (idS ▸ vs (vs vz)) ▸ (t [ wk ]t) ]t

  QWI-elim-eq : Prp Γeq ℓ
  ∣ QWI-elim-eq ∣ (((γ , i) , e) , env) = Fiber.QWI-elim₀-eq γ e
    (λ {j} x → env .₁ j .₁ x .₁)
    (λ {j} {k} {x} {y} (jk , xy) → uncoe-QEq
      {u = env .₁ j .₁ x .₁} {v = env .₁ k .₁ y .₁} (env .₂ jk x y xy .₁))
    (λ {j} x → env .₁ j .₁ x .₂)
    (λ {j} {k} {x} {y} (jk , xy) → env .₂ jk x y xy .₂)
  QWI-elim-eq .coe {γ₀} {γ₁} q eq = P .trans* _ _ _
    (P .sym* _ _ (deval-eq (lhs ε) .resp {γ₀} {γ₁} q))
    (P .trans* _ _ _ eq (deval-eq (rhs ε) .resp {γ₀} {γ₁} q))

  QWI-elim-coh : Set (ℓΓ ⊔ ℓ)
  QWI-elim-coh = Tm (Γ ▷ I S) (Π (E ε) (Π QWI-elim-env (⇑ QWI-elim-eq)))

open InternalCoherence public

QWI-elim-coh₀ : {Γ : Con ℓΓ} (S : Poly Γ) (ε : EqSys S)
  (P : Ty (Γ ▷ I S ▷ QWI S ε) ℓ) → QWI-step S ε P → Prop (ℓΓ ⊔ ℓ)
QWI-elim-coh₀ {Γ = Γ} S ε P p = (γ : ∣ Γ ∣) → FiberConstruction.Fiber.QWI-elim₀-coh S ε P p γ

module _ {Γ : Con ℓΓ} (S : Poly Γ) (ε : EqSys S)
  (P : Ty (Γ ▷ I S ▷ QWI S ε) ℓ) (p : QWI-step S ε P) where

  private
    module Q = QuotientConstruction S ε

  coherence-at : QWI-elim-coh S ε P p → QWI-elim-coh₀ S ε P p
  coherence-at coh γ {i} e ρ ρ~ ρP ρP~ = LiftS.↓ (∣ coh ∣ (γ , i) .₁ e .₁ env)
    where
      env : ∣ QWI-elim-env S ε P p ∣ ((γ , i) , e)
      env = (λ j → (λ x → ρ x , ρP x)
        , λ {x} {y} xy → Q.coe-QEq {u = ρ x} {v = ρ y} (ρ~ (I S .refl* j , xy))
          , ρP~ (I S .refl* j , xy))
        , λ {j} {k} jk x y xy → Q.coe-QEq {u = ρ x} {v = ρ y} (ρ~ (jk , xy)) , ρP~ (jk , xy)

  coherence-term : QWI-elim-coh₀ S ε P p → QWI-elim-coh S ε P p
  ∣ coherence-term coh ∣ (γ , i) = (λ e → (λ env → ↑ (coh γ e
    (λ {j} x → env .₁ j .₁ x .₁)
    (λ {j} {k} {x} {y} (jk , xy) → Q.uncoe-QEq
      {u = env .₁ j .₁ x .₁} {v = env .₁ k .₁ y .₁} (env .₂ jk x y xy .₁))
    (λ {j} x → env .₁ j .₁ x .₂)
    (λ {j} {k} {x} {y} (jk , xy) → env .₂ jk x y xy .₂))) , λ _ → tt)
    , λ _ _ _ _ → tt
  coherence-term coh .resp _ _ _ _ _ _ _ = tt

module _ {Γ : Con ℓΓ} (S : Poly Γ) (ε : EqSys S)
  (P : Ty (Γ ▷ I S ▷ QWI S ε) ℓ) (p : QWI-step S ε P)
  (equ-p : QWI-elim-coh S ε P p) where

  open At S
  open QuotientConstruction S ε hiding (QWI; sup-QWI; eval-QWI; eval-QWIₑ)
  open RelAt
  open FiberConstruction S ε P p

  elim-at : (γ : ∣ Γ ∣) → ∀ {i} (w : QWI₀ γ i) (extw : Ext γ w) → P-at γ w extw
  elim-at γ = Fiber.elim-QWI₀ γ (coherence-at S ε P p equ-p γ)

  elim-at~ : (γ : ∣ Γ ∣) → ∀ {i j} {w : QWI₀ γ i} {v : QWI₀ γ j}
    (eq : QWI₀~ γ w v) (extw : Ext γ w) (extv : Ext γ v)
    → P~-at γ eq (elim-at γ w extw) (elim-at γ v extv)
  elim-at~ γ = Fiber.elim-QWI₀~ γ (coherence-at S ε P p equ-p γ)

  elim-at-coe : ∀ {γ δ i j} (q : Γ ∋ γ ~ δ) (ij : I S ∋ q ⊢ i ~ j)
    (w : QWI₀ γ i) (extw : Ext γ w)
    → P ∋ ((q , ij) , QWI S ε .coh (q , ij) (w , extw))
        ⊢ elim-at γ w extw ~ elim-at δ (QWI-coe q ij w) (ext-coe q ij w extw)
  elim-at-coe {γ} {δ} q ij (supᵢ₀ a f) (f~ , extf) = p .resp
    ((((q , ij) , A S .coh (q , ij) a)
      , λ b b' bb → QWI-coe~ q (ar~ ((ij , A S .coh (q , ij) a) , bb))
          (ar~ (pull~ q ij b')) (f~ (branch-eq bb)))
      , λ b b' bb → P .trans* _ _ _
          (elim-at-coe q (ar~ ((ij , A S .coh (q , ij) a) , bb)) (f b) (extf b))
          (elim-at~ δ (QWI-coe~ q (ar~ ((ij , A S .coh (q , ij) a) , bb))
            (ar~ (pull~ q ij b')) (f~ (branch-eq bb)))
            (ext-coe q _ (f b) (extf b)) (ext-coe q _ (f (pull q ij b')) (extf (pull q ij b')))))
    where
      branch-eq : ∀ {b b'} → B S ∋ ((q , ij) , A S .coh (q , ij) a) ⊢ b ~ b'
        → B~ (Γ .ref γ) b (pull q ij b')
      branch-eq bb = B-trans ((ij , A S .coh (q , ij) a) , bb) (B-sym (pull~ q ij _))

  elim-QWI : Tm (Γ ▷ I S ▷ QWI S ε) P
  ∣ elim-QWI ∣ ((γ , i) , (w , extw)) = elim-at γ w extw
  elim-QWI .resp {((γ , i) , (w , extw))} {((δ , j) , (v , extv))} ((q , ij) , eq) =
    P .trans* _ _ _ (elim-at-coe q ij w extw) (elim-at~ δ eq (ext-coe q ij w extw) extv)

  elim-QWIᵢ : Tm (Γ ▷ I S) (Π (QWI S ε) P)
  elim-QWIᵢ = ƛ elim-QWI
