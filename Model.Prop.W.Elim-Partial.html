open import Prelude

module Model.Prop.W.Elim-Partial where

open import Model.Prop.CwF
open import Model.Prop.Pi
open import Model.Prop.W.Core

open ℒ using (extent; value)
open ℒ↓ using (val~)

-- As for QW-types, compute a partial value whose domain supplies coherent
-- recursive values. For W-types, preservation of equality can be proved
-- before totality, without any equivalence laws for the relations.
module _
  {A : Set} (_~A_ : A → A → Prop)
  {B : A → Set} (_~B_ : ∀ {x₀ x₁} → B x₀ → B x₁ → Prop)
  where
  open W-rel _~A_ B B _~B_

  module W-elim₀
    (P : (w : 𝒲 A B) → extW _~A_ _~B_ w → Set ℓ)
    (P~ : ∀ {w₀ w₁} → 𝒲~ w₀ w₁
      → {extw₀ : extW _~A_ _~B_ w₀} {extw₁ : extW _~A_ _~B_ w₁}
      → P w₀ extw₀ → P w₁ extw₁ → Prop ℓ)
    (p : {x : A} {f : B x → 𝒲 A B}
      → (extf : (y : B x) → extW _~A_ _~B_ (f y))
      → (f~ : ∀ {y₀ y₁} → y₀ ~B y₁ → 𝒲~ (f y₀) (f y₁))
      → (ih : (y : B x) → P (f y) (extf y))
      → (ih~ : ∀ {y₀ y₁} (y₀₁ : y₀ ~B y₁)
        → P~ (f~ y₀₁) (ih y₀) (ih y₁))
      → P (sup₀ x f) (f~ , extf))
    (p~ : {x₀ : A} {f₀ : B x₀ → 𝒲 A B}
      → (f₀~ : ∀ {y₀ y₀'} → y₀ ~B y₀' → 𝒲~ (f₀ y₀) (f₀ y₀'))
      → (extf₀ : (y₀ : B x₀) → extW _~A_ _~B_ (f₀ y₀))
      → (ih₀ : (y₀ : B x₀) → P (f₀ y₀) (extf₀ y₀))
      → (ih₀~ : ∀ {y₀ y₀'} (y₀₀' : y₀ ~B y₀')
        → P~ (f₀~ y₀₀') (ih₀ y₀) (ih₀ y₀'))
      → {x₁ : A} {f₁ : B x₁ → 𝒲 A B}
      → (f₁~ : ∀ {y₁ y₁'} → y₁ ~B y₁' → 𝒲~ (f₁ y₁) (f₁ y₁'))
      → (extf₁ : (y₁ : B x₁) → extW _~A_ _~B_ (f₁ y₁))
      → (ih₁ : (y₁ : B x₁) → P (f₁ y₁) (extf₁ y₁))
      → (ih₁~ : ∀ {y₁ y₁'} (y₁₁' : y₁ ~B y₁')
        → P~ (f₁~ y₁₁') (ih₁ y₁) (ih₁ y₁'))
      → (x₀₁ : x₀ ~A x₁)
      → (f₀₁ : ∀ {y₀ y₁} → y₀ ~B y₁ → 𝒲~ (f₀ y₀) (f₁ y₁))
      → (ih₀₁ : ∀ {y₀ y₁} (y₀₁ : y₀ ~B y₁)
        → P~ (f₀₁ y₀₁) (ih₀ y₀) (ih₁ y₁))
      → P~ (x₀₁ , f₀₁)
          (p extf₀ f₀~ ih₀ ih₀~) (p extf₁ f₁~ ih₁ ih₁~))
    where

    elim-par : (w : 𝒲 A B) (extw : extW _~A_ _~B_ w) → ℒ (P w extw)
    elim-par (sup₀ x f) (f~ , extf) =
      (∃[ ds ∈ ((y : B x) → elim-par (f y) (extf y) .extent) ]
        (∀ {y₀ y₁} (y₀₁ : y₀ ~B y₁) → P~ (f~ y₀₁)
          (elim-par (f y₀) (extf y₀) .value (ds y₀))
          (elim-par (f y₁) (extf y₁) .value (ds y₁))))
      , λ { (ds , coh) →
          p extf f~ (partial-values (λ y → elim-par (f y) (extf y)) ds) coh }

    -- Defined values respect tree equality, independently of totality.
    elim-par~ : (w₀ w₁ : 𝒲 A B)
      → (extw₀ : extW _~A_ _~B_ w₀) (extw₁ : extW _~A_ _~B_ w₁)
      → (w₀₁ : 𝒲~ w₀ w₁)
      → (d₀ : elim-par w₀ extw₀ .extent) (d₁ : elim-par w₁ extw₁ .extent)
      → P~ w₀₁ (elim-par w₀ extw₀ .value d₀) (elim-par w₁ extw₁ .value d₁)
    elim-par~ (sup₀ x₀ f₀) (sup₀ x₁ f₁) (f₀~ , extf₀) (f₁~ , extf₁)
      (x₀₁ , f₀₁) (ds₀ , coh₀) (ds₁ , coh₁) =
      p~ f₀~ extf₀ (partial-values (λ y → elim-par (f₀ y) (extf₀ y)) ds₀) coh₀
         f₁~ extf₁ (partial-values (λ y → elim-par (f₁ y) (extf₁ y)) ds₁) coh₁
         x₀₁ f₀₁ (λ {y₀} {y₁} y₀₁ →
           elim-par~ (f₀ y₀) (f₁ y₁) (extf₀ y₀) (extf₁ y₁)
             (f₀₁ y₀₁) (ds₀ y₀) (ds₁ y₁))

    -- All domain and coherence proofs live in Prop; no values are chosen.
    totality : (w : 𝒲 A B) (extw : extW _~A_ _~B_ w) → elim-par w extw .extent
    totality (sup₀ x f) (f~ , extf) = ds , λ {y₀} {y₁} y₀₁ →
      elim-par~ (f y₀) (f y₁) (extf y₀) (extf y₁) (f~ y₀₁) (ds y₀) (ds y₁)
      where
        ds : (y : B x) → elim-par (f y) (extf y) .extent
        ds y = totality (f y) (extf y)

    sound : (w₀ w₁ : 𝒲 A B)
      → (extw₀ : extW _~A_ _~B_ w₀) (extw₁ : extW _~A_ _~B_ w₁)
      → (w₀₁ : 𝒲~ w₀ w₁)
      → ℒ↓ (P~ w₀₁) (elim-par w₀ extw₀) (elim-par w₁ extw₁)
    sound w₀ w₁ extw₀ extw₁ w₀₁ =
      related (totality w₀ extw₀) (totality w₁ extw₁)
        (elim-par~ w₀ w₁ extw₀ extw₁ w₀₁ (totality w₀ extw₀) (totality w₁ extw₁))

    elim-W₀ : (w : 𝒲 A B) (extw : extW _~A_ _~B_ w) → P w extw
    elim-W₀ w extw = elim-par w extw .value (totality w extw)

    elim-W₀~ : ∀ {w₀ w₁} (w₀₁ : 𝒲~ w₀ w₁)
      → (extw₀ : extW _~A_ _~B_ w₀) (extw₁ : extW _~A_ _~B_ w₁)
      → P~ w₀₁ (elim-W₀ w₀ extw₀) (elim-W₀ w₁ extw₁)
    elim-W₀~ w₀₁ extw₀ extw₁ = sound _ _ extw₀ extw₁ w₀₁ .val~

open W-elim₀ public

-- Interpret the partial construction fibrewise, with the same interface as
-- Model.Prop.W.Elim. Its computation is independent of the ambient context.
module _ {Γ : Con ℓΓ} {A : Ty Γ lzero} (B : Ty (Γ ▷ A) lzero)
         (P : Ty (Γ ▷ W A B) ℓ)
         (p : let Γ₀    = Γ ▷ A ▷ B ⇒ W A B [ wk ]
                  f[vz] = app {Γ = Γ₀ ▷ B [ wk ]} {A = B [ wk ∘ wk ]} {B = W A B [ wk ∘ wk ∘ wk ∘ wk ]} (vs vz) ⟨ vz ⟩t
              in
              Tm (Γ₀ ▷ Π (B [ wk ]) (P [ wk ∘ wk ∘ wk ▸ f[vz] ]))
                 (P [ wk ∘ wk ∘ wk
                    ▸ (sup-W {Γ = Γ₀} (A [ wk ∘ wk ]) (B [ (wk ∘ wk) ⁺ ]) (vs vz) vz) [ wk ]t
                    ]))
  where
  open W-rel

  private
    elim-at
      : (γ : ∣ Γ ∣) (w : 𝒲 (∣ A ∣ γ) (λ x → ∣ B ∣ (γ , x)))
      → (extw : extW (A .heq (Γ .ref γ)) (B~ A B (Γ .ref γ)) w)
      → ∣ P ∣ (γ , (w , extw))
    elim-at γ = elim-W₀
      (A .heq (Γ .ref γ)) (B~ A B (Γ .ref γ))
      (λ w ext → ∣ P ∣ (γ , (w , ext)))
      (λ w₀₁ → P .heq (Γ .ref γ , w₀₁))
      (λ {x} {f} extf f~ ih ih~ →
        ∣ p ∣ (((γ , x)
               , ((λ y → f y , extf y)
                 , λ {y₀} {y₁} y₀₁ → f~ (A .refl* x , y₀₁)))
               , (ih , λ {y₀} {y₁} y₀₁ → ih~ (A .refl* x , y₀₁))))
      (λ {x₀} {f₀} f₀~ extf₀ ih₀ ih₀~ {x₁} {f₁} f₁~ extf₁ ih₁ ih₁~ x₀₁ f₀₁ ih₀₁ →
        p .resp (((Γ .ref γ , x₀₁)
                 , λ y₀ y₁ y₀₁ → f₀₁ (x₀₁ , y₀₁))
                 , λ y₀ y₁ y₀₁ → ih₀₁ (x₀₁ , y₀₁)))

    elim-at~
      : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      → ∀ {w₀} (extw₀ : extW (A .heq (Γ .ref γ₀)) (B~ A B (Γ .ref γ₀)) w₀)
          {w₁} (extw₁ : extW (A .heq (Γ .ref γ₁)) (B~ A B (Γ .ref γ₁)) w₁)
      → (w₀₁ : 𝒲~ (A .heq γ₀₁)
                    (λ x₀ → ∣ B ∣ (γ₀ , x₀)) (λ x₁ → ∣ B ∣ (γ₁ , x₁))
                    (B~ A B γ₀₁) w₀ w₁)
      → P ∋ (γ₀₁ , w₀₁) ⊢ elim-at γ₀ w₀ extw₀ ~ elim-at γ₁ w₁ extw₁
    elim-at~ γ₀₁ {sup₀ x₀ f₀} (f₀~ , extf₀) {sup₀ x₁ f₁} (f₁~ , extf₁) (x₀₁ , f₀₁) =
      p .resp (((γ₀₁ , x₀₁)
              , λ y₀ y₁ y₀₁ → f₀₁ (x₀₁ , y₀₁))
              , λ y₀ y₁ y₀₁ →
                  elim-at~ γ₀₁ (extf₀ y₀) (extf₁ y₁) (f₀₁ (x₀₁ , y₀₁)))

  elim-W : Tm (Γ ▷ W A B) P
  ∣ elim-W ∣ (γ , (w , extw)) = elim-at γ w extw
  elim-W .resp {γ₀ , (w₀ , extw₀)} {γ₁ , (w₁ , extw₁)} (γ₀₁ , w₀₁) =
    elim-at~ γ₀₁ extw₀ extw₁ w₀₁

  elim-Wₑ : (t : Tm Γ (W A B)) → Tm Γ (P ⟨ t ⟩)
  elim-Wₑ t = elim-W [ idS ▸ t ]t

-- Proof irrelevance makes substitution and constructor computation strict.
module _ {Γ : Con ℓΓ} {A : Ty Γ lzero} {B : Ty (Γ ▷ A) lzero} {σ : Sub Δ Γ}
  where

  elim-W[]
    : (P : Ty (Γ ▷ W A B) ℓ)
    → (p : _)
    → elim-W B P p [ σ ⁺ ]t ≡
      elim-W (B [ σ ⁺ ]) (P [ σ ⁺ ]) (p [ σ ⁺ ⁺ ⁺ ]t)
  elim-W[] P p = refl

  elim-sup-W
    : {P : Ty (Γ ▷ W A B) ℓ}
    → {p : _}
    → {x : Tm Γ A} {f : Tm Γ ((B ⟨ x ⟩) ⇒ W A B)}
    → elim-W B P p ⟨ sup-W A B x f ⟩t
    ≡ p [ idS ▸ x ▸ f ▸
      (ƛ (elim-W B P p [ wk {A = A} ∘ wk {A = Π B (W A B [ wk ∘ wk ])} ∘ wk {A = B [ wk ]} ▸
      (app {Γ = Γ ▷ A ▷ Π B (W A B [ wk ∘ wk ]) ▷ B [ wk ]} {B = W A B [ wk ∘ wk ∘ wk ∘ wk ]} (vs vz) ⟨ vz ⟩t) ]t)) [ idS ▸ x ▸ f ]t ]t
  elim-sup-W = refl
