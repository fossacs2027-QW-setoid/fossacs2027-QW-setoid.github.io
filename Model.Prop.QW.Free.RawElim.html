open import Prelude

module Model.Prop.QW.Free.RawElim where

open import Model.Prop.QW.Free.Core

open ℒ using (extent; value)

-- Keep the computation independent of the ambient context so that the
-- internal eliminator commutes definitionally with substitution.
module _ {A : Set} {B : A → Set} {X : Set}
  (A~ : A → A → Prop) (B~ : ∀ {a a'} → B a → B a' → Prop)
  (X~ : X → X → Prop) where

  open Term-rel A~ B B B~ X~

  private
    T : Set
    T = Σₚ (Term₀ A B X) (extTerm A~ B~ X~)

    Ext : ∀ {a} → (B a → T) → Prop
    Ext f = ∀ {b b'} → B~ b b' → Term₀~ (f b .₁) (f b' .₁)

  module Elim₀ (P : T → Set ℓ)
    (P~ : ∀ {t t'} → Term₀~ (t .₁) (t' .₁) → P t → P t' → Prop ℓ)
    (v : (x : X) → P (var₀ x , tt))
    (v~ : ∀ {x y} (xy : X~ x y) → P~ xy (v x) (v y))
    (p : (a : A) (f : B a → T) (f~ : Ext f)
      → (ih : (b : B a) → P (f b))
      → (∀ {b b'} (bb : B~ b b') → P~ (f~ bb) (ih b) (ih b'))
      → P (op₀ a (λ b → f b .₁) , f~ , λ b → f b .₂))
    (p~ : ∀ {a a'} {f : B a → T} {f' : B a' → T}
      {f~ : Ext f} {f'~ : Ext f'}
      {ih : (b : B a) → P (f b)} {ih' : (b : B a') → P (f' b)}
      → (ih~ : ∀ {b b'} (bb : B~ b b') → P~ (f~ bb) (ih b) (ih b'))
      → (ih'~ : ∀ {b b'} (bb : B~ b b') → P~ (f'~ bb) (ih' b) (ih' b'))
      → (aa : A~ a a')
      → (ff : ∀ {b b'} → B~ b b' → Term₀~ (f b .₁) (f' b' .₁))
      → (∀ {b b'} (bb : B~ b b') → P~ (ff bb) (ih b) (ih' b'))
      → P~ (aa , ff) (p a f f~ ih ih~) (p a' f' f'~ ih' ih'~)) where

    elim-par : (t : Term₀ A B X) (extt : extTerm A~ B~ X~ t) → ℒ (P (t , extt))
    elim-par (var₀ x) _ = ⊤ₚ , λ _ → v x
    elim-par (op₀ a f) (f~ , extf) =
      (∃[ ds ∈ ((b : B a) → elim-par (f b) (extf b) .extent) ]
        (∀ {b b'} (bb : B~ b b') → P~ (f~ bb)
          (elim-par (f b) (extf b) .value (ds b))
          (elim-par (f b') (extf b') .value (ds b'))))
      , λ { (ds , coh) →
          p a (λ b → f b , extf b) f~
            (partial-values (λ b → elim-par (f b) (extf b)) ds) coh }

    elim-par~ : (t t' : Term₀ A B X)
      → (extt : extTerm A~ B~ X~ t) (extt' : extTerm A~ B~ X~ t')
      → (tt' : Term₀~ t t')
      → (d : elim-par t extt .extent) (d' : elim-par t' extt' .extent)
      → P~ tt' (elim-par t extt .value d) (elim-par t' extt' .value d')
    elim-par~ (var₀ x) (var₀ y) _ _ xy _ _ = v~ xy
    elim-par~ (op₀ a f) (op₀ a' f') (f~ , extf) (f'~ , extf')
      (aa , ff) (ds , coh) (ds' , coh') =
      p~ coh coh' aa ff
        (λ {b} {b'} bb → elim-par~ (f b) (f' b') (extf b) (extf' b')
          (ff bb) (ds b) (ds' b'))

    totality : (t : Term₀ A B X) (extt : extTerm A~ B~ X~ t) → elim-par t extt .extent
    totality (var₀ x) _ = tt
    totality (op₀ a f) (f~ , extf) = ds , λ {b} {b'} bb →
      elim-par~ (f b) (f b') (extf b) (extf b') (f~ bb) (ds b) (ds b')
      where
        ds : (b : B a) → elim-par (f b) (extf b) .extent
        ds b = totality (f b) (extf b)

    val : (t : Term₀ A B X) (extt : extTerm A~ B~ X~ t) → P (t , extt)
    val t extt = elim-par t extt .value (totality t extt)

    val~ : (t t' : Term₀ A B X)
      → (extt : extTerm A~ B~ X~ t) (extt' : extTerm A~ B~ X~ t')
      → (tt' : Term₀~ t t') → P~ tt' (val t extt) (val t' extt')
    val~ t t' extt extt' tt' =
      elim-par~ t t' extt extt' tt' (totality t extt) (totality t' extt')
