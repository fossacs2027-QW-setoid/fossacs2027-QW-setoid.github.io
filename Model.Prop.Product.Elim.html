module Model.Prop.Product.Elim where

open import Model.Prop.CwF
open import Model.Prop.Product.Core

module _ {Γ : Con ℓΓ} {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB}
    (P : Ty (Γ ▷ Sigma A B) ℓC)
    (t : Tm (Γ ▷ A ▷ B) (P [ wk∘wk ▸ dpairⁱ ])) where

  elim-Sigma : Tm (Γ ▷ Sigma A B) P
  ∣ elim-Sigma ∣ (γ , (a , b)) = ∣ t ∣ ((γ , a) , b)
  elim-Sigma .resp (γ~γ' , (a~a' , b~b')) = t .resp ((γ~γ' , a~a') , b~b')
