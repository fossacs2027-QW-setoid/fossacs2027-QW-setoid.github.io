open import Prelude

module Model.Prop.W.Rec where

open import Model.Prop.CwF
open import Model.Prop.Pi
open import Model.Prop.W.Core
open import Model.Prop.W.Elim using (W-step)
import Model.Prop.W.RawElim as Raw

W-rec-step : {Γ : Con ℓΓ} {A : Ty Γ lzero} (B : Ty (Γ ▷ A) lzero)
  (C : Ty Γ ℓ) → Set _
W-rec-step B C = W-step B (C [ wk ])

module _ {Γ : Con ℓΓ}{A : Ty Γ lzero} (B : Ty (Γ ▷ A) lzero)
         (C : Ty Γ ℓ)
         (p : W-rec-step B C)
  where
  open W-rel

  private
    B~-sym-at
      : ∀ {γ} {x₀ x₁ : ∣ A ∣ γ} {y₀ : ∣ B ∣ (γ , x₀)} {y₁ : ∣ B ∣ (γ , x₁)}
      → B~ A B (Γ .ref γ) y₁ y₀ → B~ A B (Γ .ref γ) y₀ y₁
    B~-sym-at (x₁₀ , y₁₀) = A .sym* _ _ x₁₀ , B .sym* _ _ y₁₀

    B-coe-at
      : ∀ {γ} {x₀ x₁ : ∣ A ∣ γ}
      → A ∋ Γ .ref γ ⊢ x₀ ~ x₁ → ∣ B ∣ (γ , x₀) → ∣ B ∣ (γ , x₁)
    B-coe-at {γ} x₀₁ y₀ = B .coe (Γ .ref γ , x₀₁) y₀

    B-coh-at
      : ∀ {γ} {x₀ x₁ : ∣ A ∣ γ}
      → (x₀₁ : A ∋ Γ .ref γ ⊢ x₀ ~ x₁) (y₀ : ∣ B ∣ (γ , x₀))
      → B~ A B (Γ .ref γ) y₀ (B-coe-at x₀₁ y₀)
    B-coh-at {γ} x₀₁ y₀ = x₀₁ , B .coh (Γ .ref γ , x₀₁) y₀

    B-trans-at
      : ∀ {γ} {x₀ x₁ x₂ : ∣ A ∣ γ}
      → (x₀₁ : A ∋ Γ .ref γ ⊢ x₀ ~ x₁)
      → (x₁₂ : A ∋ Γ .ref γ ⊢ x₁ ~ x₂)
      → {y₀ : ∣ B ∣ (γ , x₀)} {y₂ : ∣ B ∣ (γ , x₂)}
      → B~ A B (Γ .ref γ) y₀ y₂
      → B~ A B (Γ .ref γ) (B-coe-at x₀₁ y₀) y₂
    B-trans-at {γ} x₀₁ x₁₂ {y₀} (x₀₂ , y₀₂) =
      x₁₂ , B .trans* _ _ _ (B .sym* _ _ (B .coh (Γ .ref γ , x₀₁) y₀)) y₀₂

    rec-at
      : (γ : ∣ Γ ∣) (w : 𝒲 (∣ A ∣ γ) (λ x → ∣ B ∣ (γ , x)))
      → (extw : extW (A .heq (Γ .ref γ)) (B~ A B (Γ .ref γ)) w)
      → ∣ C ∣ γ
    rec-at γ w extw = Raw.totality
      (A .heq (Γ .ref γ))
      (A .refl*) (λ {x} {y} → A .sym* x y) (λ {x} {y} {z} → A .trans* x y z)
      (B~ A B (Γ .ref γ))
      B~-sym-at B-coe-at B-coh-at B-trans-at
      (λ w ext → ∣ C ∣ γ)
      (λ w₀₁ → C .heq (Γ .ref γ))
      (λ _ _ v → C .refl* v)
      (λ {v₀ = x} {v₁ = y} → C .sym* x y) (λ {v₀ = x} {v₁ = y} {v₂ = z} → C .trans* x y z)
      (λ {x} {f} extf f~ ih ih~ →
        ∣ p ∣ (((γ , x)
               , ((λ y → f y , extf y)
                 , λ {y₀} {y₁} y₀₁ → f~ (A .refl* x , y₀₁)))
               , (ih , λ {y₀} {y₁} y₀₁ → ih~ (A .refl* x , y₀₁))))
      (λ {x₀} {f₀} f₀~ extf₀ ih₀ ih₀~ {x₁} {f₁} f₁~ extf₁ ih₁ ih₁~ x₀₁ f₀₁ ih₀₁ →
        p .resp (((Γ .ref γ , x₀₁)
                 , λ y₀ y₁ y₀₁ → f₀₁ (x₀₁ , y₀₁))
                 , λ y₀ y₁ y₀₁ → ih₀₁ (x₀₁ , y₀₁)))
      w extw .₁

    rec-at~
      : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      → ∀ {w₀} (extw₀ : extW (A .heq (Γ .ref γ₀)) (B~ A B (Γ .ref γ₀)) w₀)
          {w₁} (extw₁ : extW (A .heq (Γ .ref γ₁)) (B~ A B (Γ .ref γ₁)) w₁)
      → (w₀₁ : 𝒲~ (A .heq γ₀₁)
                    (λ x₀ → ∣ B ∣ (γ₀ , x₀)) (λ x₁ → ∣ B ∣ (γ₁ , x₁))
                    (B~ A B γ₀₁) w₀ w₁)
      → C ∋ γ₀₁ ⊢ rec-at γ₀ w₀ extw₀ ~ rec-at γ₁ w₁ extw₁
    rec-at~ γ₀₁ {sup₀ x₀ f₀} (f₀~ , extf₀) {sup₀ x₁ f₁} (f₁~ , extf₁) (x₀₁ , f₀₁) =
      p .resp (((γ₀₁ , x₀₁)
              , λ y₀ y₁ y₀₁ → f₀₁ (x₀₁ , y₀₁))
              , λ y₀ y₁ y₀₁ →
                  rec-at~ γ₀₁ (extf₀ y₀) (extf₁ y₁) (f₀₁ (x₀₁ , y₀₁)))

  rec-W : Tm (Γ ▷ W A B) (C [ wk ])
  ∣ rec-W ∣ (γ , (w , extw)) = rec-at γ w extw
  rec-W .resp {γ₀ , (w₀ , extw₀)} {γ₁ , (w₁ , extw₁)} (γ₀₁ , w₀₁) =
    rec-at~ γ₀₁ extw₀ extw₁ w₀₁

  rec-Wₑ : (t : Tm Γ (W A B)) → Tm Γ C
  rec-Wₑ t = rec-W [ idS ▸ t ]t
