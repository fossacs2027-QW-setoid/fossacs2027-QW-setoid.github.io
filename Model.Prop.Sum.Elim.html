open import Prelude

module Model.Prop.Sum.Elim where

open import Model.Prop.CwF
open import Model.Prop.Sum.Core
open import Model.Prop.Sum.Derived
 
module _ {Γ : Con ℓΓ} {A : Ty Γ ℓA} {B : Ty Γ ℓB}
    (P : Ty (Γ ▷ Sum A B) ℓ)
    (pinl : Tm (Γ ▷ A) (P [ wk ▸ Inlⁱ A B ]))
    (pinr : Tm (Γ ▷ B) (P [ wk ▸ Inrⁱ A B ])) where

  case : Tm (Γ ▷ Sum A B) P
  ∣ case ∣ (γ , z) =
   [_,_] {C = λ z → ∣ P ∣ (γ , z)}
         (λ a → ∣ pinl ∣ (γ , a))
         (λ b → ∣ pinr ∣ (γ , b))
         z
  case .resp {γ₀ , inl a₀} {γ₁ , inl a₁} (γ₀₁ , a₀₁) = pinl .resp (γ₀₁ , a₀₁ .↓)
  case .resp {γ₀ , inr b₀} {γ₁ , inr b₁} (γ₀₁ , b₀₁) = pinr .resp (γ₀₁ , b₀₁ .↓)

module _ {Γ : Con ℓΓ} {A : Ty Γ ℓA} {B : Ty Γ ℓB} (C : Ty Γ ℓ)
    (pinl : Tm (Γ ▷ A) (C [ wk ]))
    (pinr : Tm (Γ ▷ B) (C [ wk ])) where

  case′ : Tm (Γ ▷ Sum A B) (C [ wk ])
  ∣ case′ ∣ (γ , z) =
   [_,_] {C = λ z → ∣ C ∣ γ}
         (λ a → ∣ pinl ∣ (γ , a))
         (λ b → ∣ pinr ∣ (γ , b))
         z
  case′ .resp {γ₀ , inl a₀} {γ₁ , inl a₁} (γ₀₁ , a₀₁) = pinl .resp (γ₀₁ , a₀₁ .↓)
  case′ .resp {γ₀ , inr b₀} {γ₁ , inr b₁} (γ₀₁ , b₀₁) = pinr .resp (γ₀₁ , b₀₁ .↓)
