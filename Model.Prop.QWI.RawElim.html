module StepTypes {I : Set} {A : I → Set}
  (_~A_ : ∀ {i j} → A i → A j → Prop)
  {B : ∀ {i} → A i → Set} {ar : ∀ {i a} → B {i} a → I}
  (_~B_ : ∀ {i j a a'} → B {i} a → B {j} a' → Prop)
  (ε : EqSys₀ A B ar)
  (P : ∀ {i} (w : 𝒲I A B ar i) → QWI-rel.Ext _~A_ _~B_ ε w → Set ℓ)
  (P~ : ∀ {i j} {w₀ : 𝒲I A B ar i} {w₁ : 𝒲I A B ar j} → QWI-rel.QWI₀~ _~A_ _~B_ ε w₀ w₁
    → {extw₀ : QWI-rel.Ext _~A_ _~B_ ε w₀}
    → {extw₁ : QWI-rel.Ext _~A_ _~B_ ε w₁}
    → P w₀ extw₀ → P w₁ extw₁ → Prop ℓ) where

  open QWI-rel _~A_ _~B_ ε using (QWI₀~; Ext)

  Step : Set _
  Step = ∀ {i} {a : A i} {f : (b : B a) → 𝒲I A B ar (ar b)}
    → (extf : (b : B a) → Ext (f b))
    → (f~ : ∀ {b₀ b₁} → b₀ ~B b₁ → QWI₀~ (f b₀) (f b₁))
    → (ih : (b : B a) → P (f b) (extf b))
    → (ih~ : ∀ {b₀ b₁} (b₀₁ : b₀ ~B b₁)
      → P~ (f~ b₀₁) (ih b₀) (ih b₁))
    → P (supᵢ₀ a f) (f~ , extf)

  Step~ : Step → Prop _
  Step~ p = ∀ {i} {a₀ : A i} {f₀ : (b : B a₀) → 𝒲I A B ar (ar b)}
    → (f₀~ : ∀ {b₀ b₀'} → b₀ ~B b₀' → QWI₀~ (f₀ b₀) (f₀ b₀'))
    → (extf₀ : (b₀ : B a₀) → Ext (f₀ b₀))
    → (ih₀ : (b₀ : B a₀) → P (f₀ b₀) (extf₀ b₀))
    → (ih₀~ : ∀ {b₀ b₀'} (b₀₀' : b₀ ~B b₀')
      → P~ (f₀~ b₀₀') (ih₀ b₀) (ih₀ b₀'))
    → ∀ {j} {a₁ : A j} {f₁ : (b : B a₁) → 𝒲I A B ar (ar b)}
    → (f₁~ : ∀ {b₁ b₁'} → b₁ ~B b₁' → QWI₀~ (f₁ b₁) (f₁ b₁'))
    → (extf₁ : (b₁ : B a₁) → Ext (f₁ b₁))
    → (ih₁ : (b₁ : B a₁) → P (f₁ b₁) (extf₁ b₁))
    → (ih₁~ : ∀ {b₁ b₁'} (b₁₁' : b₁ ~B b₁')
      → P~ (f₁~ b₁₁') (ih₁ b₁) (ih₁ b₁'))
    → (a₀₁ : a₀ ~A a₁)
    → (f₀₁ : ∀ {b₀ b₁} → b₀ ~B b₁ → QWI₀~ (f₀ b₀) (f₁ b₁))
    → (ih₀₁ : ∀ {b₀ b₁} (b₀₁ : b₀ ~B b₁)
      → P~ (f₀₁ b₀₁) (ih₀ b₀) (ih₁ b₁))
    → P~ (qwi-sup a₀₁ f₀₁)
        (p extf₀ f₀~ ih₀ ih₀~) (p extf₁ f₁~ ih₁ ih₁~)

module _
  {I : Set} {A : I → Set} (_~A_ : ∀ {i j} → A i → A j → Prop)
  (Arefl : ∀ {i} (a : A i) → a ~A a)
  (Asym : ∀ {i j} {a₀ : A i} {a₁ : A j} → a₀ ~A a₁ → a₁ ~A a₀)
  {B : ∀ {i} → A i → Set} {ar : ∀ {i a} → B {i} a → I}
  (_~B_ : ∀ {i j a a'} → B {i} a → B {j} a' → Prop)
  (Bsym : ∀ {i j} {a₀ : A i} {a₁ : A j} {b₀ : B a₀} {b₁ : B a₁} → b₀ ~B b₁ → b₁ ~B b₀)
  (Btrans : ∀ {i j k} {a₀ : A i} {a₁ : A j} {a₂ : A k} {b₀ : B a₀} {b₁ : B a₁} {b₂ : B a₂}
    → b₀ ~B b₁ → b₁ ~B b₂ → b₀ ~B b₂)
  (Bcoe : ∀ {i j} {a₀ : A i} {a₁ : A j} → a₀ ~A a₁ → B a₀ → B a₁)
  (Bcoh : ∀ {i j} {a₀ : A i} {a₁ : A j} (aa : a₀ ~A a₁) (b : B a₀) → b ~B Bcoe aa b)
  (ε : EqSys₀ A B ar)
  (Vrefl : ∀ {i} (e : E ε i) {j} (x : V ε e j) → V~ ε e x x)
  (lhs-ext : ∀ {i} (e : E ε i) → extTerm _~A_ _~B_ (V~ ε e) (lhs ε e))
  (rhs-ext : ∀ {i} (e : E ε i) → extTerm _~A_ _~B_ (V~ ε e) (rhs ε e))
  where

  open QWI-rel _~A_ _~B_ ε
    using (eval~; QWI₀~; Ext; eval-QExt; |𝒲|; qwi-refl)

  module QWI-elim₀
    (P : ∀ {i} (w : 𝒲I A B ar i) → Ext w → Set ℓ)
    (P~ : ∀ {i j} {w₀ : 𝒲I A B ar i} {w₁ : 𝒲I A B ar j} → QWI₀~ w₀ w₁
      → {extw₀ : Ext w₀} {extw₁ : Ext w₁}
      → P w₀ extw₀ → P w₁ extw₁ → Prop ℓ)
    (P~refl : ∀ {i} (w : 𝒲I A B ar i) (extw : Ext w) (v : P w extw)
      → P~ (qwi-refl Arefl extw) v v)
    (P~sym : ∀ {i j} {w₀ : 𝒲I A B ar i} {w₁ : 𝒲I A B ar j} {extw₀ : Ext w₀} {extw₁ : Ext w₁}
      {v₀ : P w₀ extw₀} {v₁ : P w₁ extw₁} {w₀₁ : QWI₀~ w₀ w₁}
      → P~ w₀₁ v₀ v₁ → P~ (qwi-sym w₀₁) v₁ v₀)
    (P~trans : ∀ {i j k} {w₀ : 𝒲I A B ar i} {w₁ : 𝒲I A B ar j} {w₂ : 𝒲I A B ar k}
      {extw₀ : Ext w₀} {extw₁ : Ext w₁} {extw₂ : Ext w₂}
      {v₀ : P w₀ extw₀} {v₁ : P w₁ extw₁} {v₂ : P w₂ extw₂}
      {w₀₁ : QWI₀~ w₀ w₁} {w₁₂ : QWI₀~ w₁ w₂}
      → P~ w₀₁ v₀ v₁ → P~ w₁₂ v₁ v₂
      → P~ (qwi-trans extw₁ w₀₁ w₁₂) v₀ v₂)
    (p : StepTypes.Step _~A_ _~B_ ε P P~)
    (p~ : StepTypes.Step~ _~A_ _~B_ ε P P~ p)
    where

    module _ {X : I → Set} (X~ : ∀ {i j} → X i → X j → Prop)
      (ρ : ∀ {i} → X i → |𝒲| i)
      (ρ~ : ∀ {i j} {x : X i} {y : X j} → X~ x y → QWI₀~ (ρ x .₁) (ρ y .₁))
      (ρP : ∀ {i} (x : X i) → P (ρ x .₁) (ρ x .₂))
      (ρP~ : ∀ {i j} {x : X i} {y : X j} (xy : X~ x y) → P~ (ρ~ xy) (ρP x) (ρP y)) where

      private
        evaluated : ∀ {i} → Σₚ (Term₀ A B ar X i) (extTerm _~A_ _~B_ X~) → |𝒲| i
        evaluated t = eval-QExt X~ t ρ ρ~

        evaluated~ : ∀ {i j}
          {t : Σₚ (Term₀ A B ar X i) (extTerm _~A_ _~B_ X~)}
          {u : Σₚ (Term₀ A B ar X j) (extTerm _~A_ _~B_ X~)}
          → Term₀~ _~A_ _~B_ X~ (t .₁) (u .₁)
          → QWI₀~ (evaluated t .₁) (evaluated u .₁)
        evaluated~ {t = t} {u} eq = eval~ X~ {t₀ = t .₁} {t₁ = u .₁} eq ρ ρ ρ~

      open FreeRaw.Elim₀ _~A_ _~B_ X~
        (λ t → P (evaluated t .₁) (evaluated t .₂))
        (λ {t = t} {t' = t'} tt' → P~ (evaluated~ {t = t} {u = t'} tt'))
        ρP ρP~
        (λ a f f~ ih ih~ → p (λ b → evaluated (f b) .₂)
          (λ {b} {b'} bb → evaluated~ {t = f b} {u = f b'} (f~ bb)) ih ih~)
        (λ {f = f} {f' = f'} {f~} {f'~} ih~ ih'~ aa ff ii →
          p~ (λ {b} {b'} bb → evaluated~ {t = f b} {u = f b'} (f~ bb)) (λ b → evaluated (f b) .₂) _ ih~
            (λ {b} {b'} bb → evaluated~ {t = f' b} {u = f' b'} (f'~ bb)) (λ b → evaluated (f' b) .₂) _ ih'~
            aa (λ {b} {b'} bb → evaluated~ {t = f b} {u = f' b'} (ff bb)) ii)
        public using () renaming (val to deval; val~ to deval~)

    QWI-elim₀-eq : ∀ {i} (e : E ε i) (ρ : ∀ {j} → V ε e j → |𝒲| j)
      (ρ~ : ∀ {j k} {x : V ε e j} {y : V ε e k} → V~ ε e x y → QWI₀~ (ρ x .₁) (ρ y .₁))
      (ρP : ∀ {j} (x : V ε e j) → P (ρ x .₁) (ρ x .₂))
      (ρP~ : ∀ {j k} {x : V ε e j} {y : V ε e k} (xy : V~ ε e x y) → P~ (ρ~ xy) (ρP x) (ρP y))
      → Prop ℓ
    QWI-elim₀-eq e ρ ρ~ ρP ρP~ = P~ (qwi-equ e ρ ρ~)
      (deval (V~ ε e) ρ ρ~ ρP ρP~ (lhs ε e) (lhs-ext e))
      (deval (V~ ε e) ρ ρ~ ρP ρP~ (rhs ε e) (rhs-ext e))
    {-# NOINLINE QWI-elim₀-eq #-}

    QWI-elim₀-coh : Prop _
    QWI-elim₀-coh = ∀ {i} (e : E ε i) (ρ : ∀ {j} → V ε e j → |𝒲| j)
      (ρ~ : ∀ {j k} {x : V ε e j} {y : V ε e k} → V~ ε e x y → QWI₀~ (ρ x .₁) (ρ y .₁))
      (ρP : ∀ {j} (x : V ε e j) → P (ρ x .₁) (ρ x .₂))
      (ρP~ : ∀ {j k} {x : V ε e j} {y : V ε e k} (xy : V~ ε e x y) → P~ (ρ~ xy) (ρP x) (ρP y))
      → QWI-elim₀-eq e ρ ρ~ ρP ρP~

    elim-par : ∀ {i} (w : 𝒲I A B ar i) (extw : Ext w) → ℒ (P w extw)
    elim-par (supᵢ₀ a f) (f~ , extf) =
      (∃[ ds ∈ ((b : B a) → elim-par (f b) (extf b) .extent) ]
        (∀ {b₀ b₁} (bb : b₀ ~B b₁) → P~ (f~ bb)
          (elim-par (f b₀) (extf b₀) .value (ds b₀))
          (elim-par (f b₁) (extf b₁) .value (ds b₁))))
      , λ { (ds , coh) →
          p extf f~ (partial-values (λ b → elim-par (f b) (extf b)) ds) coh }

    module _ {X : I → Set} (X~ : ∀ {i j} → X i → X j → Prop)
      (ρ : ∀ {i} → X i → |𝒲| i)
      (ρ~ : ∀ {i j} {x : X i} {y : X j} → X~ x y → QWI₀~ (ρ x .₁) (ρ y .₁))
      (ds : ∀ {i} (x : X i) → elim-par (ρ x .₁) (ρ x .₂) .extent)
      (coh : ∀ {i j} {x : X i} {y : X j} (xy : X~ x y) → P~ (ρ~ xy)
        (elim-par (ρ x .₁) (ρ x .₂) .value (ds x))
        (elim-par (ρ y .₁) (ρ y .₂) .value (ds y))) where

      private
        ρP = λ {i} (x : X i) → elim-par (ρ x .₁) (ρ x .₂) .value (ds x)
        ev = deval X~ ρ ρ~ ρP coh
        ev~ = deval~ X~ ρ ρ~ ρP coh

      eval-domain : ∀ {i} (t : Term₀ A B ar X i) (extt : extTerm _~A_ _~B_ X~ t)
        → let w = eval-QExt X~ (t , extt) ρ ρ~
          in ∃[ d ∈ elim-par (w .₁) (w .₂) .extent ]
            P~ (qwi-refl Arefl (w .₂)) (elim-par (w .₁) (w .₂) .value d) (ev t extt)
      eval-domain (var₀ x) _ = ds x , P~refl _ _ _
      eval-domain (op₀ a f) (f~ , extf) = ((λ b → ih b .₁) , coherent)
        , p~ fr (λ b → ws b .₂) vals coherent
            fr (λ b → ws b .₂) (λ b → ev (f b) (extf b))
            (λ {b} {b'} bb → ev~ (f b) (f b') (extf b) (extf b') (f~ bb))
            (Arefl a) fr
            (λ {b} {b'} bb → P~trans (ih b .₂) (ev~ (f b) (f b') (extf b) (extf b') (f~ bb)))
        where
          ws = λ b → eval-QExt X~ (f b , extf b) ρ ρ~
          fr : ∀ {b b'} → b ~B b' → QWI₀~ (ws b .₁) (ws b' .₁)
          fr {b} {b'} bb = eval~ X~ {t₀ = f b} {t₁ = f b'} (f~ bb) ρ ρ ρ~
          ih = λ b → eval-domain (f b) (extf b)
          vals = λ b → elim-par (ws b .₁) (ws b .₂) .value (ih b .₁)
          coherent : ∀ {b b'} (bb : b ~B b') → P~ (fr bb) (vals b) (vals b')
          coherent {b} {b'} bb = P~trans (ih b .₂)
            (P~trans (ev~ (f b) (f b') (extf b) (extf b') (f~ bb)) (P~sym (ih b' .₂)))

    Sound : ∀ {i j} {w₀ : 𝒲I A B ar i} {w₁ : 𝒲I A B ar j} → QWI₀~ w₀ w₁ → Ext w₀ → Ext w₁ → Prop ℓ
    Sound {w₀ = w₀} {w₁} eq ext₀ ext₁ =
      ℒ↓ (P~ eq) (elim-par w₀ ext₀) (elim-par w₁ ext₁)

    module _ (equ-p : QWI-elim₀-coh) where

      sound : ∀ {i j} (w₀ : 𝒲I A B ar i) (w₁ : 𝒲I A B ar j) (eq : QWI₀~ w₀ w₁) (ext₀ : Ext w₀) (ext₁ : Ext w₁)
        → Sound eq ext₀ ext₁
      sound _ _ (qwi-sym eq) ext₀ ext₁ with sound _ _ eq ext₁ ext₀
      ... | related d₁ d₀ eqP = related d₀ d₁ (P~sym eqP)
      sound _ _ (qwi-trans mid eq₀ eq₁) ext₀ ext₂
        with sound _ _ eq₀ ext₀ mid | sound _ _ eq₁ mid ext₂
      ... | related d₀ _ eqP₀ | related _ d₂ eqP₁ = related d₀ d₂ (P~trans eqP₀ eqP₁)
      sound _ _ (qwi-sup {f₀ = f₀} {f₁} aa ff) (f₀~ , ext₀) (f₁~ , ext₁) =
        related (ds₀ , coh₀) (ds₁ , coh₁)
          (p~ f₀~ ext₀ (partial-values (λ b → elim-par (f₀ b) (ext₀ b)) ds₀) coh₀
            f₁~ ext₁ (partial-values (λ b → elim-par (f₁ b) (ext₁ b)) ds₁) coh₁
            aa ff (λ bb → cross bb .val~))
        where
          cross : ∀ {b₀ b₁} (bb : b₀ ~B b₁) → Sound (ff bb) (ext₀ b₀) (ext₁ b₁)
          cross {b₀} {b₁} bb = sound _ _ (ff bb) (ext₀ b₀) (ext₁ b₁)

          ds₀ = λ b → cross (Bcoh aa b) .def₀
          ds₁ = λ b → cross (Bsym (Bcoh (Asym aa) b)) .def₁
          coh₀ = λ {b₀} {b₁} (bb : b₀ ~B b₁) →
            P~trans (cross (Btrans bb (Bcoh aa b₁)) .val~)
              (P~sym (cross (Bcoh aa b₁) .val~))
          coh₁ = λ {b₀} {b₁} (bb : b₀ ~B b₁) →
            P~trans (P~sym (cross (Bsym (Bcoh (Asym aa) b₀)) .val~))
              (cross (Btrans (Bsym (Bcoh (Asym aa) b₀)) bb) .val~)
      sound _ _ (qwi-equ e ρ ρ~) ext₀ ext₁ = related (l .₁) (r .₁)
        (P~trans (l .₂)
          (P~trans (equ-p e ρ ρ~ vals coherent) (P~sym (r .₂))))
        where
          defined : ∀ {i} (x : V ε e i) → elim-par (ρ x .₁) (ρ x .₂) .extent
          defined x = sound _ _ (ρ~ (Vrefl e x)) (ρ x .₂) (ρ x .₂) .def₀
          vals : ∀ {i} (x : V ε e i) → P (ρ x .₁) (ρ x .₂)
          vals x = elim-par (ρ x .₁) (ρ x .₂) .value (defined x)
          coherent : ∀ {i j} {x : V ε e i} {y : V ε e j} (xy : V~ ε e x y) → P~ (ρ~ xy) (vals x) (vals y)
          coherent {x = x} {y} xy = sound _ _ (ρ~ xy) (ρ x .₂) (ρ y .₂) .val~
          l = eval-domain (V~ ε e) ρ ρ~ defined coherent (lhs ε e) (lhs-ext e)
          r = eval-domain (V~ ε e) ρ ρ~ defined coherent (rhs ε e) (rhs-ext e)

      elim-QWI₀ : ∀ {i} (w : 𝒲I A B ar i) (extw : Ext w) → P w extw
      elim-QWI₀ w extw = elim-par w extw .value (sound w w (qwi-refl Arefl extw) extw extw .def₀)

      elim-QWI₀~
        : ∀ {i j} {w₀ : 𝒲I A B ar i} {w₁ : 𝒲I A B ar j} (w₀₁ : QWI₀~ w₀ w₁)
        → (extw₀ : Ext w₀) (extw₁ : Ext w₁)
        → P~ w₀₁ (elim-QWI₀ w₀ extw₀) (elim-QWI₀ w₁ extw₁)
      elim-QWI₀~ w₀₁ extw₀ extw₁ = sound _ _ w₀₁ extw₀ extw₁ .val~
