module _
  {A : Set} (_~A_ : A → A → Prop)
  (~A-refl : ∀ x → x ~A x) (~A-sym : ∀ {x₀ x₁} → x₀ ~A x₁ → x₁ ~A x₀) (~A-trans : ∀ {x₀ x₁ x₂} → x₀ ~A x₁ → x₁ ~A x₂ → x₀ ~A x₂)
  {B : A → Set} (_~B_ : ∀ {x₀ x₁} → B x₀ → B x₁ → Prop)
  (Bsym : ∀ {x₀ x₁} {y₀ : B x₀} {y₁ : B x₁} → y₁ ~B y₀ → y₀ ~B y₁)
  (Bcoe₀₁ : ∀ {x₀ x₁ : A} → x₀ ~A x₁ → B x₀ → B x₁)
  (Bcoh₀₁ : ∀ {x₀ x₁ : A} → (x₀₁ : x₀ ~A x₁) (y : B x₀) → y ~B Bcoe₀₁ x₀₁ y)
  (Btrans : ∀ {x₀ x₁ x₂ : A} (x₀₁ : x₀ ~A x₁) (x₁₂ : x₁ ~A x₂) → {y₀ : B x₀} {y₂ : B x₂}
    → y₀ ~B y₂ → Bcoe₀₁ x₀₁ y₀ ~B y₂)

  open W-rel _~A_ B B _~B_

  module _
    (P  : ∀ t → extW _~A_ _~B_ t → Set ℓ)
    (P~ : ∀ {t₀ t₁} → 𝒲~ t₀ t₁ → ∀ {pt₀ pt₁} → P t₀ pt₀ → P t₁ pt₁ → Prop ℓ)
    (P~refl
      : ∀ (t : 𝒲 A B) (ext : extW _~A_ _~B_ t) x → P~ (𝒲~-refl _~A_ _~B_ ~A-refl t ext) {ext} {ext} x x)
    (~sym
      : ∀ {t₀ t₁} {ext₀ : extW _~A_ _~B_ t₀} {ext₁ : extW _~A_ _~B_ t₁}
        {v₀ : P t₀ ext₀} {v₁ : P t₁ ext₁} {t₀₁ : 𝒲~ t₀ t₁}
      → P~ t₀₁ v₀ v₁ → P~ (𝒲~-sym _ _ _ _ ~A-sym _ _ _ _ Bsym t₀ t₁ t₀₁) v₁ v₀ )
    (P~-trans
      : ∀ {t₀ t₁ t₂} {ext₀ : extW _ _ t₀} {ext₁ : extW _ _ t₁} {ext₂ : extW _ _ t₂}
      → {v₀ : P t₀ ext₀} {v₁ : P t₁ ext₁} {v₂ : P t₂ ext₂}
      → {t₀₁ : 𝒲~ t₀ t₁} {t₁₂ : 𝒲~ t₁ t₂}
      → P~ t₀₁ v₀ v₁ → P~ t₁₂ v₁ v₂
      → P~ (𝒲~-trans
        _ _ _ _ _ _ _ _ _ _ _ _ ~A-trans Bcoe₀₁ Bcoh₀₁ Btrans t₀₁ t₁₂) v₀ v₂)
    (p : {x : A} {f : B x → 𝒲 A B}
      → (extf : (y : B x) → extW _~A_ _~B_ (f y))
      → (f~ : {y₀ y₁ : B x} (y₀₁ : y₀ ~B y₁) → 𝒲~ (f y₀) (f y₁))
      → (extp : (y : B x) → P _ (extf y))
      → (ps~ : {y₀ y₁ : B x} (y₀₁ : y₀ ~B y₁) → P~ (f~ y₀₁) (extp y₀) (extp y₁))
      → P (sup₀ x f) (f~ , extf))
    (p~
      : {x₀ : A} {f₀ : B x₀ → 𝒲 A B}
      → (f₀~ : {y₀ y₀' : B x₀} → y₀ ~B y₀' → 𝒲~ (f₀ y₀) (f₀ y₀'))
      → (extf₀ : (y₀ : B x₀) → extW _~A_ _~B_ (f₀ y₀))
      → (extp₀ : (y₀ : B x₀) → P _ (extf₀ y₀))
      → (p₀~ : {y₀ y₀' : B x₀} → (y₀₀' : y₀ ~B y₀') → P~ (f₀~ y₀₀') (extp₀ y₀) (extp₀ y₀'))

      → {x₁ : A} {f₁ : B x₁ → 𝒲 A B}
      → (f₁~ : {y₁ y₁' : B x₁} → y₁ ~B y₁' → 𝒲~ (f₁ y₁) (f₁ y₁'))
      → (extf₁ : (y₁ : B x₁) → extW _~A_ _~B_ (f₁ y₁))
      → (extp₁ : (y₁ : B x₁) → P _ (extf₁ y₁))
      → (p₁~ : {y₁ y₁' : B x₁} → (y₁₁' : y₁ ~B y₁') → P~ (f₁~ y₁₁') (extp₁ y₁) (extp₁ y₁'))

      → (x₀₁ : x₀ ~A x₁)
      → (f₀₁ : ∀ {y₀ y₁} (y₀₁ : y₀ ~B y₁) → 𝒲~ (f₀ y₀) (f₁ y₁))
      → (p₀₁ : ∀ {y₀ y₁} (y₀₁ : y₀ ~B y₁) → P~ (f₀₁ y₀₁) (extp₀ y₀) (extp₁ y₁))
      → P~ (x₀₁ , f₀₁) (p extf₀ f₀~ extp₀ p₀~) (p extf₁ f₁~ extp₁ p₁~))

    isElim : (t : 𝒲 A B) (ext : extW _~A_ _~B_ t) (v : P t ext) → Set ℓ
    isElim (sup₀ x f) (f~ , extf) v =
      Σₚ[ |elim| ∈ ((y : B x) → Σ (P (f y) (extf y)) (isElim _ _)) ]
        ∃[ elim~ ∈ (∀ {y₀ y₁ : B x} (y₀₁ : y₀ ~B y₁) → P~ (f~ y₀₁) (|elim| y₀ .₁) (|elim| y₁ .₁)) ]
          P~ (~A-refl x , f~) (p extf f~ (λ y → |elim| y .₁) elim~) v

    uniq
      : (t₀ t₁ : 𝒲 A B)(ext₀ : extW _~A_ _~B_ t₀)(ext₁ : extW _~A_ _~B_ t₁)(v₀ : P t₀ ext₀)(v₁ : P t₁ ext₁)
      → (pv₀ : isElim t₀ ext₀ v₀) (pv₁ : isElim t₁ ext₁ v₁)
      → (t₀₁ : 𝒲~ t₀ t₁)
      → P~ t₀₁ v₀ v₁
    uniq (sup₀ x₀ f₀) (sup₀ x₁ f₁) (f₀~ , extf₀) (f₁~ , extf₁) v₀ v₁ (|elim|₀ , elim~₀ , p₀) (|elim|₁ , elim~₁ , p₁) (x₀₁ , f₀₁) =
      P~-trans (~sym p₀) (P~-trans
        (p~ f₀~ extf₀ (λ y → |elim|₀ y .₁) elim~₀ f₁~ extf₁ (λ y → |elim|₁ y .₁) elim~₁ x₀₁ f₀₁
          λ y₀₁ → uniq _ _ _ _ _ _ (|elim|₀ _ .₂) (|elim|₁ _ .₂) (f₀₁ y₀₁)) p₁)

    totality
      : (t : 𝒲 A B)(ext : extW _~A_ _~B_ t) → Σ (P t ext) (isElim t ext)
    totality (sup₀ x f) (f~ , extf) = p extf f~ (λ y → elim y .₁) elim~ , elim , elim~ , P~refl _ _ _
      where
        elim : (y : B x) → Σ (P (f y) (extf y)) (isElim (f y) (extf y))
        elim y = totality (f y) (extf y)

        elim~ : {y₀ y₁ : B x} (y₀₁ : y₀ ~B y₁)
          → P~ (f~ y₀₁) (elim y₀ .₁) (elim y₁ .₁)
        elim~ {y₀} {y₁} y₀₁ = uniq _ _ _ _ _ _ (elim y₀ .₂) (elim y₁ .₂) (f~ y₀₁)
